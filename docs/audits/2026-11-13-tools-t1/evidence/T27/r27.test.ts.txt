// Closed polygon as an imported SVG in scene mm, identity transform.
function poly(id: string, op: string, rings: P[][]): ImportedSvg {
  const xs = rings.flat().map((p) => p.x), ys = rings.flat().map((p) => p.y);
  return {
    kind: 'imported-svg', id, source: `${id}.svg`,
    bounds: { minX: Math.min(...xs), minY: Math.min(...ys), maxX: Math.max(...xs), maxY: Math.max(...ys) },
    transform: IDENTITY_TRANSFORM, operationIds: [op],
    paths: [{ color: '#000000', polylines: rings.map((r) => ({ closed: true, points: [...r, r[0]!] })) }],
  };
}
const sq = (x: number, y: number, s: number): P[] => [{ x, y }, { x: x + s, y }, { x: x + s, y: y + s }, { x, y: y + s }];
// Ring drawn the other way round, to make sure winding does not decide the side.
const sqRev = (x: number, y: number, s: number): P[] => sq(x, y, s).reverse();

function op(id: string, mode: 'line' | 'fill', extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id, color: '#000000', name: id, mode }), power: 50, speed: 1000, passes: 1, ...extra } as Layer;
}
function proj(objects: SceneObject[], layers: Layer[]): Project {
  const base = createProject();
  return { ...base, device: { ...base.device, origin: 'rear-left' }, scene: { ...base.scene, objects, layers, groups: [] } };
}

// Own G-code decoder: segments burned while M3/M4 is on with S>0 under G1 (G2/G3 flagged).
function decode(gcode: string): { segs: Seg[]; arcs: number } {
  let x = 0, y = 0, s = 0, mode = 0, on = false, arcs = 0;
  const segs: Seg[] = [];
  for (const raw of gcode.split(/\r?\n/)) {
    const l = raw.split(';')[0]!.toUpperCase();
    const w = new Map<string, number>();
    for (const m of l.matchAll(/([GMXYSF])(-?[0-9.]+)/g)) {
      const k = m[1]!, v = Number(m[2]);
      if (k === 'G') { if ([0, 1, 2, 3].includes(v)) mode = v; }
      else if (k === 'M') { if (v === 3 || v === 4) on = true; if (v === 5) on = false; }
      else w.set(k, v);
    }
    if (w.has('S')) s = w.get('S')!;
    const nx = w.get('X') ?? x, ny = w.get('Y') ?? y;
    if ((w.has('X') || w.has('Y')) && on && s > 0 && mode !== 0) {
      if (mode !== 1) arcs++;
      segs.push([{ x, y }, { x: nx, y: ny }]);
    }
    x = nx; y = ny;
  }
  return { segs, arcs };
}

async function generated(p: Project, sel: string[]): Promise<{ segs: Seg[]; toast: string }> {
  useToastStore.setState({ toasts: [] } as never);
  useStore.setState({ project: p, selectedObjectId: sel[0] ?? null, additionalSelectedIds: new Set(sel.slice(1)), undoStack: [], redoStack: [], dirty: false });
  await useStore.getState().generatePathFromCut();
  const toast = useToastStore.getState().toasts.map((t) => t.message).join(' | ');
  const objs = useStore.getState().project.scene.objects;
  if (objs.length === p.scene.objects.length) return { segs: [], toast };
  const g = objs[objs.length - 1] as ImportedSvg;
  const segs: Seg[] = [];
  for (const path of g.paths) for (const pl of path.polylines) for (let i = 1; i < pl.points.length; i++) segs.push([pl.points[i - 1]!, pl.points[i]!]);
  return { segs, toast };
}

function burned(p: Project, sel: string[]): Seg[] {
  const prepared = prepareOutput(p, sel.length ? { outputScope: { cutSelectedGraphics: true, useSelectionOrigin: false, selectedObjectIds: sel } } : {});
  if (!prepared.ok) throw new Error(prepared.preflight.issues.map((i) => i.message).join(' '));
  return decode(emitPreparedGcode(prepared).gcode).segs;
}

const len = (ss: Seg[]) => ss.reduce((a, [p, q]) => a + Math.hypot(q.x - p.x, q.y - p.y), 0);
function box(ss: Seg[]) {
  const pts = ss.flat();
  if (!pts.length) return 'empty';
  const f = (n: number) => n.toFixed(3);
  return `x ${f(Math.min(...pts.map((p) => p.x)))}..${f(Math.max(...pts.map((p) => p.x)))} y ${f(Math.min(...pts.map((p) => p.y)))}..${f(Math.max(...pts.map((p) => p.y)))}`;
}
// Segments within a window (to isolate the hole's burn from the plate's).
const within = (ss: Seg[], x0: number, y0: number, x1: number, y1: number) =>
  ss.filter(([p, q]) => [p, q].every((r) => r.x > x0 && r.x < x1 && r.y > y0 && r.y < y1));
// Max distance from any vertex of a to the nearest segment of b, both ways.
function haus(a: Seg[], b: Seg[]): number {
  const d = (p: P, [u, v]: Seg) => {
    const dx = v.x - u.x, dy = v.y - u.y, l2 = dx * dx + dy * dy;
    const t = l2 ? Math.max(0, Math.min(1, ((p.x - u.x) * dx + (p.y - u.y) * dy) / l2)) : 0;
    return Math.hypot(p.x - u.x - t * dx, p.y - u.y - t * dy);
  };
  const one = (x: Seg[], y: Seg[]) => Math.max(0, ...x.flat().map((p) => Math.min(...y.map((s) => d(p, s)))));
  if (!a.length || !b.length) return NaN;
  return Math.max(one(a, b), one(b, a));
}

async function row(name: string, p: Project, sel: string[], expect: string, win?: [number, number, number, number]) {
  const g = await generated(p, sel);
  let b: Seg[] = [];
  let err = '';
  try { b = burned(p, sel); } catch (e) { err = String(e); }
  const gw = win ? within(g.segs, ...win) : g.segs;
  const bw = win ? within(b, ...win) : b;
  say(`${name}\n  sel=[${sel.join(',')}] expect: ${expect}\n  GEN  len ${len(gw).toFixed(3)} ${box(gw)}\n  BURN len ${len(bw).toFixed(3)} ${box(bw)} ${err}\n  hausdorff ${haus(gw, bw).toFixed(3)}  toast: ${g.toast}`);
}
