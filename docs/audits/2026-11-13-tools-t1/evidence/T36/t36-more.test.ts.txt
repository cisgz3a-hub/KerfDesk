type V = { x: number; y: number };
const sq = (x: number, y: number, w: number, h: number): V[] => [
  { x, y }, { x: x + w, y }, { x: x + w, y: y + h }, { x, y: y + h },
];
function reverseOf(objects: SceneObject[], layers: Layer[], sel: string[], extra: Partial<Project> = {}) {
  const p0 = proj(objects, layers, extra);
  setProject(p0, sel);
  st().reverseSelectedPaths();
  return { before: p0, after: st().project, toast: toasts().join(' // ') };
}
const flat = (c: CurveSubpath) => {
  const r = flattenCurveSubpath(c, { toleranceMm: 0.001 });
  if (r.kind !== 'ok') throw new Error('flatten failed');
  return r.polyline.points;
};
// Max distance from each point of A to polyline B (Hausdorff one-way, via segments).
function hd(a: ReadonlyArray<V>, b: ReadonlyArray<V>): number {
  let worst = 0;
  for (const p of a) {
    let best = Infinity;
    for (let i = 0; i + 1 < b.length; i++) {
      const s = b[i]!, e = b[i + 1]!;
      const dx = e.x - s.x, dy = e.y - s.y;
      const L = dx * dx + dy * dy;
      const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - s.x) * dx + (p.y - s.y) * dy) / L));
      best = Math.min(best, Math.hypot(s.x + t * dx - p.x, s.y + t * dy - p.y));
    }
    worst = Math.max(worst, best);
  }
  return worst;
}
// Position along a polyline at arc fraction t, for checking point order.
function arcLenParam(pts: ReadonlyArray<V>, p: V): number {
  let acc = 0, best = Infinity, at = 0;
  const total = pts.slice(1).reduce((s, q, i) => s + Math.hypot(q.x - pts[i]!.x, q.y - pts[i]!.y), 0);
  for (let i = 0; i + 1 < pts.length; i++) {
    const s = pts[i]!, e = pts[i + 1]!;
    const dx = e.x - s.x, dy = e.y - s.y;
    const L = Math.hypot(dx, dy);
    const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - s.x) * dx + (p.y - s.y) * dy) / (L * L)));
    const d = Math.hypot(s.x + t * dx - p.x, s.y + t * dy - p.y);
    if (d < best) { best = d; at = acc + t * L; }
    acc += L;
  }
  return at / total;
}

// Laser tabs: find lit gaps along the contour (where the lit run breaks).
function gaps(moves: LitMove[], ring: V[] = sq(10, 10, 40, 20)): string {
  // Sample the ring every 0.05 mm; a sample not on any lit move is uncut. Report uncut intervals.
  const res: string[] = [];
  let inGap = false; let gs: V | null = null; let prev: V | null = null;
  const on = (p: V) => moves.some((m) => { const dx = m.to.x - m.from.x, dy = m.to.y - m.from.y; const L = dx * dx + dy * dy; const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - m.from.x) * dx + (p.y - m.from.y) * dy) / L)); return Math.hypot(m.from.x + t * dx - p.x, m.from.y + t * dy - p.y) < 1e-3; });
  for (let i = 0; i < ring.length; i++) {
    const a = ring[i]!, b = ring[(i + 1) % ring.length]!;
    const L = Math.hypot(b.x - a.x, b.y - a.y);
    const n = Math.round(L / 0.05);
    for (let k = 0; k < n; k++) {
      const p = { x: a.x + ((b.x - a.x) * k) / n, y: a.y + ((b.y - a.y) * k) / n };
      const lit = on(p);
      if (!lit && !inGap) { inGap = true; gs = p; }
      if (lit && inGap) { inGap = false; res.push(`(${f3(gs!.x)},${f3(gs!.y)})-(${f3(prev!.x)},${f3(prev!.y)})`); }
      prev = p;
    }
  }
  if (inGap) res.push(`(${f3(gs!.x)},${f3(gs!.y)})-end`);
  return res.sort().join(' ');
}
export function gapsOld(moves: LitMove[]): string {
  const g: string[] = [];
  for (let i = 0; i + 1 < moves.length; i++) {
    const a = moves[i]!, b = moves[i + 1]!;
    const d = Math.hypot(a.to.x - b.from.x, a.to.y - b.from.y);
    if (d > 1e-3) g.push(`(${f3((a.to.x + b.from.x) / 2)},${f3((a.to.y + b.from.y) / 2)})`);
  }
  return g.sort().join(' ');
}

function cncScene(objects: SceneObject[], cnc: Record<string, unknown>) {
  const L = { ...createLayer({ id: 'L1', color: '#ff0000' }), cnc: { ...DEFAULT_CNC_LAYER_SETTINGS, depthMm: 6, depthPerPassMm: 3, profileLead: { shape: 'none' }, ...cnc } } as Layer;
  return { layers: [L], objects };
}
function cncDesc(scene: { layers: Layer[]; objects: SceneObject[] }) {
  const job = compileCncJob(scene as never, DEFAULT_DEVICE_PROFILE, DEFAULT_CNC_MACHINE_CONFIG);
  const g = job.groups[0];
  if (g?.kind !== 'cnc') return `no cnc group ${JSON.stringify(job.groups.map((x) => x.kind))}`;
  return g.passes
    .map((p) => {
      if (p.kind === 'contour') {
        const pts = p.polyline;
        let a = 0;
        for (let i = 0; i < pts.length; i++) {
          const s = pts[i]!, e = pts[(i + 1) % pts.length]!;
          a += s.x * e.y - e.x * s.y;
        }
        return `contour z${f3(p.depthMm ?? (p as unknown as { z: number }).z ?? 0)} start ${f3(pts[0]!.x)},${f3(pts[0]!.y)} second ${f3(pts[1]!.x)},${f3(pts[1]!.y)} area ${f3(a / 2)}`;
      }
      if (p.kind === 'path3d') {
        const deep = Math.min(...p.points.map((q) => q.z));
        const rises: string[] = [];
        for (let i = 1; i < p.points.length; i++)
          if (p.points[i]!.z > p.points[i - 1]!.z + 1e-9 && Math.abs(p.points[i - 1]!.z - deep) < 1e-9) rises.push(`${f3(p.points[i]!.x)},${f3(p.points[i]!.y)}`);
        return `path3d deep ${f3(deep)} start ${f3(p.points[0]!.x)},${f3(p.points[0]!.y)} rises@ ${rises.sort().join(' ')}`;
      }
      return p.kind;
    })
    .join(' ; ');
}
