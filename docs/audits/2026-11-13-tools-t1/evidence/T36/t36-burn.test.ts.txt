type V = { x: number; y: number };
const svgObj = (
  id: string,
  pts: V[][],
  closed: boolean,
  t: Partial<Transform> = {},
  extra: Record<string, unknown> = {},
  op = 'L',
): ImportedSvg =>
  ({
    kind: 'imported-svg', id, source: `${id}.svg`, operationIds: [op],
    bounds: { minX: 0, minY: 0, maxX: 100, maxY: 100 }, transform: { ...IDENTITY_TRANSFORM, ...t },
    paths: [{ color: '#000000', polylines: pts.map((points) => ({ points, closed })) }],
    ...extra,
  }) as ImportedSvg;
const sq = (x: number, y: number, w: number, h: number): V[] => [
  { x, y }, { x: x + w, y }, { x: x + w, y: y + h }, { x, y: y + h },
];

function reverseOf(objects: SceneObject[], layers: Layer[], sel: string[], extra: Partial<Project> = {}) {
  const p0 = proj(objects, layers, extra);
  setProject(p0, sel);
  st().reverseSelectedPaths();
  return { before: p0, after: st().project, toast: toasts().join(' // ') };
}
// Lit moves split into continuous runs (a run ends where the next lit move does not start at its end).
function runs(moves: LitMove[]): LitMove[][] {
  const r: LitMove[][] = [];
  for (const m of moves) {
    const cur = r.at(-1);
    const last = cur?.at(-1);
    if (cur && last && Math.hypot(last.to.x - m.from.x, last.to.y - m.from.y) < 1e-3) cur.push(m);
    else r.push([m]);
  }
  return r;
}
const signedArea = (run: LitMove[]) =>
  run.reduce((a, m) => a + (m.from.x * m.to.y - m.to.x * m.from.y), 0) / 2;
const desc = (moves: LitMove[]) =>
  runs(moves)
    .map((r) => `[start ${f3(r[0]!.from.x)},${f3(r[0]!.from.y)} end ${f3(r.at(-1)!.to.x)},${f3(r.at(-1)!.to.y)} len ${f3(totalLen(r))} area ${f3(signedArea(r))} n${r.length}]`)
    .join(' ');
function cmp(tag: string, before: Project, after: Project) {
  const a = burn(before);
  const b = burn(after);
  if (!a.ok || !b.ok) {
    out(`${tag}: NOT OK ${a.why} ${b.why}`);
    return;
  }
  out(`${tag} before: lit ${f3(totalLen(a.moves))} ${desc(a.moves)}`);
  out(`${tag} after : lit ${f3(totalLen(b.moves))} ${desc(b.moves)}`);
  out(`${tag} gcode identical: ${a.gcode === b.gcode}`);
}

  const L = layer('L', 'line', 1000, 50);
  const t = await textObj('t', 'A', { x: 10, y: 10 }, ['L']);
  const s = svgObj('a', [sq(10, 10, 30, 20)], true, { x: 5 }, { name: 'Part A', powerScale: 80, operationOverrides: { L: { speed: 500 } }, libraryAsset: { id: 'x' } });
  const tr = { ...svgObj('tr', [sq(50, 50, 10, 10)], true), kind: 'traced-image', sourceImageId: 'img' } as unknown as SceneObject;
  const r = reverseOf([t, s, tr], [L], ['t', 'a', 'tr']);
  const a = r.after.scene.objects[1] as unknown as Record<string, unknown>;
  const b = r.before.scene.objects[1] as unknown as Record<string, unknown>;
  const diffKeys = Object.keys({ ...a, ...b }).filter((k) => JSON.stringify(a[k]) !== JSON.stringify(b[k]));
  out(`R8 toast ${r.toast}; svg keys changed ${diffKeys.join(',')}; traced kind ${r.after.scene.objects[2]!.kind} changed ${r.after.scene.objects[2] !== r.before.scene.objects[2]}; text same ${r.after.scene.objects[0] === r.before.scene.objects[0]}`);
