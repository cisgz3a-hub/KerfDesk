// T03: tone wedge on a rotary. Patches are columns in machine X (unscaled);
// rows are scan lines in emitted Y (scaled by k).
import { it } from 'vitest';
import { writeFileSync } from 'node:fs';
import { DEFAULT_DEVICE_PROFILE, emitProject, projectForScene } from './harness';
import { generateToneCalibrationWedge } from '../t01-main/src/core/job/tone-calibration-wedge';
import { DEFAULT_TONE_WEDGE_DRAFT, toneWedgeOptions } from '../t01-main/src/ui/calibration/tone-calibration-draft';
import type { DeviceProfile } from '../t01-main/src/core/scene';
import { rotaryYScale } from '../t01-main/src/core/devices/rotary';

type Raw = { x0: number; x1: number; y0: number; y1: number; f: number; s: number };
function rawLit(gcode: string): Raw[] {
  let x = 0, y = 0, s = 0, f = 0, g = 0;
  let on = false;
  const out: Raw[] = [];
  for (const raw of gcode.split('\n')) {
    const line = raw.replace(/;.*$/, '').toUpperCase();
    const w = new Map<string, number>();
    for (const m of line.matchAll(/([A-Z])\s*(-?\d*\.?\d+)/g)) {
      if (m[1] === 'M') {
        const v = Number(m[2]);
        if (v === 3 || v === 4) on = true;
        if (v === 5) on = false;
      } else w.set(m[1]!, Number(m[2]));
    }
    if (w.has('G') && [0, 1, 2, 3].includes(w.get('G')!)) g = w.get('G')!;
    if (w.has('F')) f = w.get('F')!;
    if (w.has('S')) s = w.get('S')!;
    const nx = w.get('X') ?? x;
    const ny = w.get('Y') ?? y;
    if ((w.has('X') || w.has('Y')) && g === 1 && on && s > 0) out.push({ x0: x, x1: nx, y0: y, y1: ny, f, s });
    x = nx;
    y = ny;
  }
  return out;
}

it('rotary wedge coverage', () => {
  const log: string[] = [];
  const setups = [
    { name: 'chuck 20mm/rot 60mm', rotary: { enabled: true, type: 'chuck', mmPerRotation: 20, objectDiameterMm: 60 } },
    { name: 'roller 40mm/rot 80mm obj 20mm roller', rotary: { enabled: true, type: 'roller', mmPerRotation: 40, objectDiameterMm: 80, rollerDiameterMm: 20 } },
  ];
  for (const st of setups) {
    const device = { ...DEFAULT_DEVICE_PROFILE, rotary: st.rotary } as unknown as DeviceProfile;
    const k = rotaryYScale(device.rotary as never);
    const wedge = generateToneCalibrationWedge(toneWedgeOptions(DEFAULT_TONE_WEDGE_DRAFT, null));
    const out = emitProject(projectForScene(wedge.scene, device));
    log.push(`\n## ${st.name} k=${k} ok=${out.ok}`);
    const lit = rawLit(out.gcode).filter((m) => Math.abs(m.y1 - m.y0) < 1e-9 && m.f === 3000 && m.s === 300);
    const ys = [...new Set(lit.map((m) => m.y0.toFixed(4)))].map(Number).sort((a, b) => a - b);
    const steps = ys.slice(1).map((y, i) => y - ys[i]!);
    const meanStep = steps.reduce((a, b) => a + b, 0) / Math.max(1, steps.length);
    log.push(`rows ${ys.length}; emitted Y step mean ${meanStep.toFixed(5)} -> surface ${(meanStep / k).toFixed(5)} mm (want 0.1)`);
    const minX = Math.min(...lit.map((m) => Math.min(m.x0, m.x1)));
    for (const p of wedge.patches) {
      const x0 = minX + p.index * 9 + 1;
      const x1 = x0 + 6;
      let len = 0;
      for (const m of lit) {
        const lo = Math.max(Math.min(m.x0, m.x1), x0);
        const hi = Math.min(Math.max(m.x0, m.x1), x1);
        if (hi > lo) len += hi - lo;
      }
      const cov = len / (6 * ys.length);
      log.push(`patch ${p.index} label=${p.label} commanded=${p.commandedPercent.toFixed(2)} coverage=${(cov * 100).toFixed(2)} err=${(cov * 100 - p.commandedPercent).toFixed(2)}pp`);
    }
  }
  writeFileSync('C:/Users/stolk/kd-t1/zz-t03/t03-rotary.log', log.join('\n'));
});
