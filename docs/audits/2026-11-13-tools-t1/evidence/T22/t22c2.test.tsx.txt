async function conv(all: SceneObject[], sel: SceneObject[], layers: Layer[], renderType: 'fill-all' | 'outlines' | 'use-cut-settings', brightness = 0, extra: object = {}) {
  resetStore();
  useToastStore.setState({ toasts: [] } as never);
  useStore.setState({ project: { ...project(all, layers), ...extra } as never, selectedObjectId: sel[0]!.id });
  const toasts: string[] = [];
  const outcome = await convertSelectedVectorsToBitmap(sel as unknown as ConvertibleVector[], layers, { renderType, dpi: 254, brightnessPercent: brightness }, useStore.getState().convertToBitmap, (m) => toasts.push(m));
  const storeToasts = (useToastStore.getState() as { toasts: { message: string }[] }).toasts.map((t) => t.message);
  return { outcome, toasts: [...toasts, ...storeToasts], project: useStore.getState().project };
}
