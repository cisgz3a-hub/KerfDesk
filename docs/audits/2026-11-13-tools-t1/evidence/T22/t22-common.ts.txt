export const DEV = DEFAULT_DEVICE_PROFILE;
export function logger(path: string): (s: string) => void {
  const lines: string[] = [];
  return (s: string) => {
    lines.push(s);
    writeFileSync(path, lines.join('\n'));
  };
}

export function fillLayer(id: string, color: string, extra: Partial<Layer> = {}): Layer {
  return {
    ...createLayer({ id, color, name: id, mode: 'fill' }),
    power: 50,
    speed: 3000,
    hatchSpacingMm: 0.1,
    hatchAngleDeg: 0,
    ...extra,
  } as Layer;
}
export function lineLayer(id: string, color: string, extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id, color, name: id, mode: 'line' }), power: 50, speed: 1000, ...extra } as Layer;
}

export type PathSpec = { contours: P[][]; fillRule?: 'evenodd' | 'nonzero'; closed?: boolean; color?: string; operationIds?: string[] };
/** shape / imported-svg / text whose polylines are exactly the given local contours. */
export function mkObj(
  id: string,
  kind: 'shape' | 'imported-svg' | 'text',
  paths: PathSpec[],
  transform: Transform = IDENTITY_TRANSFORM,
  operationIds?: string[],
): SceneObject {
  const all = paths.flatMap((p) => p.contours.flat());
  const bounds = {
    minX: Math.min(...all.map((p) => p.x)),
    minY: Math.min(...all.map((p) => p.y)),
    maxX: Math.max(...all.map((p) => p.x)),
    maxY: Math.max(...all.map((p) => p.y)),
  };
  const ps = paths.map((p) => ({
    color: p.color ?? '#000000',
    ...(p.fillRule ? { fillRule: p.fillRule } : {}),
    ...(p.operationIds ? { operationIds: p.operationIds } : {}),
    polylines: p.contours.map((points) => ({ points, closed: p.closed ?? true })),
  }));
  const ops = operationIds ? { operationIds } : {};
  if (kind === 'shape')
    return {
      kind,
      id,
      spec: { kind: 'polyline', points: paths[0]!.contours[0], closed: paths[0]!.closed ?? true },
      color: '#000000',
      bounds,
      transform,
      paths: ps,
      ...ops,
    } as unknown as SceneObject;
  if (kind === 'imported-svg')
    return { kind, id, source: `${id}.svg`, bounds, transform, paths: ps, ...ops } as unknown as SceneObject;
  return {
    kind: 'text', id, content: 'AB', fontKey: 'x', sizeMm: 10, alignment: 'left', lineHeight: 1,
    letterSpacing: -0.3, color: '#000000', bounds, transform, paths: ps, ...ops,
  } as unknown as SceneObject;
}
export function project(objects: SceneObject[], layers: Layer[]): Project {
  const base = createProject(DEV);
  return { ...base, scene: { ...base.scene, objects, layers, groups: [] } } as Project;
}
export function gcodeOf(p: Project): string {
  const r = prepareOutput(p);
  if (!r.ok) return `NOT OK ${JSON.stringify(r).slice(0, 800)}`;
  return emitPreparedGcode(r).gcode;
}

// ---------- decoder: every G1 move with S>0, in scene coordinates (front-left origin) ----------
export type Seg = { a: P; b: P; s: number };
export function burnedSegments(g: string, bedH = DEV.bedHeight): Seg[] {
  const out: Seg[] = [];
  let x = 0, y = 0, S = 0, mode = 0, laserOn = false;
  for (const raw of g.split('\n')) {
    const line = raw.replace(/;.*$/, '').trim();
    if (!line) continue;
    if (/\bM[34]\b/.test(line)) laserOn = true;
    if (/\bM5\b/.test(line)) laserOn = false;
    const m = line.match(/^G0?([0-3])(?![0-9])/);
    if (m) mode = Number(m[1]);
    const nx = line.match(/X(-?[\d.]+)/), ny = line.match(/Y(-?[\d.]+)/), ns = line.match(/S(-?[\d.]+)/);
    if (ns) S = Number(ns[1]);
    const x1 = nx ? Number(nx[1]) : x;
    const y1 = ny ? Number(ny[1]) : y;
    if (mode === 1 && S > 0 && laserOn && (nx || ny)) out.push({ a: { x, y: bedH - y }, b: { x: x1, y: bedH - y1 }, s: S });
    x = x1;
    y = y1;
  }
  return out;
}

// ---------- independent geometry ----------
export function xf(p: P, t: Transform): P {
  let x = p.x * t.scaleX, y = p.y * t.scaleY;
  if (t.mirrorX) x = -x;
  if (t.mirrorY) y = -y;
  const r = (t.rotationDeg * Math.PI) / 180;
  return { x: x * Math.cos(r) - y * Math.sin(r) + t.x, y: x * Math.sin(r) + y * Math.cos(r) + t.y };
}
export function winding(p: P, contours: P[][]): number {
  let w = 0;
  for (const c of contours)
    for (let i = 0; i < c.length; i++) {
      const a = c[i]!, b = c[(i + 1) % c.length]!;
      if (a.y <= p.y) {
        if (b.y > p.y && cross(a, b, p) > 0) w++;
      } else if (b.y <= p.y && cross(a, b, p) < 0) w--;
    }
  return w;
}
const cross = (a: P, b: P, p: P) => (b.x - a.x) * (p.y - a.y) - (p.x - a.x) * (b.y - a.y);
function segDist(p: P, a: P, b: P): number {
  const dx = b.x - a.x, dy = b.y - a.y;
  const L = dx * dx + dy * dy;
  const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / L));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}
export function edgeDist(p: P, contours: P[][]): number {
  let d = Infinity;
  for (const c of contours) for (let i = 0; i < c.length; i++) d = Math.min(d, segDist(p, c[i]!, c[(i + 1) % c.length]!));
  return d;
}
export function circle(cx: number, cy: number, rx: number, ry = rx, n = 96, start = 0.013): P[] {
  return Array.from({ length: n }, (_, i) => {
    const a = start + (i / n) * 2 * Math.PI;
    return { x: cx + rx * Math.cos(a), y: cy + ry * Math.sin(a) };
  });
}
export const rect = (x0: number, y0: number, x1: number, y1: number): P[] => [
  { x: x0, y: y0 }, { x: x1, y: y0 }, { x: x1, y: y1 }, { x: x0, y: y1 },
];
export const rectCW = (x0: number, y0: number, x1: number, y1: number): P[] => rect(x0, y0, x1, y1).reverse();
export function star(cx: number, cy: number, r: number): P[] {
  return [0, 2, 4, 1, 3].map((k) => {
    const a = -Math.PI / 2 + (k * 2 * Math.PI) / 5;
    return { x: cx + r * Math.cos(a), y: cy + r * Math.sin(a) };
  });
}

/** Oracle object: world contours per path with its rule. Inside = union of paths; objects combine by `combine`. */
export type OracleObj = { paths: { contours: P[][]; rule: 'evenodd' | 'nonzero' }[] };
export function insideObj(p: P, o: OracleObj): boolean {
  return o.paths.some((path) => {
    const w = winding(p, path.contours);
    return path.rule === 'nonzero' ? w !== 0 : w % 2 !== 0;
  });
}
export function insideAll(p: P, objs: OracleObj[], combine: 'evenodd' | 'union'): boolean {
  let n = 0;
  for (const o of objs) if (insideObj(p, o)) n++;
  return combine === 'union' ? n > 0 : n % 2 === 1;
}

export function compare(
  segs: Seg[],
  pitch: number,
  inside: (p: P) => boolean,
  edges: P[][],
  region: { x0: number; y0: number; x1: number; y1: number },
  g = 0.05,
) {
  const nx = Math.round((region.x1 - region.x0) / g), ny = Math.round((region.y1 - region.y0) / g);
  const burned = new Uint8Array(nx * ny);
  const half = pitch / 2;
  for (const s of segs) {
    const dx = s.b.x - s.a.x, dy = s.b.y - s.a.y;
    const L = Math.hypot(dx, dy);
    if (L < 1e-9) continue;
    const ux = dx / L, uy = dy / L;
    const minX = Math.min(s.a.x, s.b.x) - half, maxX = Math.max(s.a.x, s.b.x) + half;
    const minY = Math.min(s.a.y, s.b.y) - half, maxY = Math.max(s.a.y, s.b.y) + half;
    const i0 = Math.max(0, Math.floor((minX - region.x0) / g)), i1 = Math.min(nx - 1, Math.ceil((maxX - region.x0) / g));
    const j0 = Math.max(0, Math.floor((minY - region.y0) / g)), j1 = Math.min(ny - 1, Math.ceil((maxY - region.y0) / g));
    for (let j = j0; j <= j1; j++)
      for (let i = i0; i <= i1; i++) {
        const px = region.x0 + (i + 0.5) * g - s.a.x, py = region.y0 + (j + 0.5) * g - s.a.y;
        const along = px * ux + py * uy, perp = Math.abs(-px * uy + py * ux);
        if (along >= 0 && along <= L && perp <= half) burned[j * nx + i] = 1;
      }
  }
  let leak = 0, miss = 0, oracleArea = 0, burnArea = 0, maxLeak = 0, maxMiss = 0, leakFar = 0, missFar = 0;
  let farLeakAt: P | null = null, farMissAt: P | null = null;
  for (let j = 0; j < ny; j++)
    for (let i = 0; i < nx; i++) {
      const p = { x: region.x0 + (i + 0.5) * g, y: region.y0 + (j + 0.5) * g };
      const o = inside(p);
      const b = burned[j * nx + i] === 1;
      if (o) oracleArea++;
      if (b) burnArea++;
      if (o === b) continue;
      const d = edgeDist(p, edges);
      if (b) {
        leak++;
        if (d > maxLeak) { maxLeak = d; farLeakAt = p; }
        if (d > 2 * pitch) leakFar++;
      } else {
        miss++;
        if (d > maxMiss) { maxMiss = d; farMissAt = p; }
        if (d > 2 * pitch) missFar++;
      }
    }
  const a = g * g;
  const r3 = (v: number) => Math.round(v * 1000) / 1000;
  return {
    oracleMm2: r3(oracleArea * a), burnMm2: r3(burnArea * a), leakMm2: r3(leak * a), missMm2: r3(miss * a),
    maxLeakMm: r3(maxLeak), maxMissMm: r3(maxMiss), leakFarMm2: r3(leakFar * a), missFarMm2: r3(missFar * a),
    farLeakAt: farLeakAt && { x: r3(farLeakAt.x), y: r3(farLeakAt.y) },
    farMissAt: farMissAt && { x: r3(farMissAt.x), y: r3(farMissAt.y) },
  };
}
export function segBounds(segs: Seg[]) {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  for (const s of segs) for (const p of [s.a, s.b]) {
    minX = Math.min(minX, p.x); maxX = Math.max(maxX, p.x); minY = Math.min(minY, p.y); maxY = Math.max(maxY, p.y);
  }
  return { minX, minY, maxX, maxY };
}
