const log = logger('C:/Users/stolk/kd-t1/zz-t22/t22a.log');
const T = (t: Partial<Transform>): Transform => ({ ...IDENTITY_TRANSFORM, ...t });

type Case = {
  name: string;
  objs: { id: string; kind: 'shape' | 'imported-svg' | 'text'; paths: PathSpec[]; t?: Transform; ops?: string[] }[];
  layers?: Layer[];
  renderType?: 'fill-all' | 'outlines' | 'use-cut-settings';
  dpi?: number;
  combine?: 'evenodd' | 'union';
  fillCombine?: 'evenodd' | 'union';
};

function oracleObjs(c: Case): { objs: OracleObj[]; edges: P[][] } {
  const edges: P[][] = [];
  const objs = c.objs.map((o) => ({
    paths: o.paths
      .filter((p) => p.closed !== false)
      .map((p) => {
        const contours = p.contours.map((cc) => cc.map((q) => xf(q, o.t ?? IDENTITY_TRANSFORM)));
        edges.push(...contours);
        return { contours, rule: p.fillRule ?? (o.kind === 'text' ? 'nonzero' : 'evenodd') } as const;
      }),
  }));
  return { objs, edges };
}

async function runCase(c: Case) {
  const layers = c.layers ?? [fillLayer('F', '#000000')];
  const objects: SceneObject[] = c.objs.map((o) => mkObj(o.id, o.kind, o.paths, o.t, o.ops ?? [layers[0]!.id]));
  // Fill burn of the originals.
  const fillG = gcodeOf(project(objects, layers));
  const fillSegs = burnedSegments(fillG);
  // Convert.
  resetStore();
  useStore.setState({ project: project(objects, layers), selectedObjectId: objects[0]!.id });
  const toasts: string[] = [];
  const outcome = await convertSelectedVectorsToBitmap(
    objects as unknown as ConvertibleVector[],
    layers,
    { renderType: c.renderType ?? 'fill-all', dpi: c.dpi ?? 254, brightnessPercent: 0 },
    useStore.getState().convertToBitmap,
    (m) => toasts.push(m),
  );
  const st = useStore.getState();
  const img = st.project.scene.objects.find((o) => o.kind === 'raster-image') as RasterImage | undefined;
  const imgLayer = st.project.scene.layers.find((l) => l.mode === 'image');
  log(`\n## ${c.name}\n  outcome ${JSON.stringify(outcome)} toasts ${JSON.stringify(toasts)}`);
  if (!img || !imgLayer) { log('  NO IMAGE'); return; }
  log(`  image ${img.pixelWidth}x${img.pixelHeight}px bounds ${JSON.stringify(img.bounds)} linesPerMm ${img.linesPerMm}; layer ${imgLayer.name} power ${imgLayer.power} speed ${imgLayer.speed} lpm ${imgLayer.linesPerMm} dither ${imgLayer.ditherAlgorithm} objects ${st.project.scene.objects.length} layers ${st.project.scene.layers.map((l) => `${l.name}:${l.mode}`).join(',')}`);
  const g = gcodeOf(st.project);
  if (g.startsWith('NOT OK')) { log('  ' + g); return; }
  const segs = burnedSegments(g);
  const { objs, edges } = oracleObjs(c);
  const all = edges.flat();
  const region = {
    x0: Math.floor(Math.min(...all.map((p) => p.x))) - 2, y0: Math.floor(Math.min(...all.map((p) => p.y))) - 2,
    x1: Math.ceil(Math.max(...all.map((p) => p.x))) + 2, y1: Math.ceil(Math.max(...all.map((p) => p.y))) + 2,
  };
  const pitch = 1 / (imgLayer.linesPerMm ?? 10);
  const bmp = compare(segs, pitch, (p) => insideAll(p, objs, c.combine ?? 'evenodd'), edges, region);
  log(`  bitmap segs ${segs.length} burnBounds ${JSON.stringify(segBounds(segs))}`);
  log(`  bitmap vs oracle(${c.combine ?? 'evenodd'} across objects) ${JSON.stringify(bmp)}`);
  if (!fillG.startsWith('NOT OK')) {
    const fill = compare(fillSegs, 0.1, (p) => insideAll(p, objs, c.fillCombine ?? c.combine ?? 'evenodd'), edges, region);
    log(`  Fill    vs oracle(${c.fillCombine ?? c.combine ?? 'evenodd'}) ${JSON.stringify(fill)}`);
  } else log('  Fill: ' + fillG.slice(0, 300));
  // Undo / redo
  st.undo();
  const u = useStore.getState().project.scene;
  log(`  undo: objects ${u.objects.map((o) => o.id).join(',')} same refs ${objects.every((o) => u.objects.includes(o))} layers ${u.layers.map((l) => l.id).join(',')}`);
  useStore.getState().redo();
  const r = useStore.getState().project.scene;
  log(`  redo: objects ${r.objects.map((o) => o.kind).join(',')} layers ${r.layers.map((l) => l.mode).join(',')}`);
}
