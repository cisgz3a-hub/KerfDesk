function bytesOf(blob: Blob): Promise<Uint8Array> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(new Uint8Array(reader.result as ArrayBuffer));
    reader.onerror = () => reject(reader.error);
    reader.readAsArrayBuffer(blob);
  });
}
// jsdom 25's Blob/File have no arrayBuffer(); Chromium's do. Without this,
// readImageDensity's file.arrayBuffer() throws and is swallowed as "no density",
// so EVERY import here would fall back to 254 DPI (a jsdom artefact).
if (typeof (Blob.prototype as { arrayBuffer?: unknown }).arrayBuffer !== 'function') {
  (Blob.prototype as unknown as { arrayBuffer: () => Promise<ArrayBuffer> }).arrayBuffer =
    function (this: Blob) {
      return bytesOf(this).then((b) => b.buffer.slice(b.byteOffset, b.byteOffset + b.byteLength) as ArrayBuffer);
    };
}

// ---------- independent byte builders / parsers ----------
function chunk(type: string, data: Uint8Array): Buffer {
  const len = Buffer.alloc(4);
  len.writeUInt32BE(data.length);
  const td = Buffer.concat([Buffer.from(type, 'latin1'), Buffer.from(data)]);
  const crc = Buffer.alloc(4);
  crc.writeUInt32BE(crc32(td) >>> 0);
  return Buffer.concat([len, td, crc]);
}
function pngRgba(w: number, h: number, rgba: Uint8Array, dpi?: number): Uint8Array {
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(w, 0);
  ihdr.writeUInt32BE(h, 4);
  ihdr[8] = 8;
  ihdr[9] = 6;
  const raw = Buffer.alloc((w * 4 + 1) * h);
  for (let y = 0; y < h; y += 1) {
    raw[y * (w * 4 + 1)] = 0;
    Buffer.from(rgba.buffer, rgba.byteOffset + y * w * 4, w * 4).copy(raw, y * (w * 4 + 1) + 1);
  }
  const parts = [Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]), chunk('IHDR', ihdr)];
  if (dpi !== undefined) {
    const p = Buffer.alloc(9);
    const ppm = Math.round(dpi / 0.0254);
    p.writeUInt32BE(ppm, 0);
    p.writeUInt32BE(ppm, 4);
    p[8] = 1;
    parts.push(chunk('pHYs', p));
  }
  parts.push(chunk('IDAT', deflateSync(raw)), chunk('IEND', new Uint8Array()));
  return new Uint8Array(Buffer.concat(parts));
}
function decodeOwnPng(bytes: Uint8Array): { w: number; h: number; rgba: Uint8Array } {
  const b = Buffer.from(bytes);
  let o = 8;
  let w = 0;
  let h = 0;
  const idat: Buffer[] = [];
  while (o < b.length) {
    const len = b.readUInt32BE(o);
    const type = b.toString('latin1', o + 4, o + 8);
    const data = b.subarray(o + 8, o + 8 + len);
    if (type === 'IHDR') {
      w = data.readUInt32BE(0);
      h = data.readUInt32BE(4);
    }
    if (type === 'IDAT') idat.push(data);
    o += 12 + len;
  }
  const raw = inflateSync(Buffer.concat(idat));
  const rgba = new Uint8Array(w * h * 4);
  for (let y = 0; y < h; y += 1) raw.copy(rgba, y * w * 4, y * (w * 4 + 1) + 1, (y + 1) * (w * 4 + 1));
  return { w, h, rgba };
}
function jfif(dpi: number): Uint8Array {
  return Uint8Array.from([
    0xff, 0xd8, 0xff, 0xe0, 0x00, 0x10, 0x4a, 0x46, 0x49, 0x46, 0x00, 0x01, 0x01, 0x01,
    dpi >> 8, dpi & 0xff, dpi >> 8, dpi & 0xff, 0x00, 0x00, 0xff, 0xd9,
  ]);
}
function bmpHeaderDpi(bytes: Uint8Array): { x: number; y: number; w: number; h: number } {
  const v = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
  // BITMAPINFOHEADER: biWidth @18, biHeight @22, biXPelsPerMeter @38, biYPelsPerMeter @42
  return {
    w: v.getInt32(18, true),
    h: v.getInt32(22, true),
    x: v.getInt32(38, true) * 0.0254,
    y: v.getInt32(42, true) * 0.0254,
  };
}

async function importOne(
  file: File,
): Promise<{ object: RasterImage; toasts: string[] }> {
  const toasts: string[] = [];
  let object: SceneObject | null = null;
  await importImageFile(
    file,
    (o) => {
      object = o;
      return undefined;
    },
    (m) => toasts.push(m),
  );
  if (object === null) throw new Error(`import failed: ${toasts.join(' | ')}`);
  return { object: object as RasterImage, toasts };
}

const RED = [220, 40, 40, 255];
function redRgba(w: number, h: number): Uint8Array {
  const a = new Uint8Array(w * h * 4);
  for (let i = 0; i < w * h; i += 1) a.set(RED, i * 4);
  return a;
}
const LUMA = Math.round(0.299 * 220 + 0.587 * 40 + 0.114 * 40);

function installCanvasPng(): () => void {
  const proto = HTMLCanvasElement.prototype as unknown as Record<string, unknown>;
  const oldGet = proto.getContext;
  const oldToBlob = proto.toBlob;
  proto.getContext = function (this: HTMLCanvasElement & { __rgba?: Uint8ClampedArray }) {
    const canvas = this;
    return {
      canvas,
      createImageData: (w: number, h: number) => new ImageData(w, h),
      putImageData: (img: ImageData) => {
        canvas.__rgba = img.data;
      },
    };
  };
  proto.toBlob = function (
    this: HTMLCanvasElement & { __rgba?: Uint8ClampedArray },
    cb: (b: Blob | null) => void,
  ) {
    const rgba = this.__rgba ?? new Uint8ClampedArray(this.width * this.height * 4);
    cb(new Blob([pngRgba(this.width, this.height, new Uint8Array(rgba.buffer))], { type: 'image/png' }));
  };
  return () => {
    proto.getContext = oldGet;
    proto.toBlob = oldToBlob;
  };
}
