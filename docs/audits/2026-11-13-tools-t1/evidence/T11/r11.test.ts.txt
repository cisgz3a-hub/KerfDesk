function rectCurve(w: number, h: number): CurveSubpath {
  return {
    start: { x: 0, y: 0 },
    segments: [
      { kind: 'line', to: { x: w, y: 0 } },
      { kind: 'line', to: { x: w, y: h } },
      { kind: 'line', to: { x: 0, y: h } },
      { kind: 'line', to: { x: 0, y: 0 } },
    ],
    closed: true,
  };
}
const kinds = (c: CurveSubpath) => c.segments.map((s) => (s.kind === 'line' ? 'l' : s.kind === 'cubic' ? 'C' : 'A')).join('');

function obj(id: string, curves: CurveSubpath[], transform: Transform = { ...IDENTITY_TRANSFORM, x: 20, y: 20 }): ImportedSvg {
  const polylines = curves.map((c) => {
    const r = flattenCurveSubpath(c, { toleranceMm: 0.01, segmentBudget: 2_000_000 });
    if (r.kind !== 'ok') throw new Error('flatten');
    return { points: [...r.polyline.points], closed: c.closed };
  });
  const paths: ColoredPath[] = [{ color: '#ff0000', polylines, curves }];
  const all = polylines.flatMap((p) => p.points);
  return {
    kind: 'imported-svg', id, source: `${id}.svg`,
    bounds: bbox(all),
    transform, operationIds: ['L'], paths,
  } as unknown as ImportedSvg;
}
function load(objects: SceneObject[], picked: any[] = []): void {
  const base = createProject();
  const layer = { ...createLayer({ id: 'L', color: '#ff0000', name: 'L', mode: 'line' }), speed: 777, power: 30 };
  const project: Project = { ...base, optimization: { ...DEFAULT_PROJECT_OPTIMIZATION }, scene: { ...base.scene, objects, layers: [layer], groups: [] } };
  useStore.setState({ project, selectedObjectId: objects[0]!.id, additionalSelectedIds: new Set(objects.slice(1).map((o) => o.id)), undoStack: [], redoStack: [], selectedPathNodes: picked } as any);
}
function apply(r: number, a: 'round' | 'reverse' | 'remove'): string {
  useToastStore.setState({ toasts: [] } as any);
  st().filletSelection(r, a);
  return ((useToastStore.getState() as any).toasts ?? []).map((t: any) => t.message).join(' | ');
}
function burn() {
  const p = prepareOutput(st().project);
  if (!p.ok) return { ok: false, why: JSON.stringify((p as any).preflight).slice(0, 400), lit: 0, box: null as any, g2g3: 0 };
  const g = emitPreparedGcode(p).gcode;
  return { ok: true, why: '', ...decode(g) };
}

it('P2-1: exact-fit radius, own area + corner-distance oracle', () => {
  out('== P2-1 core, own oracle ==');
  for (const [name, w, h, r] of [['10x10 r5', 10, 10, 5], ['30x10 r5', 30, 10, 5], ['25.4x12.7 r6.35', 25.4, 12.7, 6.35], ['20x20 r10', 20, 20, 10], ['3x1 r0.5', 3, 1, 0.5]] as const) {
    for (const rr of [r * (1 - 1e-12), r, r * (1 + 1e-12), r - 1e-6]) {
      const res = filletCurveCorners(rectCurve(w, h), rr, 'round');
      const pts = res ? sample(res.curve as Curve) : sample(rectCurve(w, h) as Curve);
      const cornerD = [[0, 0], [w, 0], [w, h], [0, h]].map(([x, y]) => distTo(pts, { x: x!, y: y! }));
      const expA = w * h - 4 * rr * rr + Math.PI * rr * rr;
      out(`${name} r=${rr}: count ${res?.count ?? 0}, segs ${res ? kinds(res.curve) : '-'}, area ${f(area(pts))} (all-round ${f(expA)}, sharp ${f(w * h)}), corner->outline ${cornerD.map((d) => f(d, 3)).join(',')}`);
    }
  }
  // Second attempt on the half-rounded shape: can the user fix it by re-applying?
  const half = filletCurveCorners(rectCurve(10, 10), 5, 'round')!.curve;
  const again = filletCurveCorners(half, 5, 'round');
  const againPick = filletCurveCorners(half, 5, 'round', [0, 1, 2, 3, 4, 5, 6, 7]);
  const smaller = filletCurveCorners(half, 4.9999, 'round');
  out(`re-apply r5 on half-rounded 10x10: ${again?.count ?? 0} more; with all nodes picked: ${againPick?.count ?? 0}; r 4.9999: ${smaller?.count ?? 0} more (segs ${smaller ? kinds(smaller.curve) : '-'})`);
  // Other shapes at exact fit.
  const tri = (s: number) => ({ start: { x: 0, y: 0 }, segments: [{ kind: 'line' as const, to: { x: s, y: 0 } }, { kind: 'line' as const, to: { x: s / 2, y: (s * Math.sqrt(3)) / 2 } }], closed: true });
  const inr = 10 / (2 * Math.sqrt(3));
  out(`equilateral 10 r=inradius ${f(inr, 4)}: count ${filletCurveCorners(tri(10), inr, 'round')?.count ?? 0} of 3; r=inradius-1e-6: ${filletCurveCorners(tri(10), inr - 1e-6, 'round')?.count ?? 0}`);
  // Reverse mode exact fit.
  out(`reverse 10x10 r5: ${filletCurveCorners(rectCurve(10, 10), 5, 'reverse')?.count ?? 0} of 4`);
});

  out('== P2-1 alternatives + store + own decode ==');
  for (const [w, h, r] of [[10, 10, 5], [30, 10, 5], [25.4, 12.7, 6.35], [30, 10, 50]] as const) {
    const c = rectangleToCurve({ widthMm: w, heightMm: h, cornerRadiusMm: r });
    const pts = sample(c as Curve);
    const rr = Math.min(r, Math.min(w, h) / 2);

  }
  for (const [w, h, r, sc] of [[10, 10, 5, 1], [30, 10, 5, 1], [25.4, 12.7, 6.35, 1], [1, 0.5, 6.35, 25.4], [100, 50, 2.5, 0.1]] as const) {

function taperedSlot(halfDeg: number, Rbig: number, Rsmall: number, endKind: 'arc' | 'cubic'): CurveSubpath {
  const a = (halfDeg * Math.PI) / 180;
  const L = (Rbig - Rsmall) / Math.sin(a);
  const n = { x: -Math.sin(a), y: Math.cos(a) };
  const cy = 10;
  const p1 = { x: Rbig * n.x, y: Rbig * n.y + cy };
  const p2 = { x: L + Rsmall * n.x, y: Rsmall * n.y + cy };
  const q2 = { x: L + Rsmall * n.x, y: -Rsmall * n.y + cy };
  const q1 = { x: Rbig * n.x, y: -Rbig * n.y + cy };
  // one cubic approximating the small-end arc: tangent handles along the side directions
  const sweep = Math.PI - 2 * a; // arc angle of small end = 180 - taper
  const k = (4 / 3) * Math.tan(sweep / 4) * Rsmall;
  const dir = { x: Math.cos(a), y: -Math.sin(a) }; // travel direction along top side, toward +x
  const dir2 = { x: -Math.cos(a), y: -Math.sin(a) }; // travel direction along bottom side back to -x
  const end = endKind === 'arc'
    ? { kind: 'elliptical-arc' as const, radiusX: Rsmall, radiusY: Rsmall, rotationDeg: 0, largeArc: false, sweep: false, to: q2 }
    : { kind: 'cubic' as const, control1: { x: p2.x + k * dir.x, y: p2.y + k * dir.y }, control2: { x: q2.x - k * dir2.x, y: q2.y - k * dir2.y }, to: q2 };
  return {
    start: p1,
    segments: [
      { kind: 'line', to: p2 },
      end,
      { kind: 'line', to: q1 },
      { kind: 'elliptical-arc', radiusX: Rbig, radiusY: Rbig, rotationDeg: 0, largeArc: true, sweep: false, to: p1 },
    ],
    closed: true,
  };
}

it('P2-2: remove on tapered ends — arc vs single cubic, own sampled bbox', () => {
  out('== P2-2 core ==');
  for (const half of [1, 5, 15, 30]) {
    for (const k of ['arc', 'cubic'] as const) {
      const c = taperedSlot(half, 5, 4.5 - (half > 1 ? 1.5 : 0), k);
      const before = bbox(sample(c as Curve));
      const res = removeCurveFillets(c);
      const after = res ? bbox(sample(res.curve as Curve)) : before;
      const Rs = 4.5 - (half > 1 ? 1.5 : 0);
      const L = (5 - Rs) / Math.sin((half * Math.PI) / 180);
      const apexBeyondArc = Rs / Math.sin((half * Math.PI) / 180) - Rs;
      out(`half-taper ${half} deg, small end R${Rs} as ${k}: removed ${res?.count ?? 0}; maxX ${f(before.maxX)} -> ${f(after.maxX)} (oracle apex ${f(L + Rs / Math.sin((half * Math.PI) / 180))}, ${f(apexBeyondArc)} beyond arc)`);
    }
  }
});

it('P2-2: how a single-arc tapered end arises (SVG A, DXF bulge, Optimize Shapes)', () => {
  out('== P2-2 sources ==');
  // SVG with A command, mm units
  const half = (1 * Math.PI) / 180;
  const R1 = 5, R2 = 4.5, L = (R1 - R2) / Math.sin(half);
  const n = { x: -Math.sin(half), y: Math.cos(half) };
  const cy = 10, x0 = 10;
  const p1 = { x: x0 + R1 * n.x, y: R1 * n.y + cy }, p2 = { x: x0 + L + R2 * n.x, y: R2 * n.y + cy };
  const q2 = { x: x0 + L + R2 * n.x, y: -R2 * n.y + cy }, q1 = { x: x0 + R1 * n.x, y: -R1 * n.y + cy };
  const d = `M${p1.x} ${p1.y} L${p2.x} ${p2.y} A${R2} ${R2} 0 0 0 ${q2.x} ${q2.y} L${q1.x} ${q1.y} A${R1} ${R1} 0 1 0 ${p1.x} ${p1.y} Z`;
  for (const [label, attrs] of [['mm, viewBox 1:1', 'width="100mm" height="40mm" viewBox="0 0 100 40"'], ['px, no viewBox', 'width="378" height="151"'], ['mm, viewBox 1:1, transform translate', 'width="100mm" height="40mm" viewBox="0 0 100 40"']] as const) {
    const tr = label.includes('translate') ? ' transform="translate(1 0)"' : '';
    const svg = `<svg xmlns="http://www.w3.org/2000/svg" ${attrs}><path${tr} d="${d}" stroke="#ff0000" fill="none"/></svg>`;
    const o = parseSvg({ svgText: svg, id: 's', source: 's.svg' }).object!;
    const c = o.paths[0]!.curves![0]!;
    load([{ ...o, operationIds: ['L'] } as any]);
    const b0 = burn();
    const t = apply(0, 'remove');
    const b1 = burn();
    out(`SVG A (${label}): imported segs ${kinds(c)}; Remove toast "${t}"; burn ok ${b1.ok}, lit ${f(b0.lit)} -> ${f(b1.lit)}, box ${boxs(b0.box)} -> ${boxs(b1.box)} ${b1.why}`);
  }
  // DXF LWPOLYLINE with bulges (same slot)
  const bulgeSmall = -Math.tan((Math.PI - 2 * half) / 4);
  const bulgeBig = -Math.tan((Math.PI + 2 * half) / 4);
  const dxf = ['0', 'SECTION', '2', 'HEADER', '9', '$INSUNITS', '70', '4', '0', 'ENDSEC', '0', 'SECTION', '2', 'ENTITIES', '0', 'LWPOLYLINE', '8', '0', '90', '4', '70', '1',
    '10', String(p1.x), '20', String(p1.y), '42', '0',
    '10', String(p2.x), '20', String(p2.y), '42', String(bulgeSmall),
    '10', String(q2.x), '20', String(q2.y), '42', '0',
    '10', String(q1.x), '20', String(q1.y), '42', String(bulgeBig),
    '0', 'ENDSEC', '0', 'EOF'].join('\n');
  const dr: any = parseDxf({ dxfText: dxf, id: 'd', source: 'd.dxf' });
  const dobj = dr.object ?? dr.objects?.[0] ?? dr.result?.object;
  const dc = dobj?.paths?.[0]?.curves?.[0];
  out(`DXF bulge: result kind ${dr.kind}; segs ${dc ? kinds(dc) : 'n/a ' + Object.keys(dr).join(',')}; removeCurveFillets -> ${dc ? (removeCurveFillets(dc)?.count ?? 0) : 'n/a'}`);
  // Optimize Shapes on a dense (traced-like) polyline of the same slot
  const dense = sample(taperedSlot(1, 5, 4.5, 'arc') as Curve, 0.05);
  const densePoly: CurveSubpath = { start: dense[0]!, segments: dense.slice(1).map((to) => ({ kind: 'line' as const, to })), closed: true };
  for (const fitWith of ['lines-arcs-curves', 'lines-arcs'] as const) {
    const opt = optimizeContour(densePoly, IDENTITY_TRANSFORM, { ...DEFAULT_SHAPE_OPTIMIZE_OPTIONS, fitWith });
    const rem = removeCurveFillets(opt.curve);
    const bb = rem ? bbox(sample(rem.curve as Curve)) : null;
    out(`Optimize Shapes (${fitWith}) on dense 2-deg slot: changed ${opt.changed}, segs ${kinds(opt.curve)}; Remove -> ${rem?.count ?? 0}${bb ? `, maxX ${f(bb.maxX)}` : ''}`);
  }
  // Tool's own output: fillet a sharp 2-deg wedge tip, then remove (inverse)
  const wedge: CurveSubpath = { start: { x: 0, y: 5 }, segments: [{ kind: 'line', to: { x: 286.5, y: 10 } }, { kind: 'line', to: { x: 0, y: 15 } }], closed: true };
  const fw = filletCurveCorners(wedge, 4.5, 'round', [1]);
  const rw = fw ? removeCurveFillets(fw.curve) : null;
  out(`own fillet on sharp 2-deg wedge tip r4.5: ${fw?.count ?? 0}; then Remove restores maxX ${rw ? f(bbox(sample(rw.curve as Curve)).maxX) : '-'} (original tip 286.5)`);
});
