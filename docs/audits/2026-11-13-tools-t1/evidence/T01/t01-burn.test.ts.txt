// compare with its burned labels. Report-style: prints mismatches.

const log: string[] = [];

  log.push(s);
}

  passes: 1,

  cellHeightMm: 5,
  gapMm: 1,
};

const CASES: Case[] = [

function parsePercent(text: string): number {
  return Number(text);
}

describe('T01 per-cell decode', () => {
  for (const c of CASES) {
    it(c.title, () => {
      const device: DeviceProfile = { ...DEFAULT_DEVICE_PROFILE, ...(c.device ?? {}) };
      const grid = generateMaterialTestGrid({
        ...c.options,
        maxFeedMmPerMin: device.maxFeed,
        accelMmPerSec2: device.accelMmPerSec2,
      });
      const out = emit(grid, device, c.laserMaxPowerS);
      const maxS = c.laserMaxPowerS ?? device.maxPowerS;
      note(`\n## ${c.title}`);
      note(`ok=${out.ok} cells=${grid.cells.length} rowP=${grid.rowParameter} colP=${grid.columnParameter} mode=${grid.mode}`);
      if (!out.ok || out.gcode === '') {
        note(`NO GCODE: ${JSON.stringify(out.preflight).slice(0, 600)}`);
        return;
      }
      const pf = out.preflight as { issues?: { code: string; message: string }[] };
      if (pf.issues && pf.issues.length > 0)
        note(`preflight: ${pf.issues.map((i) => i.code).join(', ')}`);
      const moves = litMoves(out.gcode, device);
      const { cells, labelMoves, stray } = perCell(grid, moves);
      const lab = labels(grid);
      const colLabels = lab.filter((l) => l.id.startsWith('material-test-power-c')).map((l) => l.text);
      const rowLabels = lab.filter((l) => l.id.startsWith('material-test-speed-r')).map((l) => l.text);
      note(`rowLabels=${rowLabels.join('|')}`);
      note(`colLabels=${colLabels.join('|')}`);
      note(`layer names: ${grid.scene.layers.map((l) => l.name).join(' || ')}`);
      const mismatches: string[] = [];
      for (const cell of grid.cells) {
        const r = cells.get(name(cell))!;
        const rowLabel = rowLabels[cell.row]!;
        const colLabel = colLabels[cell.column]!;
        const labelFor = (p: string): string | undefined =>
          grid.rowParameter === p ? rowLabel : grid.columnParameter === p ? colLabel : undefined;
        if (r.litLen === 0) {
          mismatches.push(`${r.name}: nothing burned (power ${cell.power})`);
          continue;
        }
        // speed
        const sl = labelFor('speed');
        if (r.fWords.length !== 1) mismatches.push(`${r.name}: F words ${r.fWords.join(',')}`);
        if (sl !== undefined && Number(r.fWords[0]) !== Number(sl))
          mismatches.push(`${r.name}: F${r.fWords[0]} vs speed label ${sl}`);
        if (sl !== undefined && r.fWords[0] !== sl)
          mismatches.push(`${r.name}: F word text '${r.fWords[0]}' vs label text '${sl}'`);
        // power
        const pl = labelFor('power');
        if (r.sValues.length !== 1) mismatches.push(`${r.name}: S values ${r.sValues.join(',')}`);
        const burnedPct = (r.sValues[0]! / maxS) * 100;
        const expectedPct = pl !== undefined ? parsePercent(pl) : cell.power;
        const tol = maxS >= 100 ? 50 / maxS + 1e-9 : 0.05 + 1e-9;
        if (Math.abs(burnedPct - expectedPct) > tol)
          mismatches.push(`${r.name}: S${r.sValues[0]} = ${burnedPct.toFixed(3)}% vs ${pl !== undefined ? 'label' : 'fixed'} ${expectedPct}%`);
        // passes
        if (grid.mode === 'line') {
          const per = 2 * (cell.bounds.maxX - cell.bounds.minX + cell.bounds.maxY - cell.bounds.minY);
          const ratio = r.litLen / per;
          const pal = labelFor('passes');
          const expectedPasses = pal !== undefined ? Number(pal) : cell.passes;
          if (Math.abs(ratio - expectedPasses) > 0.02)
            mismatches.push(`${r.name}: traced ${ratio.toFixed(3)} perimeters vs passes ${expectedPasses}`);
        }
      }
      if (grid.mode === 'fill' && (grid.rowParameter === 'passes' || grid.columnParameter === 'passes')) {
        const row0 = grid.cells.filter((x) => x.row === 0).map((x) => cells.get(name(x))!.litLen.toFixed(1));
        note(`fill passes row0 litLen: ${row0.join(', ')}`);
      }
      // order (gentlest first)
      const order = burnOrder(grid, moves);
      const generated = grid.cells.map(name);
      if (order.join() !== generated.join()) note(`ORDER differs: ${order.slice(0, 12).join(',')} ...`);
      // labels
      const lf = [...new Set(labelMoves.map((m) => m.fWord))];
      const ls = [...new Set(labelMoves.map((m) => m.s))];
      note(`label moves=${labelMoves.length} F=${lf.join(',')} S=${ls.join(',')}; stray lit moves=${stray.length}`);
      if (stray.length > 0) note(`stray sample: ${JSON.stringify(stray.slice(0, 3))}`);
      // label/cell overlap
      for (const l of lab)
        for (const cell of grid.cells) {
          const b = cell.bounds;
          const ov = !(l.bounds.maxX <= b.minX || l.bounds.minX >= b.maxX || l.bounds.maxY <= b.minY || l.bounds.minY >= b.maxY);
          if (ov) mismatches.push(`label ${l.id} overlaps ${name(cell)}`);
        }
      const xy = allXY(out.gcode, device);
      note(`machine XY extent: X ${xy.minX}..${xy.maxX} Y ${xy.minY}..${xy.maxY} (bed ${device.bedWidth}x${device.bedHeight})`);
      note(mismatches.length === 0 ? 'per-cell: ALL MATCH' : `MISMATCHES (${mismatches.length}):\n  ${mismatches.slice(0, 25).join('\n  ')}`);
      expect(true).toBe(true);
    });
  }
  it('write log', () => {
    writeFileSync('C:/Users/stolk/kd-t1/zz-t01/t01-burn.log', log.join('\n'));
  });
});
