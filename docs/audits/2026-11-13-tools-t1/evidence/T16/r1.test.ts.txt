type P = { x: number; y: number };
const LOG = 'C:/Users/stolk/kd-t1/zz-t16r/r1.log';
writeFileSync(LOG, '');
const out = (s: string) => {
  appendFileSync(LOG, s + '\n');
};

function wind(p: P, contours: P[][]): number {
  let w = 0;
  for (const c of contours) {
    for (let i = 0; i < c.length; i += 1) {
      const a = c[i]!;
      const b = c[(i + 1) % c.length]!;
      if (a.y <= p.y) {
        if (b.y > p.y && (b.x - a.x) * (p.y - a.y) - (p.x - a.x) * (b.y - a.y) > 0) w += 1;
      } else if (b.y <= p.y && (b.x - a.x) * (p.y - a.y) - (p.x - a.x) * (b.y - a.y) < 0) w -= 1;
    }
  }
  return w;
}

const STEP = 0.05;

// Signed crossings of a horizontal line, sorted by x; winding at x = sum of dirs of crossings <= x.
function rowCrossings(contours: P[][], py: number): Array<[number, number]> {
  const xs: Array<[number, number]> = [];
  for (const c of contours) {
    for (let i = 0; i < c.length; i += 1) {
      const a = c[i]!;
      const b = c[(i + 1) % c.length]!;
      if ((a.y <= py) === (b.y <= py)) continue;
      const x = a.x + ((py - a.y) / (b.y - a.y)) * (b.x - a.x);
      xs.push([x, b.y > a.y ? 1 : -1]);
    }
  }
  return xs.sort((u, v) => u[0] - v[0]);
}

function measure(label: string, obj: SceneObject, oracleContours: P[][], withFill = true) {
  const b = obj.bounds;
  const t = obj.transform;
  const x0 = t.x + b.minX - 1;
  const y0 = t.y + b.minY - 1;
  const w = b.maxX - b.minX + 2;
  const h = b.maxY - b.minY + 2;
  const W = Math.round(w / STEP);
  const H = Math.round(h / STEP);
  const image = {
    kind: 'raster-image',
    id: 'IMG',
    bounds: { minX: 0, minY: 0, maxX: w, maxY: h },
    transform: { ...IDENTITY_TRANSFORM, x: x0, y: y0 },
    imageMaskId: obj.id,
  } as unknown as RasterImage;
  const test = createImageMaskPixelTest(image, obj, W, H);
  if (test === null) throw new Error('no mask test');
  // Fill: the real per-object contours for a Fill layer, mapped back to scene space.
  let fillScene: P[][] = [];
  if (withFill) {
    const layer = { ...createLayer({ id: 'L', color: '#000000', name: 'F', mode: 'fill' }) };
    const objForFill = { ...obj, paths: obj.paths.map((p) => ({ ...p, color: '#000000' })) } as SceneObject;
    fillScene = collectFillContoursForLayer([objForFill], layer as never, DEFAULT_DEVICE_PROFILE)
      .filter((l) => l.closed)
      .map((l) => l.points.map((q) => toSceneCoords(q, DEFAULT_DEVICE_PROFILE)));
  }
  const cell = STEP * STEP;
  let oracle = 0, mask = 0, fill = 0, maskMiss = 0, maskLeak = 0, fillMiss = 0, fillLeak = 0, interiorMiss = 0;
  const missGrid = new Uint8Array(W * H);
  const oracleGrid = new Uint8Array(W * H);
  for (let y = 0; y < H; y += 1) {
    const py = y0 + (y + 0.5) * STEP;
    const oc = rowCrossings(oracleContours, py);
    const fc = withFill ? rowCrossings(fillScene, py) : [];
    let oi = 0, ow = 0, fi = 0, fw = 0;
    for (let x = 0; x < W; x += 1) {
      const px = x0 + (x + 0.5) * STEP;
      while (oi < oc.length && oc[oi]![0] <= px) { ow += oc[oi]![1]; oi += 1; }
      while (fi < fc.length && fc[fi]![0] <= px) { fw += 1; fi += 1; }
      const o = ow !== 0;
      const m = test(x, y);
      const f = withFill ? fw % 2 === 1 : false;
      if (!o && f) fillLeak += 1;
      if (o) oracle += 1;
      if (m) mask += 1;
      if (f) fill += 1;
      if (o && !m) { maskMiss += 1; missGrid[y * W + x] = 1; }
      if (!o && m) maskLeak += 1;
      if (o && !f) fillMiss += 1;
      if (o) oracleGrid[y * W + x] = 1;
    }
  }
  // interior miss: missed cell whose +-4 cell (0.2 mm) neighbourhood is all oracle ink
  for (let y = 4; y < H - 4; y += 1) {
    for (let x = 4; x < W - 4; x += 1) {
      if (missGrid[y * W + x] !== 1) continue;
      let all = true;
      for (let dy = -4; dy <= 4 && all; dy += 1)
        for (let dx = -4; dx <= 4; dx += 1) if (oracleGrid[(y + dy) * W + x + dx] !== 1) { all = false; break; }
      if (all) interiorMiss += 1;
    }
  }
  const r = (n: number) => (n * cell).toFixed(2);
  out(
    `${label}: nonzero-oracle ${r(oracle)} mm2 | mask ${r(mask)} (miss ${r(maskMiss)}, leak ${r(maskLeak)}, interior miss >=0.2mm from edge ${r(interiorMiss)})` +
      (withFill ? ` | Fill ${r(fill)} (miss ${r(fillMiss)}, leak ${r(fillLeak)})` : ''),
  );
  return { maskMiss: maskMiss * cell, interiorMiss: interiorMiss * cell };
}

const FONTS: Array<[string, string]> = [
  ['dancing-script-regular', 'DancingScript-Regular'],
  ['pacifico-regular', 'Pacifico-Regular'],
  ['great-vibes-regular', 'GreatVibes-Regular'],
  ['allura-regular', 'Allura-Regular'],
  ['alex-brush-regular', 'AlexBrush-Regular'],
  ['parisienne-regular', 'Parisienne-Regular'],

  ['corinthia-regular', 'Corinthia-Regular'],
  ['roboto-regular', 'Roboto-Regular'],
  ['anton-regular', 'Anton-Regular'],
];

async function render(file: string, content: string, sizeMm: number, lineHeight = 1.4) {
  const bytes = readFileSync(`C:/Users/stolk/kd-t1/t13-main/src/ui/text/fonts/${file}.ttf`);
  return textToPolylines({
    fontBuffer: bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength),
    content,
    sizeMm,
    alignment: 'left',
    lineHeight,
    letterSpacing: 0,
    color: '#000000',
  } as never);
}

function textObj(key: string, content: string, sizeMm: number, rendered: { paths: SceneObject extends never ? never : any; bounds: any }, weld: boolean): SceneObject {
  return {
    kind: 'text', id: 'T', content, fontKey: key, sizeMm, alignment: 'left', lineHeight: 1.4, letterSpacing: 0,
    weldOverlaps: weld, color: '#000000', bounds: { ...rendered.bounds }, transform: { ...IDENTITY_TRANSFORM, x: 20, y: 20 },
    paths: rendered.paths,
  } as unknown as SceneObject;
}

function rawContours(rendered: any, tx = 20, ty = 20): P[][] {
  return rendered.paths.flatMap((p: any) =>
    p.polylines.filter((l: any) => l.closed !== false && l.points.length >= 3).map((l: any) => l.points.map((q: P) => ({ x: q.x + tx, y: q.y + ty }))),
  );
}

it('R1a: bundled fonts, welded (new-text default, ADR-321) vs unwelded, as an image mask', async () => {
  for (const [key, file] of FONTS) {
    const raw = await render(file, 'Wedding Love', 30);
    const welded = weldTextRender(raw as never);
    if (welded.kind === 'error') { out(`${key}: weld error`); continue; }
    const oracle = rawContours(raw);
    measure(`${key} UNWELDED`, textObj(key, 'Wedding Love', 30, raw as never, false), oracle);
    measure(`${key} WELDED  `, textObj(key, 'Wedding Love', 30, welded.value as never, true), oracle);
  }
});

it('R1b: welded multi-line script text with tight line height (lines overlap)', async () => {
  for (const [key, file] of [['dancing-script-regular', 'DancingScript-Regular'], ['great-vibes-regular', 'GreatVibes-Regular']] as const) {
    const raw = await render(file, 'Joy\nfly', 30, 0.75);
    out(`${key} multi-line: render paths ${raw.paths.length}`);
    const welded = weldTextRender(raw as never);
    if (welded.kind === 'error') { out('weld error'); continue; }
    measure(`${key} 2 lines lh0.75 WELDED`, textObj(key, 'Joy\nfly', 30, welded.value as never, true), rawContours(raw));
  }
});

it('R1c: Inkscape-style SVG, two overlapping shapes each with fill-rule:evenodd in style (separate paths)', () => {
  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="60mm" height="40mm" viewBox="0 0 60 40">
<path style="fill:#000000;fill-rule:evenodd" d="M 5 5 H 35 V 35 H 5 Z"/>
<path style="fill:#000000;fill-rule:evenodd" d="M 25 5 H 55 V 35 H 25 Z"/></svg>`;
  const parsed = parseSvg({ svgText: svg, id: 'S', source: 's.svg' });
  const obj = parsed.object!;
  out(`R1c parsed paths ${obj.paths.length}, rules ${obj.paths.map((p) => p.fillRule).join(',')}, transform ${JSON.stringify(obj.transform)}`);
  const contours: P[][] = obj.paths.flatMap((p) => p.polylines.map((l) => l.points.map((q) => ({ x: q.x * obj.transform.scaleX + obj.transform.x, y: q.y * obj.transform.scaleY + obj.transform.y }))));
  // oracle here: union of the two squares (what each path paints) = nonzero of same-wound squares
  measure('R1c two overlapping evenodd paths', obj as SceneObject, contours);
  const svg2 = svg.replace(/;fill-rule:evenodd/g, '');
  const obj2 = parseSvg({ svgText: svg2, id: 'S', source: 's.svg' }).object!;
  out(`R1c' no fill-rule: paths ${obj2.paths.length}`);
  measure("R1c' same, no fill-rule attribute", obj2 as SceneObject, contours);
});
