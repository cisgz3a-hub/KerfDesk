// T16 refuter R2: whole design-canvas frame (drawScene, not drawRasterImage) after the REAL
// store applyImageMask, vs the same image carrying the REAL bakedImageMaskClip (what Crop and
// Flatten store). A recording 2D context tracks the clip stack at every drawImage.
import { appendFileSync, writeFileSync } from 'node:fs';
import { it, vi } from 'vitest';
import { bakedImageMaskClip } from '../t13-main/src/core/raster/image-mask-clip';
import { createLayer, createProject, IDENTITY_TRANSFORM, type RasterImage } from '../t13-main/src/core/scene';
import { createEllipse } from '../t13-main/src/core/shapes/primitives/create-ellipse';
import { useStore } from '../t13-main/src/ui/state/store';
import { resetStore } from '../t13-main/src/ui/state/test-helpers';
import { drawScene } from '../t13-main/src/ui/workspace/draw-scene';

const LOG = 'C:/Users/stolk/kd-t1/zz-t16r/r2.log';
writeFileSync(LOG, '');
const out = (s: string) => appendFileSync(LOG, s + '\n');

class ReadyImage {
  complete = true;
  naturalWidth = 80;
  naturalHeight = 60;
  width = 80;
  height = 60;
  src = '';
  onload: unknown = null;
  onerror: unknown = null;
  decoding = 'async';
  addEventListener() {}
  removeEventListener() {}
  decode() {
    return Promise.resolve();
  }
}

function recorder() {
  const log: Array<{ op: string; clipDepth: number; args: unknown[] }> = [];
  const stack: number[] = [];
  let clipDepth = 0;
  const target: Record<string, unknown> = {};
  const ctx = new Proxy(target, {
    get: (_t, k) => {
      if (typeof k !== 'string' || k === 'then') return undefined;
      if (k in target) return target[k];
      if (k === 'canvas') return { width: 800, height: 600 };
      if (k === 'measureText') return () => ({ width: 10 });
      if (k === 'getTransform') return () => ({ a: 1, b: 0, c: 0, d: 1, e: 0, f: 0 });
      if (k === 'createLinearGradient' || k === 'createPattern' || k === 'createRadialGradient')
        return () => ({ addColorStop() {}, setTransform() {} });
      if (k === 'getLineDash') return () => [];
      return (...args: unknown[]) => {
        if (k === 'save') stack.push(clipDepth);
        if (k === 'restore') clipDepth = stack.pop() ?? 0;
        if (k === 'clip') clipDepth += 1;
        if (k === 'drawImage' || k === 'clip') log.push({ op: k, clipDepth, args: args.slice(1) });
      };
    },
    set: (_t, k, v) => {
      target[k as string] = v;
      return true;
    },
  });
  return { ctx: ctx as unknown as CanvasRenderingContext2D, log };
}

it('R2: canvas frame after Apply Mask vs after the baked (Crop/Flatten) clip', () => {
  vi.stubGlobal('Image', ReadyImage);
  resetStore();
  const base = createProject();
  const image: RasterImage = {
    kind: 'raster-image', id: 'PHOTO', source: 'p.png', dataUrl: 'data:image/png;base64,r2photo',
    pixelWidth: 80, pixelHeight: 60, bounds: { minX: 0, minY: 0, maxX: 40, maxY: 30 },
    transform: { ...IDENTITY_TRANSFORM, x: 50, y: 50 }, color: '#808080', dither: 'threshold', linesPerMm: 5,
  } as RasterImage;
  const circle = createEllipse({ id: 'CIRC', color: '#000000', spec: { widthMm: 20, heightMm: 20 } } as never);
  const mask = { ...circle, transform: { ...IDENTITY_TRANSFORM, x: 60, y: 55 } };
  useStore.getState().setProject({
    ...base,
    scene: {
      objects: [image, mask],
      layers: [
        createLayer({ id: 'img', color: '#808080', mode: 'image' }),
        createLayer({ id: 'line', color: '#000000', mode: 'line' }),
      ],
    },
  });
  useStore.getState().applyImageMask('PHOTO', 'CIRC');
  const applied = useStore.getState().project;
  const photo = applied.scene.objects.find((o) => o.id === 'PHOTO') as RasterImage;
  out(`after store applyImageMask: imageMaskId=${photo.imageMaskId}, objects=${applied.scene.objects.length}`);

  const live = recorder();
  drawScene(live.ctx, 800, 600, applied, { selectedId: null, preview: false });
  const liveDraws = live.log.filter((e) => e.op === 'drawImage');
  out(`LIVE mask, design canvas: drawImage calls ${liveDraws.length}; clip depth at each: ${liveDraws.map((e) => e.clipDepth).join(',')}; dest rect args ${JSON.stringify(liveDraws.map((e) => e.args.slice(-4)))}`);

  const clip = bakedImageMaskClip(photo, mask as never);
  const baked = { ...photo, imageClip: clip ?? undefined } as RasterImage;
  delete (baked as { imageMaskId?: string }).imageMaskId;
  const bakedProject = { ...applied, scene: { ...applied.scene, objects: [baked, mask] } };
  const own = recorder();
  drawScene(own.ctx, 800, 600, bakedProject as never, { selectedId: null, preview: false });
  const ownDraws = own.log.filter((e) => e.op === 'drawImage');
  out(`BAKED clip (what Crop/Flatten store), design canvas: drawImage calls ${ownDraws.length}; clip depth at each: ${ownDraws.map((e) => e.clipDepth).join(',')}`);
  vi.unstubAllGlobals();
});
