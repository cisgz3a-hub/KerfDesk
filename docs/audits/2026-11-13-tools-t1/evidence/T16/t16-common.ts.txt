export type P = { x: number; y: number };
export const DEV = DEFAULT_DEVICE_PROFILE;
export function logger(path: string): (s: string) => void {
  const lines: string[] = [];
  return (s: string) => { lines.push(s); writeFileSync(path, lines.join('\n')); };
}
export const b64 = (u: Uint8Array) => Buffer.from(u).toString('base64');

export function mkImage(id: string, W: number, H: number, box: { x: number; y: number; w: number; h: number }, transform: Transform = IDENTITY_TRANSFORM, extra: Partial<RasterImage> = {}, lumaValue = 0): RasterImage {
  const luma = new Uint8Array(W * H).fill(lumaValue);
  return {
    kind: 'raster-image', id, source: `${id}.png`, dataUrl: 'data:image/png;base64,iVBORw0KGgo=', pixelWidth: W, pixelHeight: H,
    bounds: { minX: box.x, minY: box.y, maxX: box.x + box.w, maxY: box.y + box.h }, transform, color: '#808080',
    dither: 'threshold', linesPerMm: 5, lumaBase64: b64(luma), ...extra,
  } as RasterImage;
}
export function imageLayer(extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id: 'img', color: '#808080', name: 'Photo', mode: 'image' }), power: 80, minPower: 0, speed: 3000, linesPerMm: 5,
    ditherAlgorithm: 'threshold', dotWidthCorrectionMm: 0, ...extra } as Layer;
}
export function maskLayer(output: boolean, mode: Layer['mode'] = 'line', extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id: 'mask', color: '#000000', name: 'Mask', mode }), output, power: 50, speed: 1000, ...extra } as Layer;
}
/** A shape/imported-svg/text object whose polylines are exactly the given local contours. */
export function mkMask(id: string, contours: P[][], transform: Transform = IDENTITY_TRANSFORM, kind: 'shape' | 'imported-svg' | 'text' = 'shape', fillRule?: 'evenodd' | 'nonzero', closed = true): SceneObject {
  const all = contours.flat();
  const bounds = { minX: Math.min(...all.map((p) => p.x)), minY: Math.min(...all.map((p) => p.y)), maxX: Math.max(...all.map((p) => p.x)), maxY: Math.max(...all.map((p) => p.y)) };
  const paths = [{ color: '#000000', ...(fillRule ? { fillRule } : {}), polylines: contours.map((points) => ({ points, closed })) }];
  if (kind === 'shape') return { kind, id, spec: { kind: 'polyline', points: contours[0], closed }, color: '#000000', bounds, transform, paths } as unknown as SceneObject;
  if (kind === 'imported-svg') return { kind, id, source: 'mask.svg', bounds, transform, paths } as unknown as SceneObject;
  return { kind: 'text', id, content: 'AB', fontKey: 'x', sizeMm: 10, alignment: 'left', lineHeight: 1, letterSpacing: -0.3, color: '#000000', bounds, transform, paths } as unknown as SceneObject;
}
export function project(objects: SceneObject[], layers: Layer[]): Project {
  const base = createProject(DEV);
  return { ...base, scene: { ...base.scene, objects, layers, groups: [] } } as Project;
}
export function gcodeOf(p: Project): string {
  const r = prepareOutput(p);
  if (!r.ok) return `NOT OK ${JSON.stringify(r).slice(0, 800)}`;
  return emitPreparedGcode(r).gcode;
}

// ---------- decoder: every G1 move with S>0, in scene coordinates (front-left origin) ----------
export type Seg = { a: P; b: P; s: number };
export function burnedSegments(g: string, bedH = DEV.bedHeight): Seg[] {
  const out: Seg[] = [];
  let x = 0, y = 0, S = 0, mode = 0, laserOn = false;
  for (const raw of g.split('\n')) {
    const line = raw.replace(/;.*$/, '').trim(); if (!line) continue;
    if (/^M[34]\b/.test(line) || /\bM[34]\b/.test(line)) laserOn = true;
    if (/\bM5\b/.test(line)) laserOn = false;
    const m = line.match(/^G0?([0-3])(?![0-9])/); if (m) mode = Number(m[1]);
    const nx = line.match(/X(-?[\d.]+)/), ny = line.match(/Y(-?[\d.]+)/), ns = line.match(/S(-?[\d.]+)/);
    if (ns) S = Number(ns[1]);
    const x1 = nx ? Number(nx[1]) : x; const y1 = ny ? Number(ny[1]) : y;
    if (mode === 1 && S > 0 && laserOn && (nx || ny)) out.push({ a: { x, y: bedH - y }, b: { x: x1, y: bedH - y1 }, s: S });
    x = x1; y = y1;
  }
  return out;
}

// ---------- independent geometry ----------
export function xf(p: P, t: Transform): P {
  let x = p.x * t.scaleX, y = p.y * t.scaleY;
  if (t.mirrorX) x = -x; if (t.mirrorY) y = -y;
  const r = (t.rotationDeg * Math.PI) / 180;
  return { x: x * Math.cos(r) - y * Math.sin(r) + t.x, y: x * Math.sin(r) + y * Math.cos(r) + t.y };
}
/** Winding number of point about all contours (signed crossing count). */
export function winding(p: P, contours: P[][]): number {
  let w = 0;
  for (const c of contours) for (let i = 0; i < c.length; i++) {
    const a = c[i]!, b = c[(i + 1) % c.length]!;
    if (a.y <= p.y) { if (b.y > p.y && cross(a, b, p) > 0) w++; }
    else if (b.y <= p.y && cross(a, b, p) < 0) w--;
  }
  return w;
}
const cross = (a: P, b: P, p: P) => (b.x - a.x) * (p.y - a.y) - (p.x - a.x) * (b.y - a.y);
export const inEvenOdd = (p: P, c: P[][]) => winding(p, c) % 2 !== 0;
export const inNonZero = (p: P, c: P[][]) => winding(p, c) !== 0;
function segDist(p: P, a: P, b: P): number {
  const dx = b.x - a.x, dy = b.y - a.y; const L = dx * dx + dy * dy;
  const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / L));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}
export function edgeDist(p: P, contours: P[][]): number {
  let d = Infinity;
  for (const c of contours) for (let i = 0; i < c.length; i++) d = Math.min(d, segDist(p, c[i]!, c[(i + 1) % c.length]!));
  return d;
}
export function circle(cx: number, cy: number, rx: number, ry = rx, n = 96, start = 0.013): P[] {
  return Array.from({ length: n }, (_, i) => { const a = start + (i / n) * 2 * Math.PI; return { x: cx + rx * Math.cos(a), y: cy + ry * Math.sin(a) }; });
}
export const rect = (x0: number, y0: number, x1: number, y1: number): P[] => [{ x: x0, y: y0 }, { x: x1, y: y0 }, { x: x1, y: y1 }, { x: x0, y: y1 }];

/**
 * Compare the burned ribbon (each burned segment widened +-pitch/2 perpendicular) with the
 * oracle region (inside image footprint AND inside mask under `rule`) on a fine grid.
 */
export function compare(segs: Seg[], pitch: number, imageQuad: P[], maskContours: P[][], rule: 'evenodd' | 'nonzero' | 'none', region: { x0: number; y0: number; x1: number; y1: number }, g = 0.05) {
  const nx = Math.round((region.x1 - region.x0) / g), ny = Math.round((region.y1 - region.y0) / g);
  const burned = new Uint8Array(nx * ny);
  const half = pitch / 2;
  for (const s of segs) {
    const dx = s.b.x - s.a.x, dy = s.b.y - s.a.y; const L = Math.hypot(dx, dy); if (L < 1e-9) continue;
    const ux = dx / L, uy = dy / L;
    const minX = Math.min(s.a.x, s.b.x) - half, maxX = Math.max(s.a.x, s.b.x) + half, minY = Math.min(s.a.y, s.b.y) - half, maxY = Math.max(s.a.y, s.b.y) + half;
    const i0 = Math.max(0, Math.floor((minX - region.x0) / g)), i1 = Math.min(nx - 1, Math.ceil((maxX - region.x0) / g));
    const j0 = Math.max(0, Math.floor((minY - region.y0) / g)), j1 = Math.min(ny - 1, Math.ceil((maxY - region.y0) / g));
    for (let j = j0; j <= j1; j++) for (let i = i0; i <= i1; i++) {
      const px = region.x0 + (i + 0.5) * g - s.a.x, py = region.y0 + (j + 0.5) * g - s.a.y;
      const along = px * ux + py * uy, perp = Math.abs(-px * uy + py * ux);
      if (along >= 0 && along <= L && perp <= half) burned[j * nx + i] = 1;
    }
  }
  let leak = 0, miss = 0, both = 0, oracleArea = 0, burnArea = 0, maxLeak = 0, maxMiss = 0;
  let leakFar = 0, missFar = 0; let farLeakAt: P | null = null, farMissAt: P | null = null;
  const bound = [...maskContours, imageQuad];
  for (let j = 0; j < ny; j++) for (let i = 0; i < nx; i++) {
    const p = { x: region.x0 + (i + 0.5) * g, y: region.y0 + (j + 0.5) * g };
    const inImg = inNonZero(p, [imageQuad]);
    const inMask = rule === 'none' ? true : rule === 'evenodd' ? inEvenOdd(p, maskContours) : inNonZero(p, maskContours);
    const o = inImg && inMask; const b = burned[j * nx + i] === 1;
    if (o) oracleArea++; if (b) burnArea++;
    if (o && b) both++;
    else if (b) { leak++; const d = edgeDist(p, bound); if (d > maxLeak) { maxLeak = d; farLeakAt = p; } if (d > 2 * pitch) leakFar++; }
    else if (o) { miss++; const d = edgeDist(p, bound); if (d > maxMiss) { maxMiss = d; farMissAt = p; } if (d > 2 * pitch) missFar++; }
  }
  const a = g * g;
  const r3 = (v: number) => Math.round(v * 1000) / 1000;
  return { oracleMm2: r3(oracleArea * a), burnMm2: r3(burnArea * a), leakMm2: r3(leak * a), missMm2: r3(miss * a), maxLeakMm: r3(maxLeak), maxMissMm: r3(maxMiss), leakFarMm2: r3(leakFar * a), missFarMm2: r3(missFar * a), farLeakAt, farMissAt };
}
