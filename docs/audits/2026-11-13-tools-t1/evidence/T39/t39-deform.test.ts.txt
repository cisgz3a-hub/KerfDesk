// ---------- independent oracle: de Casteljau on rows, then on the column ----------
type Box = { minX: number; minY: number; maxX: number; maxY: number };
function casteljau(p: Vec[], t: number): Vec {
  let q = p.map((v) => ({ ...v }));
  while (q.length > 1) {
    const n: Vec[] = [];
    for (let i = 1; i < q.length; i++) n.push({ x: q[i - 1]!.x + (q[i]!.x - q[i - 1]!.x) * t, y: q[i - 1]!.y + (q[i]!.y - q[i - 1]!.y) * t });
    q = n;
  }
  return q[0]!;
}
function oracleDeform(box: Box, h: Vec[]): (p: Vec) => Vec {
  return (p) => {
    const u = (p.x - box.minX) / (box.maxX - box.minX);
    const v = (p.y - box.minY) / (box.maxY - box.minY);
    const rows = [0, 1, 2, 3].map((r) => casteljau(h.slice(r * 4, r * 4 + 4), u));
    return casteljau(rows, v);
  };
}
function segDist(p: Vec, a: Vec, b: Vec): number {
  const dx = b.x - a.x, dy = b.y - a.y;
  const l2 = dx * dx + dy * dy;
  const t = l2 === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / l2));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}
function distToPoly(p: Vec, pts: ReadonlyArray<Vec>): number {
  let best = Infinity;
  for (let i = 1; i < pts.length; i++) best = Math.min(best, segDist(p, pts[i - 1]!, pts[i]!));
  if (pts.length === 1) best = Math.hypot(p.x - pts[0]!.x, p.y - pts[0]!.y);
  return best;
}
function densify(pts: ReadonlyArray<Vec>, step: number): Vec[] {
  const res: Vec[] = [];
  for (let i = 1; i < pts.length; i++) {
    const a = pts[i - 1]!, b = pts[i]!;
    const n = Math.max(1, Math.ceil(Math.hypot(b.x - a.x, b.y - a.y) / step));
    for (let k = 0; k < n; k++) res.push({ x: a.x + ((b.x - a.x) * k) / n, y: a.y + ((b.y - a.y) * k) / n });
  }
  res.push(pts.at(-1)!);
  return res;
}
function worldPolys(o: SceneObject): Vec[][] {
  if (!('paths' in o)) return [];
  return o.paths.flatMap((p) =>
    p.polylines.map((l) => {
      const w = l.points.map((q) => applyTransform(q, o.transform));
      return l.closed && w.length > 0 ? [...w, w[0]!] : w;
    }),
  );
}
function hausdorff(expected: Vec[][], actual: Vec[][], step = 0.25): { fwd: number; back: number } {
  let fwd = 0, back = 0;
  for (const poly of expected)
    for (const p of densify(poly, step)) {
      let best = Infinity;
      for (const a of actual) best = Math.min(best, distToPoly(p, a));
      fwd = Math.max(fwd, best);
    }
  for (const poly of actual)
    for (const p of densify(poly, step)) {
      let best = Infinity;
      for (const e of expected) best = Math.min(best, distToPoly(p, e));
      back = Math.max(back, best);
    }
  return { fwd: r3(fwd), back: r3(back) };
}
function polyLen(p: Vec[]): number {
  let s = 0;
  for (let i = 1; i < p.length; i++) s += Math.hypot(p[i]!.x - p[i - 1]!.x, p[i]!.y - p[i - 1]!.y);
  return s;
}
function layer(id = 'L', mode: 'line' | 'fill' = 'line', extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id, color: '#000000', name: id, mode }), speed: 1000, power: 40, passes: 1, ...extra } as Layer;

function svg(id: string, polylines: { closed: boolean; points: Vec[] }[], extra: Partial<ImportedSvg> = {}): ImportedSvg {
  const all = polylines.flatMap((p) => p.points);
  return {
    kind: 'imported-svg', id, source: `${id}.svg`,
    bounds: { minX: Math.min(...all.map((p) => p.x)), minY: Math.min(...all.map((p) => p.y)), maxX: Math.max(...all.map((p) => p.x)), maxY: Math.max(...all.map((p) => p.y)) },
    transform: IDENTITY_TRANSFORM, operationIds: ['L'],
    paths: [{ color: '#000000', operationIds: ['L'], polylines }],
    ...extra,
  } as ImportedSvg;
}
function denseRect(id: string, x: number, y: number, w: number, h: number, step = 5, extra: Partial<ImportedSvg> = {}): ImportedSvg {
  const corners = [{ x, y }, { x: x + w, y }, { x: x + w, y: y + h }, { x, y: y + h }, { x, y }];
  return svg(id, [{ closed: true, points: densify(corners, step).slice(0, -1) }], extra);
}
function lastToast(): string {
  const toasts = (useToastStore.getState() as unknown as { toasts: { message: string }[] }).toasts;
  return toasts.at(-1)?.message ?? '';
}
function deform(dragged: (h: Vec[]) => Vec[]): { box: Box; handles: Vec[] } {
  startWarpDeformTool('deform');
  const s = activeWarpDeformSession();
  if (s === null) throw new Error(`no session: ${lastToast()}`);
  const handles = dragged(s.handles.map((p) => ({ ...p })));
  useWarpDeformSession.getState().setHandles(handles);
  return { box: { ...s.box }, handles };
}
const moveH = (moves: Record<number, Vec>) => (h: Vec[]) => h.map((p, i) => (moves[i] ? { x: p.x + moves[i]!.x, y: p.y + moves[i]!.y } : p));
function burn(project: Project) {
  const prepared = prepareOutput(project);
  if (!prepared.ok) return { ok: false, moves: [] as LitMove[], issues: JSON.stringify(prepared.preflight).slice(0, 400) };
  const emitted = emitPreparedGcode(prepared);
  return { ok: true, moves: litMoves(emitted.gcode, project.device), issues: '' };
}
function obj(id: string): SceneObject {
  return st().project.scene.objects.find((o) => o.id === id)!;
}
function selfCrossings(poly: Vec[]): number {
  let n = 0;
  const o = (p: Vec, q: Vec, r: Vec) => (q.x - p.x) * (r.y - p.y) - (q.y - p.y) * (r.x - p.x);
  const inter = (a: Vec, b: Vec, c: Vec, d: Vec) => o(a, b, c) * o(a, b, d) < 0 && o(c, d, a) * o(c, d, b) < 0;
  for (let i = 1; i < poly.length; i++)
    for (let j = i + 2; j < poly.length; j++) {
      if (i === 1 && j === poly.length - 1) continue;
      if (inter(poly[i - 1]!, poly[i]!, poly[j - 1]!, poly[j]!)) n++;
    }
  return n;
}
function shoelace(p: Vec[]): number {
  let a = 0;
  for (let i = 1; i < p.length; i++) a += p[i - 1]!.x * p[i]!.y - p[i]!.x * p[i - 1]!.y;
  return a / 2;
}
// max turning angle (deg) along an open polyline, with the u (fraction) where it happens
function maxTurn(p: Vec[]): number {
  let best = 0;
  for (let i = 1; i < p.length - 1; i++) {
    const a = Math.atan2(p[i]!.y - p[i - 1]!.y, p[i]!.x - p[i - 1]!.x);
    const b = Math.atan2(p[i + 1]!.y - p[i]!.y, p[i + 1]!.x - p[i]!.x);
    let d = Math.abs(b - a);
    if (d > Math.PI) d = 2 * Math.PI - d;
    best = Math.max(best, (d * 180) / Math.PI);
  }
  return best;
}

    fail = warpDeformPreview(st().project, activeWarpDeformSession()).failure;
    applyWarpDeformTool();
    out(`D4c-${tag}`, { fail, n: worldPolys(obj('c'))[0]?.length, hd: fail ? null : hausdorff([truthEllipse(cx, cy, 3, 3, oracleDeform(s.box, s.handles), 40000)], worldPolys(obj('c')), 0.05), toast: lastToast() });
  }

// ---------- D6 placed tabs ----------
it('D6 placed laser tabs follow the deform', () => {
  const tabLayer = layer('L', 'line', { tabsEnabled: true, tabSizeMm: 2, tabsPerShape: 4 } as Partial<Layer>);
  const fr = [12.5 / 300, 50 / 300, 125 / 300, 200 / 300, 275 / 300];
  const r = denseRect('r', 100, 100, 100, 50, 100, {
    laserTabAnchors: fr.map((pathT) => ({ layerColor: '#000000', pathIndex: 0, polylineIndex: 0, pathT })),
  });
  const placeAt = (t: number): Vec => {
    let d = t * 300;
    const c = [{ x: 100, y: 100 }, { x: 200, y: 100 }, { x: 200, y: 150 }, { x: 100, y: 150 }, { x: 100, y: 100 }];
    for (let i = 1; i < c.length; i++) {
      const L = Math.hypot(c[i]!.x - c[i - 1]!.x, c[i]!.y - c[i - 1]!.y);
      if (d <= L) return { x: c[i - 1]!.x + ((c[i]!.x - c[i - 1]!.x) * d) / L, y: c[i - 1]!.y + ((c[i]!.y - c[i - 1]!.y) * d) / L };
      d -= L;
    }
    return c[0]!;
  };
  const gaps = (moves: LitMove[]) => {
    const res: Vec[] = [];
    for (let i = 1; i < moves.length; i++) {
      const a = moves[i - 1]!.to, b = moves[i]!.from;
      const d = Math.hypot(a.x - b.x, a.y - b.y);
      if (d > 0.5 && d < 5) res.push({ x: r3((a.x + b.x) / 2), y: r3((a.y + b.y) / 2) });
    }
    const wa = moves.at(-1)!.to, wb = moves[0]!.from;
    const wd = Math.hypot(wa.x - wb.x, wa.y - wb.y);
    if (wd > 0.5 && wd < 5) res.push({ x: r3((wa.x + wb.x) / 2), y: r3((wa.y + wb.y) / 2) });
    return res;
  };
  for (const [tag, fn] of [
    ['edge-bulge', moveH({ 1: { x: 0, y: -40 }, 2: { x: 0, y: -40 }, 7: { x: 30, y: 0 } })],
    ['interior+corner', moveH({ 5: { x: 25, y: 20 }, 15: { x: 40, y: 15 } })],
  ] as const) {
    load([r], ['r'], [tabLayer]);
    const g0 = burn(st().project);
    const { box, handles } = deform(fn as (h: Vec[]) => Vec[]);
    applyWarpDeformTool();
    const oracle = oracleDeform(box, handles);
    const want = fr.map((t) => oracle(placeAt(t)));
    const g1 = burn(st().project);
    const got = gaps(g1.moves);
    const err = want.map((w) => r3(Math.min(...got.map((g) => Math.hypot(g.x - w.x, g.y - w.y)))));
    out(`D6-${tag}`, { gapsBefore: gaps(g0.moves).length, nGapsAfter: got.length, want: want.map((w) => ({ x: r3(w.x), y: r3(w.y) })), errMm: err, toast: lastToast(), issues: g1.issues });
  }
});
