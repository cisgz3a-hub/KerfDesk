// ---------- independent oracle ----------
function jarvis(points: ReadonlyArray<Vec>): Vec[] {
  const pts = dedupe(points);
  if (pts.length < 3) return pts;
  let start = pts[0]!;
  for (const p of pts) if (p.x < start.x || (p.x === start.x && p.y < start.y)) start = p;
  const hull: Vec[] = [];
  let cur = start;
  for (let guard = 0; guard < pts.length + 5; guard++) {
    hull.push(cur);
    let cand = pts[0] === cur ? pts[1]! : pts[0]!;
    for (const p of pts) {
      if (p === cur) continue;
      const c = (cand.x - cur.x) * (p.y - cur.y) - (cand.y - cur.y) * (p.x - cur.x);
      const farther = Math.hypot(p.x - cur.x, p.y - cur.y) > Math.hypot(cand.x - cur.x, cand.y - cur.y);
      if (c < 0 || (c === 0 && farther)) cand = p;
    }
    cur = cand;
    if (cur === start) break;
  }
  return hull;
}
function dedupe(points: ReadonlyArray<Vec>): Vec[] {
  const seen = new Map<string, Vec>();
  for (const p of points) seen.set(`${p.x},${p.y}`, p);
  return [...seen.values()];
}
function area(poly: ReadonlyArray<Vec>): number {
  let a = 0;
  for (let i = 0; i < poly.length; i++) {
    const p = poly[i]!;
    const q = poly[(i + 1) % poly.length]!;
    a += p.x * q.y - q.x * p.y;
  }
  return Math.abs(a / 2);
}
function perimeter(poly: ReadonlyArray<Vec>): number {
  let l = 0;
  for (let i = 0; i < poly.length; i++) {
    const p = poly[i]!;
    const q = poly[(i + 1) % poly.length]!;
    l += Math.hypot(q.x - p.x, q.y - p.y);
  }
  return l;
}
/** Distance a point lies outside a convex polygon (0 when inside/on). */
function outsideBy(poly: ReadonlyArray<Vec>, p: Vec): number {
  // orientation
  let s = 0;
  for (let i = 0; i < poly.length; i++) {
    const a = poly[i]!;
    const b = poly[(i + 1) % poly.length]!;
    s += a.x * b.y - b.x * a.y;
  }
  const sign = s >= 0 ? 1 : -1;
  let worst = 0;
  for (let i = 0; i < poly.length; i++) {
    const a = poly[i]!;
    const b = poly[(i + 1) % poly.length]!;
    const len = Math.hypot(b.x - a.x, b.y - a.y);
    if (len === 0) continue;
    const c = (sign * ((b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x))) / len;
    if (c < 0) worst = Math.max(worst, -c);
  }
  return worst;
}
function maxOutside(poly: ReadonlyArray<Vec>, pts: ReadonlyArray<Vec>): number {
  let m = 0;
  for (const p of pts) m = Math.max(m, outsideBy(poly, p));
  return m;
}

  const base = createProject();

function svg(id: string, polylines: ImportedSvg['paths'][number]['polylines'], t: Partial<Transform> = {}, extra: Partial<ImportedSvg> = {}): ImportedSvg {
  const xs = polylines.flatMap((p) => p.points.map((q) => q.x));
  const ys = polylines.flatMap((p) => p.points.map((q) => q.y));
  return {
    kind: 'imported-svg',
    id,
    source: `${id}.svg`,
    bounds: { minX: Math.min(...xs), minY: Math.min(...ys), maxX: Math.max(...xs), maxY: Math.max(...ys) },
    transform: { ...IDENTITY_TRANSFORM, ...t },
    operationIds: ['L'],
    paths: [{ color: '#000000', polylines }],
    ...extra,
  };
}

function outlineRing(): { obj: ImportedSvg; ring: Vec[] } | null {
  const objs = st().project.scene.objects;
  const obj = objs.find((o) => o.kind === 'imported-svg' && o.source === 'Rubber-band outline') as ImportedSvg | undefined;
  if (obj === undefined) return null;
  const pts = obj.paths[0]!.polylines[0]!.points.map((p) => applyTransform(p, obj.transform));
  return { obj, ring: pts.slice(0, -1) };
}
function burn(project: Project) {
  const prepared = prepareOutput(project);
  if (!prepared.ok) return { ok: false, gcode: '', moves: [] as LitMove[], why: JSON.stringify(prepared.preflight).slice(0, 500) };
  const g = emitPreparedGcode(prepared).gcode;
  return { ok: true, gcode: g, moves: litMoves(g, project.device), why: '' };
}
function densify(moves: ReadonlyArray<LitMove>, step = 0.05): Vec[] {
  const res: Vec[] = [];
  for (const m of moves) {
    const n = Math.max(1, Math.ceil(m.len / step));
    for (let k = 0; k <= n; k++) res.push({ x: m.from.x + ((m.to.x - m.from.x) * k) / n, y: m.from.y + ((m.to.y - m.from.y) * k) / n });
  }
  return res;
}
const toasts = () => (useToastStore.getState() as unknown as { toasts: { message: string; variant: string }[] }).toasts.map((t) => `[${t.variant}] ${t.message}`);

/** Compare outline with an oracle point cloud; also burn the source alone and the outline. */
function compare(label: string, oraclePts: Vec[], sourceProject: Project) {
  const o = outlineRing();
  if (o === null) {
    out(`${label}: NO OUTLINE; toasts ${JSON.stringify(toasts())}`);
    return;
  }
  const oracle = jarvis(oraclePts);
  const outsideTrue = maxOutside(o.ring, oraclePts);
  const src = burn(sourceProject);
  const srcPts = densify(src.moves);
  const outsideBurn = srcPts.length ? maxOutside(o.ring, srcPts) : NaN;
  const after = burn(st().project);
  // outline-only lit moves: compare with ring perimeter
  const ringPerim = perimeter(o.ring);
  const outlineMoves = after.moves.filter((m) => {
    const mid = { x: (m.from.x + m.to.x) / 2, y: (m.from.y + m.to.y) / 2 };
    return outsideBy(o.ring, mid) < 1e-3 && ringEdgeDist(o.ring, mid) < 1e-3;
  });
  const lit = outlineMoves.reduce((a, m) => a + m.len, 0);
  const fs = [...new Set(outlineMoves.map((m) => `F${m.f}/S${m.s}`))].join(',');
  out(
    `${label}: hull v=${o.ring.length} area=${f3(area(o.ring))} oracle v=${oracle.length} area=${f3(area(oracle))} ` +
      `dArea=${f3(area(oracle) - area(o.ring))} trueOutside=${f3(outsideTrue)} burnOutside=${f3(outsideBurn)} srcBurnOk=${src.ok} ` +
      `ringPerim=${f3(ringPerim)} outlineLit=${f3(lit)} outlineFS=${fs} afterOk=${after.ok} ${after.why} toasts=${JSON.stringify(toasts())}`,
  );
}
function ringEdgeDist(ring: ReadonlyArray<Vec>, p: Vec): number {
  let best = Infinity;
  for (let i = 0; i < ring.length; i++) {
    const a = ring[i]!;
    const b = ring[(i + 1) % ring.length]!;
    const dx = b.x - a.x;
    const dy = b.y - a.y;
    const L = dx * dx + dy * dy;
    const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / L));
    best = Math.min(best, Math.hypot(a.x + t * dx - p.x, a.y + t * dy - p.y));
  }
  return best;
}
function rectPts(w: number, h: number, t: Partial<Transform>): Vec[] {
  const T = { ...IDENTITY_TRANSFORM, ...t };
  return [{ x: 0, y: 0 }, { x: w, y: 0 }, { x: w, y: h }, { x: 0, y: h }].map((p) => applyTransform(p, T));
}
function ellipsePts(w: number, h: number, t: Partial<Transform>, n = 20000): Vec[] {
  const T = { ...IDENTITY_TRANSFORM, ...t };
  const res: Vec[] = [];
  for (let i = 0; i < n; i++) {
    const a = (2 * Math.PI * i) / n;
    res.push(applyTransform({ x: w / 2 + (w / 2) * Math.cos(a), y: h / 2 + (h / 2) * Math.sin(a) }, T));
  }
  return res;
}

function run(label: string, objects: SceneObject[], layers: Layer[], selected: string[], oracle: Vec[]) {
  const p = proj(objects, layers);
  setProject(p, selected);
  const ok = st().addRubberBandOutline();
  out(`-- ${label}: returned ${ok}`);
  compare(label, oracle, p);
}

function cutStats(g: string) {
  let x = 0, y = 0, z = 0, mode = 0, len = 0;
  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const raw of g.split('\n')) {
    const line = raw.replace(/;.*$/, '').replace(/\(.*?\)/g, '').toUpperCase();
    const w = new Map<string, number>();
    for (const m of line.matchAll(/([GXYZ])\s*(-?\d*\.?\d+)/g)) w.set(m[1]!, Number(m[2]));
    if (w.has('G')) { const gv = w.get('G')!; if ([0, 1, 2, 3].includes(gv)) mode = gv; }
    const nx = w.get('X') ?? x, ny = w.get('Y') ?? y, nz = w.get('Z') ?? z;
    if (mode === 1 && nz < 0 && z < 0 && (nx !== x || ny !== y)) {
      len += Math.hypot(nx - x, ny - y);
      minX = Math.min(minX, x, nx); maxX = Math.max(maxX, x, nx); minY = Math.min(minY, y, ny); maxY = Math.max(maxY, y, ny);
    }
    x = nx; y = ny; z = nz;
  }
  return `cutXY=${len.toFixed(1)}mm bbox=[${minX.toFixed(1)},${minY.toFixed(1)}..${maxX.toFixed(1)},${maxY.toFixed(1)}]`;
}
