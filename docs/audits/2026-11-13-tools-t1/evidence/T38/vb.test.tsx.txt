function Probe() {
  const project = useStore((s) => s.project);
  shown = useCanvasDisplay(project, false, null, none).project;
  return null;
}
async function settle(done: () => boolean) {
  for (let i = 0; i < 100 && !done(); i++)
    await act(async () => {
      await new Promise((r) => setTimeout(r, 5));
    });
  expect(done()).toBe(true);
}
function qr(project: Project) {
  return project.scene.objects.find(isBarcodeObject)!;
}
function scan(object: SceneObject, original: BarcodeObject, dx = 0, dy = 0) {
  if (!('paths' in object)) throw new Error('No vector');
  const spec = original.spec;
  const size =
    Math.round((original.bounds.maxX - original.bounds.minX) / spec.moduleMm) -
    2 * spec.quietZoneModules;
  const quiet = spec.quietZoneModules * spec.moduleMm;
  const polylines = object.paths.flatMap((p) =>
    p.polylines.map((l) => ({
      ...l,
      points: l.points.map((p) => applyTransform(p, object.transform)),
    })),
  );
  const grid = sampleGrid(polylines, size, size, spec.moduleMm, {
    x: original.transform.x + quiet + dx,
    y: original.transform.y + quiet + dy,
  });
  const decoded = decodeQrModules(grid, size);
  return decoded.ok ? decoded.text : null;
}
async function insert() {
  useStore.getState().setVariableSettings({ serialValue: 1, advancement: 'manual' });
  const draft = {
    ...draftFromSpec(defaultBarcodeSpec('qr')),
    data: 'SN-{{serial:4}}',
    variable: true,
  };
  const standIn: ShapeObject = {
    kind: 'shape',
    id: 'barcode-preview',
    spec: defaultBarcodeSpec('qr'),
    color: '#000000',
    bounds: { minX: 0, minY: 0, maxX: 0, maxY: 0 },
    transform: IDENTITY_TRANSFORM,
    paths: [],
  };
  const preview = previewBarcode(draft, {
    project: useStore.getState().project,
    object: standIn,
    now: new Date(),
  });
  if (preview.kind !== 'ready') throw new Error(preview.kind);
  expect(await commitBarcode({ spec: preview.spec, value: preview.value })).toMatchObject({
    ok: true,
  });
  const original = qr(useStore.getState().project);
  expect(original.transform).toMatchObject({ rotationDeg: 0, scaleX: 1, scaleY: 1 });
  useStore.setState({ selectedObjectId: original.id, additionalSelectedIds: new Set() });
  root = createRoot(document.createElement('div'));
  await act(async () => root!.render(createElement(Probe)));
  await settle(() => shown !== null && scan(qr(shown), original) === 'SN-0001');
  return original;
}

it('unchanged serial preview and Apply agree; Cancel preserves current variable barcode', async () => {
  const original = await insert();
  act(() => startWarpDeformTool('warp'));
  let start = activeWarpDeformSession()!;
  act(() =>
    useWarpDeformSession
      .getState()
      .setHandles(start.handles.map((p) => ({ x: p.x + 10, y: p.y + 5 }))),
  );
  expect(scan(shown!.scene.objects.find((o) => o.id === original.id)!, original, 10, 5)).toBe(
    'SN-0001',
  );
  act(() => applyWarpDeformTool());
  expect(
    scan(
      useStore.getState().project.scene.objects.find((o) => o.id === original.id)!,
      original,
      10,
      5,
    ),
  ).toBe('SN-0001');
  act(() => useStore.getState().undo());
  act(() => useStore.getState().advanceVariablesManually());
  await settle(() => shown !== null && scan(qr(shown), original) === 'SN-0002');
  const before = useStore.getState().project;
  const depth = useStore.getState().undoStack.length;
  act(() => startWarpDeformTool('warp'));
  start = activeWarpDeformSession()!;
  act(() =>
    useWarpDeformSession
      .getState()
      .setHandles(start.handles.map((p) => ({ x: p.x + 10, y: p.y + 5 }))),
  );
  act(() => cancelWarpDeformTool());
  expect(useStore.getState().project).toBe(before);
  expect(useStore.getState().undoStack.length).toBe(depth);
  expect(scan(qr(shown!), original)).toBe('SN-0002');
});
