// ---------- independent oracle (Heckbert unit-square -> quad) ----------
type Box = { minX: number; minY: number; maxX: number; maxY: number };
function oracleProjective(box: Box, q: Vec[]): (p: Vec) => Vec {
  const [p0, p1, p2, p3] = q as [Vec, Vec, Vec, Vec];
  const dx1 = p1.x - p2.x, dx2 = p3.x - p2.x, dx3 = p0.x - p1.x + p2.x - p3.x;
  const dy1 = p1.y - p2.y, dy2 = p3.y - p2.y, dy3 = p0.y - p1.y + p2.y - p3.y;
  const den = dx1 * dy2 - dx2 * dy1;
  const g = (dx3 * dy2 - dx2 * dy3) / den;
  const h = (dx1 * dy3 - dx3 * dy1) / den;
  const a = p1.x - p0.x + g * p1.x, b = p3.x - p0.x + h * p3.x, c = p0.x;
  const d = p1.y - p0.y + g * p1.y, e = p3.y - p0.y + h * p3.y, f = p0.y;
  return (p) => {
    const u = (p.x - box.minX) / (box.maxX - box.minX);
    const v = (p.y - box.minY) / (box.maxY - box.minY);
    const w = g * u + h * v + 1;
    return { x: (a * u + b * v + c) / w, y: (d * u + e * v + f) / w };
  };
}
function oracleBilinear(box: Box, q: Vec[]): (p: Vec) => Vec {
  const [p0, p1, p2, p3] = q as [Vec, Vec, Vec, Vec];
  return (p) => {
    const u = (p.x - box.minX) / (box.maxX - box.minX);
    const v = (p.y - box.minY) / (box.maxY - box.minY);
    return {
      x: (1 - u) * (1 - v) * p0.x + u * (1 - v) * p1.x + u * v * p2.x + (1 - u) * v * p3.x,
      y: (1 - u) * (1 - v) * p0.y + u * (1 - v) * p1.y + u * v * p2.y + (1 - u) * v * p3.y,
    };
  };
}
function segDist(p: Vec, a: Vec, b: Vec): number {
  const dx = b.x - a.x, dy = b.y - a.y;
  const l2 = dx * dx + dy * dy;
  const t = l2 === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / l2));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}
function distToPoly(p: Vec, pts: ReadonlyArray<Vec>): number {
  let best = Infinity;
  for (let i = 1; i < pts.length; i++) best = Math.min(best, segDist(p, pts[i - 1]!, pts[i]!));
  return best;
}
function densify(pts: ReadonlyArray<Vec>, step: number): Vec[] {
  const res: Vec[] = [];
  for (let i = 1; i < pts.length; i++) {
    const a = pts[i - 1]!, b = pts[i]!;
    const n = Math.max(1, Math.ceil(Math.hypot(b.x - a.x, b.y - a.y) / step));
    for (let k = 0; k < n; k++) res.push({ x: a.x + ((b.x - a.x) * k) / n, y: a.y + ((b.y - a.y) * k) / n });
  }
  res.push(pts.at(-1)!);
  return res;
}
function worldPolys(o: SceneObject): Vec[][] {
  if (!('paths' in o)) return [];
  return o.paths.flatMap((p) =>
    p.polylines.map((l) => {
      const w = l.points.map((q) => applyTransform(q, o.transform));
      return l.closed && w.length > 0 ? [...w, w[0]!] : w;
    }),
  );
}
// two-way Hausdorff between oracle-mapped dense source and output polylines
function hausdorff(expected: Vec[][], actual: Vec[][]): { fwd: number; back: number } {
  let fwd = 0, back = 0;
  const actAll = actual;
  for (const poly of expected)
    for (const p of densify(poly, 0.25)) {
      let best = Infinity;
      for (const a of actAll) best = Math.min(best, distToPoly(p, a));
      fwd = Math.max(fwd, best);
    }
  for (const poly of actual)
    for (const p of densify(poly, 0.25)) {
      let best = Infinity;
      for (const e of expected) best = Math.min(best, distToPoly(p, e));
      back = Math.max(back, best);
    }
  return { fwd: r3(fwd), back: r3(back) };
}

// ---------- scene helpers ----------
function layer(id = 'L', mode: 'line' | 'fill' = 'line', extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id, color: '#000000', name: id, mode }), speed: 1000, power: 40, passes: 1, ...extra } as Layer;
}
function load(objects: SceneObject[], selected: string[], layers: Layer[] = [layer()]): void {
  const base = createProject();
  const project: Project = {
    ...base,
    optimization: { ...DEFAULT_PROJECT_OPTIMIZATION },
    scene: { ...base.scene, objects, layers, groups: [] },
  };
  useStore.setState({
    project,
    selectedObjectId: selected[0] ?? null,
    additionalSelectedIds: new Set(selected.slice(1)),
    undoStack: [],
    redoStack: [],
    dirty: false,
  });
}
function denseRect(id: string, x: number, y: number, w: number, h: number, step = 5, extra: Partial<ImportedSvg> = {}): ImportedSvg {
  const corners = [{ x, y }, { x: x + w, y }, { x: x + w, y: y + h }, { x, y: y + h }, { x, y }];
  const pts = densify(corners, step).slice(0, -1);
  return {
    kind: 'imported-svg',
    id,
    source: `${id}.svg`,
    bounds: { minX: x, minY: y, maxX: x + w, maxY: y + h },
    transform: IDENTITY_TRANSFORM,
    operationIds: ['L'],
    paths: [{ color: '#000000', operationIds: ['L'], polylines: [{ closed: true, points: pts }] }],
    ...extra,
  };
}
function lastToast(): string {
  const toasts = (useToastStore.getState() as unknown as { toasts: { message: string }[] }).toasts;
  return toasts.at(-1)?.message ?? '';
}
function warp(dragged: (h: Vec[]) => Vec[]): { box: Box; handles: Vec[] } {
  startWarpDeformTool('warp');
  const s = activeWarpDeformSession();
  if (s === null) throw new Error(`no session: ${lastToast()}`);
  const handles = dragged(s.handles.map((p) => ({ ...p })));
  useWarpDeformSession.getState().setHandles(handles);
  return { box: { ...s.box }, handles };
}
function burn(project: Project) {
  const prepared = prepareOutput(project);
  if (!prepared.ok) return { ok: false, moves: [] as LitMove[], issues: JSON.stringify(prepared.preflight).slice(0, 400) };
  const emitted = emitPreparedGcode(prepared);
  return { ok: true, moves: litMoves(emitted.gcode, project.device), issues: '' };
}
function obj(id: string): SceneObject {
  return st().project.scene.objects.find((o) => o.id === id)!;
}
function selfCrossings(poly: Vec[]): number {
  let n = 0;
  const inter = (a: Vec, b: Vec, c: Vec, d: Vec) => {
    const o = (p: Vec, q: Vec, r: Vec) => (q.x - p.x) * (r.y - p.y) - (q.y - p.y) * (r.x - p.x);
    return o(a, b, c) * o(a, b, d) < 0 && o(c, d, a) * o(c, d, b) < 0;
  };
  for (let i = 1; i < poly.length; i++)
    for (let j = i + 2; j < poly.length; j++) {
      if (i === 1 && j === poly.length - 1) continue;
      if (inter(poly[i - 1]!, poly[i]!, poly[j - 1]!, poly[j]!)) n++;
    }
  return n;
}
function shoelace(p: Vec[]): number {
  let a = 0;
  for (let i = 1; i < p.length; i++) a += p[i - 1]!.x * p[i]!.y - p[i]!.x * p[i - 1]!.y;
  return a / 2;
}

// ---------- S2 single-corner perspective drag ----------
it('S2 single corner + G-code', () => {
  load([denseRect('r', 100, 100, 100, 50)], ['r']);
  const before = st().project;
  const gBefore = burn(before);
  const { box, handles } = warp((h) => h.map((p, i) => (i === 1 ? { x: p.x + 40, y: p.y - 20 } : p)));
  const preview = warpDeformPreview(before, activeWarpDeformSession());
  applyWarpDeformTool();
  const after = obj('r');
  const oracle = oracleProjective(box, handles);
  const src = worldPolys(before.scene.objects[0]!)[0]!;
  const act = worldPolys(after)[0]!;
  let maxVertex = 0;
  for (let i = 0; i < src.length; i++) {
    const o = oracle(src[i]!);
    maxVertex = Math.max(maxVertex, Math.hypot(o.x - act[i]!.x, o.y - act[i]!.y));
  }
  // edge midpoints of the four sides
  const mids = [{ x: 150, y: 100 }, { x: 200, y: 125 }, { x: 150, y: 150 }, { x: 100, y: 125 }].map((m) => {
    const o = oracle(m);
    return r3(distToPoly(o, act));
  });
  const prevObj = preview.project.scene.objects.find((o) => o.id === 'r')!;
  const gAfter = burn(st().project);
  const expectedPerim = (() => {
    const q = [...handles, handles[0]!];
    let s = 0;
    for (let i = 1; i < q.length; i++) s += Math.hypot(q[i]!.x - q[i - 1]!.x, q[i]!.y - q[i - 1]!.y);
    return s;
  })();
  const litMax = Math.max(...gAfter.moves.flatMap((m) => [m.from, m.to]).map((p) => {
    let best = Infinity;
    const q = [...handles, handles[0]!];
    for (let i = 1; i < q.length; i++) best = Math.min(best, segDist(p, q[i - 1]!, q[i]!));
    return best;
  }));
  out('S2', {
    nSrc: src.length, nAct: act.length, maxVertexErrMm: maxVertex, midDistMm: mids,
    previewEqualsApply: JSON.stringify(prevObj) === JSON.stringify(after),
    litBefore: r3(totalLen(gBefore.moves)), litAfter: r3(totalLen(gAfter.moves)), expectedPerim: r3(expectedPerim),
    litMaxOffQuadMm: r3(litMax), toast: lastToast(), undo: st().undoStack.map((u: { label?: string }) => u.label ?? '?'),
  });
  // undo / redo / save-reopen
  const applied = st().project;
  st().undo();
  const undoneSame = st().project.scene.objects[0] === before.scene.objects[0];
  st().redo();
  const redoSame = st().project === applied;
  const reopened = deserializeProject(serializeProject(st().project));
  const rr = reopened as unknown as { kind: string; project?: Project };
  const ro = rr.project?.scene.objects[0] ?? null;
  out('S2-hist', { undoneSame, redoSame, reopenKind: rr.kind, reopenSame: ro !== null && JSON.stringify(worldPolys(ro)) === JSON.stringify(worldPolys(applied.scene.objects[0]!)) });
});

  });
  load([r], ['r'], [tabLayer]);
  const before = st().project;
  const gaps = (moves: LitMove[]) => {
    const res: Vec[] = [];
    for (let i = 1; i < moves.length; i++) {
      const a = moves[i - 1]!.to, b = moves[i]!.from;
      const d = Math.hypot(a.x - b.x, a.y - b.y);
      if (d > 0.5 && d < 5) res.push({ x: r3((a.x + b.x) / 2), y: r3((a.y + b.y) / 2) });
    }
    const wa = moves.at(-1)!.to, wb = moves[0]!.from;
    const wd = Math.hypot(wa.x - wb.x, wa.y - wb.y);
    if (wd > 0.5 && wd < 5) res.push({ x: r3((wa.x + wb.x) / 2), y: r3((wa.y + wb.y) / 2) });
    return res;
  };
  const g0 = burn(before);
  const { box, handles } = warp((h) => h.map((p, i) => (i === 1 ? { x: p.x + 60, y: p.y } : p)));
  applyWarpDeformTool();
  const oracle = oracleProjective(box, handles);
  const placeAt = (t: number): Vec => {
    let d = t * 300;
    const c = [{ x: 100, y: 100 }, { x: 200, y: 100 }, { x: 200, y: 150 }, { x: 100, y: 150 }, { x: 100, y: 100 }];
    for (let i = 1; i < c.length; i++) {
      const L = Math.hypot(c[i]!.x - c[i - 1]!.x, c[i]!.y - c[i - 1]!.y);
      if (d <= L) return { x: c[i - 1]!.x + ((c[i]!.x - c[i - 1]!.x) * d) / L, y: c[i - 1]!.y + ((c[i]!.y - c[i - 1]!.y) * d) / L };
      d -= L;
    }
    return c[0]!;
  };
  const want = fr.map((t) => oracle(placeAt(t)));
  const g1 = burn(st().project);
  const got = gaps(g1.moves);
  const err = want.map((w) => r3(Math.min(...got.map((g) => Math.hypot(g.x - w.x, g.y - w.y)))));
  out('S12', { gapsBefore: gaps(g0.moves), want: want.map((w) => ({ x: r3(w.x), y: r3(w.y) })), got, errMm: err, anchorsAfter: (obj('r') as ImportedSvg).laserTabAnchors, toast: lastToast(), issues: g1.issues });
});
