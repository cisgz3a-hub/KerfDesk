function burn(p: Project): Seg[] {
  const prepared = prepareOutput(p);
  if (!prepared.ok) return [];
  return readGcode(emitPreparedGcode(prepared).gcode);
}
async function snapBurn(p: Project) {
  const prepared = await prepareOutputSnapshot(p, { clock: () => new Date(0), renderVariableText: render });
  if (!prepared.ok) return { ok: false as const, segs: [] as Seg[], warn: [] as string[], why: JSON.stringify(prepared).slice(0, 300) };
  const anyP = prepared as unknown as { warnings?: unknown; preflight?: { issues?: { message?: string; severity?: string }[] } };
  const warn = [JSON.stringify(anyP.warnings ?? null).slice(0, 300), ...(anyP.preflight?.issues ?? []).map((i) => `${i.severity}:${i.message}`)];
  return { ok: true as const, segs: readGcode(emitPreparedGcode(prepared).gcode), warn };
}

/** Start Warp on the current selection and move handles with `move`; returns null when refused. */
function warpWith(move: (h: ReadonlyArray<{ x: number; y: number }>) => { x: number; y: number }[]) {
  startWarpDeformTool('warp');
  const s = activeWarpDeformSession();
  if (s === null) return null;
  useWarpDeformSession.getState().setHandles(move(s.handles));
  const objectIds = [...s.objectIds];
  applyWarpDeformTool();
  return { objectIds, box: s.box };
}

// ---------- the jig, made through the real editor (as T28's refuter did) ----------
function buildJig() {
  const A = op('art', 'line', 2222, 90, '#ff0000');
  const E = op('eng', 'line', 1111, 10, '#0000ff');
  load(project([], [E, A]), []);
  let session = createJigEditorSession(st() as never);
  const d1 = disc('C1', 8, 10, 10, ['art'], '#ff0000');
  const d2 = disc('C2', 8, 18, 12, ['art'], '#ff0000');
  const sq = rect('SQ', '#0000ff', 5, 5, 12, 22, ['eng']);
  const dp = session.draft.project;
  session = changeJigDraft(session, { project: { ...dp, scene: { ...dp.scene, objects: [...dp.scene.objects, d1, d2, sq], layers: [...dp.scene.layers.filter((l) => l.id !== 'art' && l.id !== 'eng'), E, A] } } });
  session = replaceJigLayout(session, { outline: { kind: 'rectangle', widthMm: 60, heightMm: 50 }, rows: 1, columns: 2, spacingX: 10, spacingY: 10 });
  session = selectJigObjects(session, new Set(['C1', 'C2', 'SQ']));
  session = centerJigArtwork(session);
  const committed = commitJigEditorSession(st() as never, session);
  if (committed.kind !== 'saved') throw new Error(`commit ${committed.reason}`);
  useStore.setState(committed.change as never);
  st().setRegistrationOutput('artwork' as never);
  useStore.setState({ selectedObjectId: null, additionalSelectedIds: new Set() });
  const p0 = st().project;
  const boxes = findRegistrationBoxes(p0.scene);
  const bx = boxes.map((b) => b.transform.x).sort((a, b) => a - b);
  const pitch = bx[1]! - bx[0]!;
  const midX = bx[1]! - 5;
  const region = (x: number) => (x < midX ? 'A' : 'B');
  return { p0, boxes, pitch, midX, region };
}
function jigStats(tag: string, p: Project, jig: ReturnType<typeof buildJig>) {
  const segs = burn(p);
  const a = segs.filter((g) => (g.x0 + g.x1) / 2 < jig.midX);
  const b = segs.filter((g) => (g.x0 + g.x1) / 2 >= jig.midX);
  const sa = segSet(a), sb = segSet(b, jig.pitch, 0);
  const boxes = findRegistrationBoxes(p.scene);
  out(`${tag}: boxes ${boxes.length} (locked ${boxes.map((x) => x.locked === true).join(',')}); jig runs ${registrationJigCompilationRuns(p.scene)?.length ?? 0}; ` +
    `lit A ${f3(litWhere(segs, (x) => x < jig.midX))} B ${f3(litWhere(segs, (x) => x >= jig.midX))}; A and B pieces identical after shift: ${JSON.stringify(sa) === JSON.stringify(sb)}`);
  out(`${tag}: order ${order(segs, jig.region).join(' > ')}`);
  out(`${tag}: kinds ${p.scene.objects.map((o) => `${o.id.slice(0, 30)}:${o.kind}${o.locked ? '[L]' : ''}`).join(' | ')}`);
}

// ---------- tabs spot check (R3-P2-2) with an own homography oracle ----------
function homography(src: { x: number; y: number }[], dst: { x: number; y: number }[]) {
  // 8x8 DLT solve, h33 = 1
  const A: number[][] = [];
  const bv: number[] = [];
  for (let i = 0; i < 4; i++) {
    const { x, y } = src[i]!, { x: u, y: v } = dst[i]!;
    A.push([x, y, 1, 0, 0, 0, -u * x, -u * y]); bv.push(u);
    A.push([0, 0, 0, x, y, 1, -v * x, -v * y]); bv.push(v);
  }
  const n = 8;
  for (let c = 0; c < n; c++) {
    let p = c;
    for (let r = c + 1; r < n; r++) if (Math.abs(A[r]![c]!) > Math.abs(A[p]![c]!)) p = r;
    [A[c], A[p]] = [A[p]!, A[c]!]; [bv[c], bv[p]] = [bv[p]!, bv[c]!];
    for (let r = 0; r < n; r++) {
      if (r === c) continue;
      const k = A[r]![c]! / A[c]![c]!;
      for (let j = c; j < n; j++) A[r]![j]! -= k * A[c]![j]!;
      bv[r]! -= k * bv[c]!;
    }
  }
  const h = bv.map((v, i) => v / A[i]![i]!);
  return (p: { x: number; y: number }) => {
    const w = h[6]! * p.x + h[7]! * p.y + 1;
    return { x: (h[0]! * p.x + h[1]! * p.y + h[2]!) / w, y: (h[3]! * p.x + h[4]! * p.y + h[5]!) / w };
  };
}
