// T30 Join Paths: geometry against an independent oracle.
import { it } from 'vitest';
import { createLayer, IDENTITY_TRANSFORM, type ImportedSvg, type Transform } from '../t29-main/src/core/scene';
import { joinOpenVectorPaths } from '../t29-main/src/core/geometry/vector-path-join';
import { hausdorff, len, sample, xf, type Curve, type V } from './oracle';
import { logger, f3 } from './common';

const out = logger('C:/Users/stolk/kd-t1/zz-t30/t30-geom.log');
const cut = createLayer({ id: 'cut', color: '#000000' });
const T = (t: Partial<Transform> = {}) => ({ ...IDENTITY_TRANSFORM, ...t });
const art = (id: string, curves: Curve[], t: Partial<Transform> = {}, extra: Partial<ImportedSvg> = {}): ImportedSvg => ({
  kind: 'imported-svg', id, source: id, operationIds: ['cut'],
  bounds: { minX: 0, minY: 0, maxX: 100, maxY: 100 }, transform: T(t),
  paths: [{ color: '#000000', polylines: [], curves }] as never, ...extra,
});
const poly = (...pts: V[]): Curve => ({ start: pts[0]!, segments: pts.slice(1).map((to) => ({ kind: 'line' as const, to })), closed: false });
const join = (objs: ImportedSvg[], tol = 0.05) => {
  const r = joinOpenVectorPaths(objs, [cut], tol);
  if (r.kind === 'error') return { err: r.error.message } as const;
  return r.value;
};
const curvesOf = (objs: ReadonlyArray<{ paths: ReadonlyArray<{ curves?: ReadonlyArray<Curve> }> }>) =>
  objs.flatMap((o) => o.paths.flatMap((p) => (p.curves ?? []) as Curve[]));
const summary = (r: ReturnType<typeof join>) =>
  'err' in r ? `ERR ${r.err}` : `joins ${r.joins} closures ${r.closures} remaining ${r.remainingOpenPaths} ambiguous ${r.ambiguousEndpoints} objects ${r.objects.length}`;

it('G1 tolerance boundary (gap at, under and over the maximum gap)', () => {
  for (const [gap, tol] of [[0.05, 0.05], [0.0499, 0.05], [0.0501, 0.05], [0.1, 0.1], [0.3, 0.3], [0.5, 0.5], [0, 0], [1e-10, 0]] as const) {
    const a = art('a', [poly({ x: 10, y: 3 }, { x: 10.3, y: 3 })]);
    const b = art('b', [poly({ x: 10.3 + gap, y: 3 }, { x: 20, y: 3 })]);
    const actualGap = 10.3 + gap - 10.3;
    out(`G1 gap typed ${gap} (float gap ${actualGap}) tol ${tol}: ${summary(join([a, b], tol))}`);
  }
  // same, gap measured at x ~ 100 mm (typical bed coordinates)
  for (const x0 of [0, 100.1, 333.3]) {
    const a = art('a', [poly({ x: x0 - 10, y: 0 }, { x: x0, y: 0 })]);
    const b = art('b', [poly({ x: x0 + 0.05, y: 0 }, { x: x0 + 10, y: 0 })]);
    out(`G1b gap 0.05 at x=${x0} (float ${x0 + 0.05 - x0}) tol 0.05: ${summary(join([a, b], 0.05))}`);
  }
  for (const bad of [-1, NaN, Infinity, 1e9]) out(`G1c tol ${bad}: ${summary(join([art('a', [poly({ x: 0, y: 0 }, { x: 1, y: 0 })]), art('b', [poly({ x: 500, y: 0 }, { x: 501, y: 0 })])], bad))}`);
});

it('G2 direction, endpoints kept, connector exactly the gap', () => {
  // A: 0->10, B reversed: 20 -> 10.03, C: 20.02 -> 20.02,10  (chain A + B(rev) + C)
  const A = poly({ x: 0, y: 0 }, { x: 10, y: 0 });
  const B = poly({ x: 20, y: 0 }, { x: 10.03, y: 0 });
  const C = poly({ x: 20.02, y: 0.01 }, { x: 20.02, y: 10 });
  const r = join([art('C', [C]), art('A', [A]), art('B', [B])]);
  out(`G2 ${summary(r)}`);
  if ('err' in r) return;
  const cs = curvesOf(r.objects);
  const pts = sample(cs[0]!);
  const srcLen = len(sample(A)) + len(sample(B)) + len(sample(C));
  const gaps = Math.hypot(0.03, 0) + Math.hypot(0.02, 0.01);
  out(`G2 result id ${r.objects.map((o) => o.id)}; points ${pts.map((p) => `(${p.x},${p.y})`).join(' ')}`);
  out(`G2 length ${f3(len(pts))} oracle ${f3(srcLen + gaps)} (sources ${f3(srcLen)} + gaps ${f3(gaps)})`);
  const srcPts = [...sample(A), ...sample(B), ...sample(C)];
  const moved = srcPts.filter((p) => !pts.some((q) => q.x === p.x && q.y === p.y));
  out(`G2 source vertices not found bit-exact in result: ${moved.length}`);
});

it('G3 ambiguous junctions: T, X, Y with one near and one far candidate', () => {
  const o = { x: 50, y: 50 };
  const T3 = [art('a', [poly({ x: 40, y: 50 }, o)]), art('b', [poly({ x: 50.01, y: 50 }, { x: 60, y: 50 })]), art('c', [poly({ x: 50, y: 50.02 }, { x: 50, y: 60 })])];
  out(`G3 T (3 ends within 0.05): ${summary(join(T3))}`);
  const X4 = [...T3, art('d', [poly({ x: 50, y: 49.99 }, { x: 50, y: 40 })])];
  out(`G3 X (4 ends): ${summary(join(X4))}`);
  // 3 ends: two within tol of each other, third at 0.06 from one and 0.04 from the other
  const Y = [art('a', [poly({ x: 40, y: 50 }, { x: 50, y: 50 })]), art('b', [poly({ x: 50.03, y: 50 }, { x: 60, y: 50 })]), art('c', [poly({ x: 50.07, y: 50 }, { x: 50.07, y: 60 })])];
  out(`G3 chain-ish a.end-b.start 0.03, b.start-c.start 0.04, a.end-c.start 0.07: ${summary(join(Y))}`);
});

it('G4 closing on itself, closed chains, doubled lines', () => {
  const sq = poly({ x: 0, y: 0 }, { x: 10, y: 0 }, { x: 10, y: 10 }, { x: 0, y: 10 }, { x: 0, y: 0.02 });
  let r = join([art('sq', [sq])]);
  out(`G4a open square gap 0.02: ${summary(r)}`);
  if (!('err' in r)) {
    const c = curvesOf(r.objects)[0]!;
    out(`G4a closed=${c.closed}; segments ${c.segments.length}; length ${f3(len(sample(c), c.closed))} oracle 40.000`);
  }
  const four = [
    art('e1', [poly({ x: 0, y: 0 }, { x: 10, y: 0 })]),
    art('e2', [poly({ x: 10.01, y: 0 }, { x: 10, y: 10 })]),
    art('e3', [poly({ x: 0, y: 10 }, { x: 9.99, y: 10 })]), // reversed
    art('e4', [poly({ x: 0, y: 10 - 0.001 }, { x: 0, y: 0.03 })]),
  ];
  r = join(four);
  out(`G4b four edges, one reversed: ${summary(r)}`);
  if (!('err' in r)) {
    const c = curvesOf(r.objects)[0]!;
    const pts = sample(c);
    let area = 0;
    for (let i = 0; i < pts.length; i++) {
      const p = pts[i]!;
      const q = pts[(i + 1) % pts.length]!;
      area += (p.x * q.y - q.x * p.y) / 2;
    }
    out(`G4b closed=${c.closed}; |area| ${f3(Math.abs(area))} oracle ~100; length ${f3(len(pts, true))} oracle ~${f3(40 + 0.01 + 0.01 + 0.001 + 0.03)}`);
  }
  // doubled line (DXF double entity)
  r = join([art('d1', [poly({ x: 0, y: 0 }, { x: 30, y: 0 })]), art('d2', [poly({ x: 0, y: 0 }, { x: 30, y: 0 })])]);
  out(`G4c doubled identical line: ${summary(r)}`);
  if (!('err' in r)) {
    const c = curvesOf(r.objects)[0];
    if (c) out(`G4c result closed=${c.closed}; points ${sample(c).map((p) => `(${p.x},${p.y})`).join(' ')}`);
  }
  // out-and-back single path (retrace), 2 segments
  r = join([art('rb', [poly({ x: 0, y: 0 }, { x: 30, y: 0 }, { x: 0, y: 0.01 })])]);
  out(`G4d out-and-back single path: ${summary(r)}`);
  if (!('err' in r)) {
    const c = curvesOf(r.objects)[0];
    if (c) out(`G4d closed=${c.closed}`);
  }
  // overlapping collinear segments sharing one end
  r = join([art('o1', [poly({ x: 0, y: 0 }, { x: 30, y: 0 })]), art('o2', [poly({ x: 30, y: 0 }, { x: 10, y: 0 })])]);
  out(`G4e overlapping collinear (o2 doubles back over o1): ${summary(r)}`);
});

it('G5 curves and arcs under rotate, non-uniform scale and mirror', () => {
  const cases: Array<[string, Partial<Transform>]> = [
    ['identity', {}],
    ['rot 33', { rotationDeg: 33, x: 50, y: 40 }],
    ['scale 3x0.5', { scaleX: 3, scaleY: 0.5, x: 10 }],
    ['mirrorX + rot 120 + scale 2x1.3', { mirrorX: true, rotationDeg: 120, scaleX: 2, scaleY: 1.3, x: 100, y: 100 }],
    ['mirrorY scale 0.2x4', { mirrorY: true, scaleX: 0.2, scaleY: 4 }],
  ];
  for (const [name, t] of cases) {
    const tt = T(t);
    // a: cubic + arc, local; b: arc continuing, gap 0.01 mm in WORLD space
    const a: Curve = {
      start: { x: 0, y: 0 },
      segments: [
        { kind: 'cubic', control1: { x: 3, y: 6 }, control2: { x: 7, y: -4 }, to: { x: 10, y: 2 } },
        { kind: 'elliptical-arc', radiusX: 5, radiusY: 3, rotationDeg: 20, largeArc: false, sweep: true, to: { x: 16, y: 6 } },
      ],
      closed: false,
    };
    const aEnd = xf({ x: 16, y: 6 }, tt);
    const bStartWorld = { x: aEnd.x + 0.01, y: aEnd.y };
    // b is in world (identity) so the gap is known in mm
    const b: Curve = {
      start: bStartWorld,
      segments: [{ kind: 'elliptical-arc', radiusX: 8, radiusY: 4, rotationDeg: -30, largeArc: true, sweep: false, to: { x: bStartWorld.x + 6, y: bStartWorld.y + 5 } }],
      closed: false,
    };
    const r = join([art('a', [a], t), art('b', [b])]);
    if ('err' in r) {
      out(`G5 ${name}: ERR ${r.err}`);
      continue;
    }
    const c = curvesOf(r.objects)[0]!;
    const oracle = [...sample(a).map((p) => xf(p, tt)), ...sample(b)];
    const got = sample(c);
    const d1 = hausdorff(oracle, got);
    const d2 = hausdorff(got, oracle);
    out(`G5 ${name}: ${summary(r)}; kinds ${c.segments.map((s) => s.kind[0]).join('')}; hausdorff oracle->result ${d1.toExponential(2)} result->oracle ${d2.toExponential(2)} (gap bridge 0.01 expected <=0.01)`);
  }
});

it('G6 tolerance in world mm after scale', () => {
  // local gap 0.004 mm, scaled x10 = 0.04 mm world -> joins at 0.05; local 0.006 -> 0.06 -> no
  for (const lg of [0.004, 0.006]) {
    const r = join([art('a', [poly({ x: 0, y: 0 }, { x: 1, y: 0 })], { scaleX: 10, scaleY: 10 }), art('b', [poly({ x: 1 + lg, y: 0 }, { x: 2, y: 0 })], { scaleX: 10, scaleY: 10 })]);
    out(`G6 local gap ${lg} x10: ${summary(r)}`);
  }
  // shrunk 0.01: local gap 3 mm -> 0.03 world -> joins
  const r = join([art('a', [poly({ x: 0, y: 0 }, { x: 100, y: 0 })], { scaleX: 0.01, scaleY: 0.01 }), art('b', [poly({ x: 103, y: 0 }, { x: 200, y: 0 })], { scaleX: 0.01, scaleY: 0.01 })]);
  out(`G6 local gap 3 x0.01: ${summary(r)}`);
});

it('G7 degenerate input', () => {
  const z = art('z', [{ start: { x: 5, y: 5 }, segments: [{ kind: 'line', to: { x: 5, y: 5 } }], closed: false }]);
  const a = art('a', [poly({ x: 0, y: 5 }, { x: 5, y: 5 })]);
  const b = art('b', [poly({ x: 5, y: 5 }, { x: 10, y: 5 })]);
  out(`G7a zero-length piece at a junction: ${summary(join([a, z, b]))}`);
  out(`G7b zero-length alone: ${summary(join([z]))}`);
  const empty = art('e', [{ start: { x: 5, y: 5 }, segments: [], closed: false }]);
  out(`G7c no-segment subpath: ${summary(join([empty, a]))}`);
  out(`G7d zero scale: ${summary(join([art('s', [poly({ x: 0, y: 0 }, { x: 1, y: 0 })], { scaleX: 0 }), a]))}`);
  out(`G7e NaN point: ${summary(join([art('n', [poly({ x: NaN, y: 0 }, { x: 1, y: 0 })]), a]))}`);
  const tiny = art('t', [poly({ x: 0, y: 0 }, { x: 0.01, y: 0 }, { x: 0.01, y: 0.01 })]);
  out(`G7f 3-point open path 0.01 mm long, tol 0.05: ${summary(join([tiny]))}`);
  const r = join([tiny]);
  if (!('err' in r)) out(`G7f closed=${curvesOf(r.objects)[0]?.closed}`);
});

it('G8 performance: many segments', () => {
  for (const n of [1000, 10000, 50000]) {
    const objs: ImportedSvg[] = [];
    // n short collinear pieces in shuffled order, alternate reversed
    const order = Array.from({ length: n }, (_, i) => i).sort((p, q) => ((p * 7919) % n) - ((q * 7919) % n));
    for (const i of order) {
      const s = { x: i, y: 0 };
      const e = { x: i + 1, y: 0 };
      objs.push(art(`p${i}`, [i % 2 ? poly(e, s) : poly(s, e)]));
    }
    // single object, n subpaths
    const one = art('one', order.map((i) => (i % 2 ? poly({ x: i + 1, y: 0 }, { x: i, y: 0 }) : poly({ x: i, y: 0 }, { x: i + 1, y: 0 }))));
    let t0 = performance.now();
    const r1 = join(objs);
    const t1 = performance.now() - t0;
    t0 = performance.now();
    const r2 = join([one]);
    const t2 = performance.now() - t0;
    const c = 'err' in r2 ? null : curvesOf(r2.objects);
    out(`G8 n=${n}: separate objects ${t1.toFixed(0)} ms (${summary(r1)}); one object ${t2.toFixed(0)} ms (${summary(r2)}; result curves ${c?.length}, length ${c ? f3(len(sample(c[0]!))) : '-'} oracle ${n})`);
  }
});
