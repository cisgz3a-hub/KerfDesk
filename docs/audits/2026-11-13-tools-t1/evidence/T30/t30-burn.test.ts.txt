type V = { x: number; y: number };
const line = (id: string, pts: V[][], extra: Record<string, unknown> = {}): ImportedSvg =>
  ({
    kind: 'imported-svg', id, source: `${id}.dxf`, operationIds: ['L'],
    bounds: { minX: 0, minY: 0, maxX: 100, maxY: 100 }, transform: IDENTITY_TRANSFORM,
    paths: [{ color: '#000000', polylines: pts.map((points) => ({ points, closed: false })) }],
    ...extra,
  }) as ImportedSvg;
const ids = (p: Project) => p.scene.objects.map((o) => o.id).join(',');
/** pierces = lit runs; travel = G0 distance (scene mm). */
function stats(gcode: string, moves: LitMove[]) {
  let pierces = 0;
  let prev: V | null = null;
  for (const m of moves) {
    if (prev === null || Math.hypot(prev.x - m.from.x, prev.y - m.from.y) > 1e-6) pierces += 1;
    prev = m.to;
  }
  let x = 0, y = 0, g = 0, travel = 0;
  for (const raw of gcode.split('\n')) {
    const l = raw.replace(/;.*$/, '').toUpperCase();
    const gm = /G0*([0-3])\b/.exec(l);
    if (gm) g = Number(gm[1]);
    const xm = /X(-?\d*\.?\d+)/.exec(l);
    const ym = /Y(-?\d*\.?\d+)/.exec(l);
    const nx = xm ? Number(xm[1]) : x;
    const ny = ym ? Number(ym[1]) : y;
    if ((xm || ym) && g === 0) travel += Math.hypot(nx - x, ny - y);
    x = nx;
    y = ny;
  }
  return `lit ${f3(totalLen(moves))} pierces ${pierces} travel ${f3(travel)}`;
}
const burnStats = (p: Project) => {
  const b = burn(p);
  return b.ok ? stats(b.gcode, b.moves) : `NOT OK ${b.why}`;
};

it('B1 burn before/after: two lines, an open square, undo/redo/save', () => {
  const L = layer('L', 'line', 1000, 50);
  const A = line('A', [[{ x: 10, y: 10 }, { x: 40, y: 10 }]], { name: 'Alpha' });
  const B = line('B', [[{ x: 70, y: 10 }, { x: 40.04, y: 10 }]]); // reversed, gap 0.04
  const S = line('S', [[{ x: 10, y: 30 }, { x: 30, y: 30 }, { x: 30, y: 50 }, { x: 10, y: 50 }, { x: 10, y: 30.03 }]]);
  const p0 = proj([A, B, S], [L]);
  setProject(p0, ['A', 'B', 'S']);
  out(`B1 menu enabled ${selectionCanJoinPaths(p0, ['A', 'B', 'S'])}`);
  out(`B1 before: ${burnStats(p0)} (oracle lit 30+29.96+79.97=139.930, pierces 3)`);
  const ok = st().joinSelectedPaths(0.05);
  const p1 = st().project;
  out(`B1 applied ${ok}; objects ${ids(p1)}; selected ${st().selectedObjectId},${[...st().additionalSelectedIds]}; undo ${st().undoStack.length}; dirty ${st().dirty}; toasts ${JSON.stringify(toasts())}`);
  out(`B1 after: ${burnStats(p1)} (oracle lit 139.930+0.04+0.03=140.000, pierces 2)`);
  const a1 = p1.scene.objects.find((o) => o.id === 'A') as ImportedSvg;
  out(`B1 A name ${a1.name}; source ${a1.source}; transform identity ${JSON.stringify(a1.transform) === JSON.stringify(IDENTITY_TRANSFORM)}`);
  st().undo();
  out(`B1 undo exact: ${JSON.stringify(st().project) === JSON.stringify(p0)}`);
  st().redo();
  out(`B1 redo exact: ${JSON.stringify(st().project) === JSON.stringify(p1)}`);
  const re = deserializeProject(serializeProject(p1));
  out(`B1 save/reopen: ${re.kind}; burn same ${re.kind === 'ok' ? burnStats((re as { project: Project }).project) === burnStats(p1) : '-'}`);
});

it('B4 variable text in a single-line font', async () => {
  const L = layer('L', 'line', 1234, 37);
  const renderer: VariableTextRenderer = ({ text, content }) =>
    textToPolylines({ content, sizeMm: text.sizeMm, alignment: text.alignment, lineHeight: text.lineHeight, letterSpacing: text.letterSpacing, color: text.color, geometry: 'single-line', fontKey: text.fontKey });
  for (const [template, tokens] of [
    ['SN {{serial:4}}', [{ kind: 'literal', value: 'SN ' }, { kind: 'serial', prefix: '', width: 4 }]],
    ['{{serial:4}}', [{ kind: 'serial', prefix: '', width: 4 }]],
  ] as const) {
    for (const fontKey of ['relief-single-line', 'ems-nixish', 'ems-casual-hand']) {
      const base = {
        kind: 'text', id: 'VT', content: template, variableTemplate: { tokens },
        fontKey, sizeMm: 10, alignment: 'left', lineHeight: 1.2, letterSpacing: 0, color: '#000000',
        bounds: { minX: 0, minY: 0, maxX: 1, maxY: 1 }, transform: { ...IDENTITY_TRANSFORM, x: 50, y: 50 }, paths: [], operationIds: ['L'],
      } as unknown as TextObject;
      let src;
      try {
        src = await renderer({ text: base, content: template, project: proj([], []) } as never);
      } catch (e) {
        out(`B4 ${fontKey}: render failed ${String(e).slice(0, 120)}`);
        continue;
      }
      const text = { ...base, bounds: src.bounds, paths: src.paths } as TextObject;
      const p0 = proj([text], [L], { variables: { ...DEFAULT_PROJECT_VARIABLE_DATA, serialValue: 42 } });
      const snap = async (p: Project, serial: number) => {
        const prepared = await prepareOutputSnapshot({ ...p, variables: { ...p.variables, serialValue: serial } }, { clock: () => new Date(0), renderVariableText: renderer });
        if (!prepared.ok) return { len: NaN, w: NaN };
        const moves = litMoves(emitPreparedGcode(prepared).gcode, p.device);
        const b = bbox(moves);
        return { len: totalLen(moves), w: b.maxX - b.minX };
      };
      const plan = joinOpenVectorPaths([text], [L], 0.05);
      setProject(p0, ['VT']);
      clearToasts();
      const menu = selectionCanJoinPaths(p0, ['VT']);
      const applied = st().joinSelectedPaths(0.05);
      const p1 = st().project;
      const o = p1.scene.objects[0]!;
      const b42 = await snap(p0, 42);
      const b43 = await snap(p0, 43);
      const a42 = await snap(p1, 42);
      const a43 = await snap(p1, 43);
      const value = await renderer({ text: base, content: template.replace('{{serial:4}}', '0042'), project: p0 } as never);
      out(`B4 "${template}" ${fontKey}: menu ${menu}; plan ${plan.kind === 'ok' ? `joins ${plan.value.joins} closures ${plan.value.closures}` : plan.error.message}; applied ${applied}; kind after ${o.kind}; variableTemplate kept ${'variableTemplate' in o}; toasts ${JSON.stringify(toasts())}`);
      out(`   before: serial 42 lit ${f3(b42.len)} w ${f3(b42.w)}; serial 43 lit ${f3(b43.len)}`);
      out(`   after : serial 42 lit ${f3(a42.len)} w ${f3(a42.w)}; serial 43 lit ${f3(a43.len)} (changes with serial: ${Math.abs(a43.len - a42.len) > 1e-6})`);
      out(`   oracle "0042" value width ${f3(value.bounds.maxX - value.bounds.minX)}; template width ${f3(src.bounds.maxX - src.bounds.minX)}`);
    }
  }
});

it('B5 registration jig copies', () => {
  const BOX_A = '379754e4-b1f1-45b1-853d-75a769147a71';
  const BOX_B = '389754e4-b1f1-45b1-853d-75a769147a72';
  const boxes = [BOX_A, BOX_B].map((id, i) => createRegistrationBox({ id, widthMm: 60, heightMm: 60, x: 10 + i * 80, y: 10 }));
  const layers = [{ ...createRegistrationLayer(), output: false }, layer('L', 'line', 2222, 90), layer('E', 'line', 1111, 10)];
  // an "L" drawn as two strokes meeting with a 0.02 gap
  const S = line('SRC', [[{ x: 20, y: 20 }, { x: 20, y: 40 }], [{ x: 20.02, y: 40 }, { x: 40, y: 40 }]]);
  const E = rect('ENG', 6, 6, { x: 30, y: 50 }, { operationIds: ['E'] });
  for (const style of ['legacy id', 'v2 uuid id'] as const) {
    const copyId = style === 'legacy id' ? registrationJigCopyId('SRC', BOX_B) : 'registration-jig-copy-v2:0b0e6f8e-8f1c-4c55-9b33-0a3a8c4a1d01';
    const copy = withRegistrationJigChild(S, { ...S, id: copyId, transform: { ...IDENTITY_TRANSFORM, x: 80 } }, BOX_B, 'copy');
    const engId = style === 'legacy id' ? registrationJigCopyId('ENG', BOX_B) : 'registration-jig-copy-v2:1b0e6f8e-8f1c-4c55-9b33-0a3a8c4a1d02';
    const engC = withRegistrationJigChild(E, { ...E, id: engId, transform: { ...E.transform, x: 110 } } as SceneObject, BOX_B, 'copy');
    const p0 = proj([...boxes, S, E, copy, engC], layers);
    const order = (p: Project) =>
      burn(p).moves.reduce<string[]>((acc, m) => {
        const tag = `F${m.f}@${m.from.x < 85 ? 'A' : 'B'}`;
        if (acc.at(-1) !== tag) acc.push(tag);
        return acc;
      }, []);
    for (const sel of [[copyId], ['SRC', copyId]]) {
      setProject(p0, sel);
      clearToasts();
      const menu = selectionCanJoinPaths(p0, sel);
      st().joinSelectedPaths(0.05);
      const p1 = st().project;
      const c1 = p1.scene.objects.find((o) => o.id === copyId)!;
      out(`B5 ${style} select [${sel.map((s) => s.slice(0, 26))}]: menu ${menu}; copy jig field before ${JSON.stringify(copy.registrationJig?.boxId?.slice(0, 4))} after ${JSON.stringify(c1.registrationJig?.boxId?.slice(0, 4))}; identity after ${registrationJigObjectIdentity(c1) === null ? 'NONE' : 'kept'}; runs ${registrationJigCompilationRuns(p0.scene)?.length ?? 0} -> ${registrationJigCompilationRuns(p1.scene)?.length ?? 0}`);
      out(`   order before ${order(p0).join(' > ')}; after ${order(p1).join(' > ')}; toasts ${JSON.stringify(toasts())}`);
    }
  }
});

it('B7 image mask and text guide references, groups', () => {
  const N = 40;
  const photo = {
    kind: 'raster-image', id: 'photo', source: 'p.png', dataUrl: 'data:image/png;base64,x', pixelWidth: N, pixelHeight: N,
    bounds: { minX: 0, minY: 0, maxX: 40, maxY: 40 }, transform: { ...IDENTITY_TRANSFORM, x: 100, y: 100 }, color: '#808080',
    operationIds: ['img'], dither: 'threshold', linesPerMm: 1, lumaBase64: lumaToBase64(new Uint8Array(N * N).fill(40)),
  } as RasterImage;
  const layers = [{ ...createLayer({ id: 'img', color: '#808080', mode: 'image' }), linesPerMm: 1 } as Layer, layer('L', 'line', 500, 90)];
  const cells = (p: Project) => {
    const g = compileJob(p.scene, p.device).groups.find((c) => c.kind === 'raster' && c.sourceObjectId === 'photo');
    return g?.kind === 'raster' ? g.sValues.filter((s) => s > 0).length : -1;
  };
  // mask: closed-enough open-flagged box (end returns to start), plus a separate open stroke in the mask object
  const M = line('M', [[{ x: 105, y: 105 }, { x: 125, y: 105 }, { x: 125, y: 125 }, { x: 105, y: 125 }, { x: 105, y: 105 }], [{ x: 130, y: 105 }, { x: 130, y: 125 }]]);
  const A = line('A', [[{ x: 130.01, y: 125 }, { x: 130, y: 140 }]]);
  const p0 = proj([A, photo, M], layers);
  setProject(p0, []);
  st().applyImageMask('photo', 'M');
  const pm = st().project;
  out(`B7 masked cells ${cells(pm)} (unmasked ${cells(p0)})`);
  setProject(pm, ['A', 'M']);
  clearToasts();
  st().joinSelectedPaths(0.05);
  const p1 = st().project;
  const ph = p1.scene.objects.find((o) => o.id === 'photo') as RasterImage;
  out(`B7a mask M with an extra stroke joined into earlier A: objects ${ids(p1)}; imageMaskId ${ph.imageMaskId}; cells ${cells(p1)}; toasts ${JSON.stringify(toasts())}`);
  // groups: A in group G1 with X, B in group G2 with Y; joining A+B
  const X = rect('X', 5, 5, { x: 0, y: 50 }, { operationIds: ['L'] });
  const Y = rect('Y', 5, 5, { x: 20, y: 50 }, { operationIds: ['L'] });
  const GA = line('GA', [[{ x: 0, y: 0 }, { x: 10, y: 0 }]]);
  const GB = line('GB', [[{ x: 10, y: 0 }, { x: 20, y: 0 }]]);
  const pg = { ...proj([GA, GB, X, Y], [layer('L', 'line', 500, 90)]) };
  const pg2 = { ...pg, scene: { ...pg.scene, groups: [{ id: 'G1', objectIds: ['GA', 'X'] }, { id: 'G2', objectIds: ['GB', 'Y'] }] } } as Project;
  setProject(pg2, ['GA', 'GB']);
  st().joinSelectedPaths(0.05);
  out(`B7b groups after: ${JSON.stringify(st().project.scene.groups)}; objects ${ids(st().project)}`);
});
