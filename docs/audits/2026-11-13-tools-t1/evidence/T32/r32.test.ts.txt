// ---------- own G-code decoder ----------
type P = { x: number; y: number };
type Seg = { a: P; b: P; s: number; len: number };
function decode(gcode: string, project: Project): Seg[] {
  let x = 0, y = 0, s = 0, motion = 0, on = false, abs = true;
  const segs: Seg[] = [];
  for (const raw of gcode.split(/\r?\n/)) {
    const line = raw.split(';')[0]!.replace(/\([^)]*\)/g, '').trim().toUpperCase();
    if (line === '') continue;
    const words = line.match(/[A-Z][-+]?[0-9]*\.?[0-9]+/g) ?? [];
    let nx: number | undefined, ny: number | undefined;
    for (const w of words) {
      const k = w[0]!, v = Number(w.slice(1));
      if (k === 'G') {
        if (v === 0 || v === 1) motion = v;
        else if (v === 90) abs = true;
        else if (v === 91) abs = false;
      } else if (k === 'M') {
        if (v === 3 || v === 4) on = true;
        else if (v === 5 || v === 2 || v === 30) on = false;
      } else if (k === 'S') s = v;
      else if (k === 'X') nx = v;
      else if (k === 'Y') ny = v;
    }
    if (nx === undefined && ny === undefined) continue;
    const tx = nx === undefined ? x : abs ? nx : x + nx;
    const ty = ny === undefined ? y : abs ? ny : y + ny;
    if (motion === 1 && on && s > 0) {
      const a = toSceneCoords({ x, y }, project.device);
      const b = toSceneCoords({ x: tx, y: ty }, project.device);
      segs.push({ a, b, s, len: Math.hypot(b.x - a.x, b.y - a.y) });
    }
    x = tx; y = ty;
  }
  return segs;
}
function burn(project: Project): Seg[] {
  const prepared = prepareOutput(project);
  if (!prepared.ok) throw new Error('prepare failed ' + JSON.stringify(prepared).slice(0, 300));
  return decode(emitPreparedGcode(prepared).gcode, project);
}
const withOutputOff = (p: Project, opId: string): Project => ({
  ...p, scene: { ...p.scene, layers: p.scene.layers.map((l) => (l.id === opId ? { ...l, output: false } : l)) },
});

// ---------- face grid: 0.1 mm cells over a world rectangle; face(x,y) is the oracle ----------
type Grid = { minX: number; minY: number; nx: number; ny: number; step: number; face: Uint8Array; faceCells: number };
function makeGrid(minX: number, minY: number, maxX: number, maxY: number, step: number, isFace: (x: number, y: number) => boolean): Grid {
  const nx = Math.round((maxX - minX) / step), ny = Math.round((maxY - minY) / step);
  const face = new Uint8Array(nx * ny);
  let faceCells = 0;
  for (let j = 0; j < ny; j++) for (let i = 0; i < nx; i++)
    if (isFace(minX + (i + 0.5) * step, minY + (j + 0.5) * step)) { face[j * nx + i] = 1; faceCells++; }
  return { minX, minY, nx, ny, step, face, faceCells };
}
function measure(segs: Seg[], g: Grid) {
  const hit = new Uint8Array(g.nx * g.ny);
  let onFace = 0, total = 0;
  for (const sg of segs) {
    total += sg.len;
    const n = Math.max(1, Math.ceil(sg.len / 0.01));
    for (let k = 0; k < n; k++) {
      const t = (k + 0.5) / n;
      const x = sg.a.x + (sg.b.x - sg.a.x) * t, y = sg.a.y + (sg.b.y - sg.a.y) * t;
      const i = Math.floor((x - g.minX) / g.step), j = Math.floor((y - g.minY) / g.step);
      if (i < 0 || j < 0 || i >= g.nx || j >= g.ny) continue;
      if (g.face[j * g.nx + i]) { onFace += sg.len / n; hit[j * g.nx + i] = 1; }
    }
  }
  let burnedCells = 0;
  for (let c = 0; c < hit.length; c++) burnedCells += hit[c]!;
  return { onFace, total, burnedCells, pct: (100 * burnedCells) / g.faceCells };
}

function evenOdd(x: number, y: number, rings: P[][]): boolean {
  let inside = false;
  for (const r of rings) for (let i = 0, j = r.length - 1; i < r.length; j = i++) {
    const a = r[i]!, b = r[j]!;
    if ((a.y > y) !== (b.y > y) && x < ((b.x - a.x) * (y - a.y)) / (b.y - a.y) + a.x) inside = !inside;
  }
  return inside;
}
const ringsOf = (paths: ReadonlyArray<{ polylines: ReadonlyArray<{ points: ReadonlyArray<P> }> }>, dx: number, dy: number): P[][] =>
  paths.flatMap((p) => p.polylines.map((pl) => pl.points.map((q) => ({ x: q.x + dx, y: q.y + dy }))));

const op = (id: string, mode: 'fill' | 'image'): Layer => ({ ...createLayer({ id, color: '#000000', name: id, mode }) }) as Layer;
function proj(objects: SceneObject[], layers: Layer[], extra: Partial<Project> = {}): Project {
  const base = createProject();
  return { ...base, optimization: { ...DEFAULT_PROJECT_OPTIMIZATION }, scene: { ...base.scene, objects, layers, groups: [] }, ...extra };
}

/** Real Prepare Stamp path minus the worker: prepareStampSource → prepareStampHeight → stampDraftImage → applyStampDraft. */
async function stamp(project: Project, ids: string[], o: { taper?: number; mirror?: boolean; dpi?: number } = {}) {
  useStore.setState({ project, projectDocumentEpoch: 3, selectedObjectId: ids[0]!, additionalSelectedIds: new Set(ids.slice(1)), undoStack: [], redoStack: [] });
  const source = await prepareStampSource(project, ids, o.dpi ?? 254, new AbortController().signal);
  const draft = prepareStampHeight(source.pixels, { threshold: 127, taperMm: o.taper ?? 0.5, mirror: o.mirror ?? true });
  const { face, luma, ...meta } = draft;
  const enc = { ...meta, dataUrl: 'data:image/png;base64,AA==', lumaBase64: lumaToBase64(luma), sourceDataUrl: '', faceDataUrl: '' };
  const image = stampDraftImage(source, enc, 'STAMP-R');
  useStore.setState((s) => applyStampDraft(s, { project, documentEpoch: 3, ids }, image));
  return { source, draft, face, image, after: useStore.getState().project };
}

// ---------- R1: raster source, asymmetric "F" + bar, 0.1 mm pixels, every default ----------
it('R1 raster source with defaults: face burned by the kept source op', async () => {
  const w = 300, h = 150; // 30 x 15 mm at 0.1 mm
  const luma = new Uint8Array(w * h).fill(255);
  const dark = (x0: number, y0: number, x1: number, y1: number) => { for (let y = y0; y < y1; y++) for (let x = x0; x < x1; x++) luma[y * w + x] = 0; };
  dark(20, 20, 45, 130); dark(45, 20, 110, 45); dark(45, 65, 95, 88); // "F"
  dark(150, 20, 280, 40); dark(240, 40, 260, 130); // "T"-ish, off-centre
  const img = {
    kind: 'raster-image', id: 'SRC', source: 'logo.png', pixelWidth: w, pixelHeight: h,
    bounds: { minX: 0, minY: 0, maxX: 30, maxY: 15 }, transform: { ...IDENTITY_TRANSFORM, x: 40, y: 30 },
    color: '#000000', dither: 'threshold', linesPerMm: 10, dataUrl: 'data:image/png;base64,AA==', lumaBase64: lumaToBase64(luma), operationIds: ['SRCOP'],
  } as unknown as RasterImage;
  const srcOp = { ...op('SRCOP', 'image'), linesPerMm: 10 } as Layer;
  const p0 = proj([img], [srcOp]);
  for (const mirror of [true, false]) {
    const r = await stamp(p0, ['SRC'], { mirror });
    const st = r.after.scene.objects.find((o) => o.id === 'STAMP-R') as RasterImage;
    const newOp = r.after.scene.layers.find((l) => st.operationIds?.includes(l.id))!;
    // oracle: world point → source pixel; face if the (mirrored) source pixel is dark
    const g = makeGrid(40, 30, 70, 45, 0.1, (x, y) => {
      const i = Math.floor((x - 40) / 0.1), j = Math.floor((y - 30) / 0.1);
      const si = mirror ? w - 1 - i : i;
      return luma[j * w + si]! <= 127;
    });
    const both = measure(burn(r.after), g);
    const stampOnly = measure(burn(withOutputOff(r.after, 'SRCOP')), g);
    const srcOnly = measure(burn(p0), g);
    out(`R1 mirror=${mirror}: layers ${r.after.scene.layers.map((l) => `${l.name}[${l.mode},out=${l.output},lpm=${l.linesPerMm},S%=${l.power}]`).join(' + ')}; stamp ${st.pixelWidth}x${st.pixelHeight} px, face cells oracle ${g.faceCells}, tool facePixels ${r.draft.facePixels}`);
    out(`R1 mirror=${mirror}: AS APPLIED  lit ${f3(both.total)} mm, on face ${f3(both.onFace)} mm, face cells burned ${both.burnedCells}/${g.faceCells} = ${f3(both.pct)} %`);
    out(`R1 mirror=${mirror}: SOURCE OFF  lit ${f3(stampOnly.total)} mm, on face ${f3(stampOnly.onFace)} mm, face cells burned ${stampOnly.burnedCells} = ${f3(stampOnly.pct)} %`);
    out(`R1 mirror=${mirror}: SOURCE ONLY lit ${f3(srcOnly.total)} mm, on face ${f3(srcOnly.onFace)} mm (${f3(srcOnly.pct)} % of face cells)`);
    const warn = detectJobIntentWarnings(r.after);
    out(`R1 mirror=${mirror}: Job Review intent warnings ${warn.length}; any naming overlap/copies/stamp/source: ${warn.filter((x) => /overlap|cop(y|ies)|stamp|logo\.png/i.test(x)).map((x) => x.slice(0, 120)).join(' || ') || 'none'}`);
    out(`R1 mirror=${mirror}: new op mode ${newOp.mode} name "${newOp.name}"; stamp dither ${st.dither}`);
  }
});

// ---------- R3: variable text; what does the stamp face spell? ----------
it('R3 variable text "LOT {{serial:3}}" Tinos 8 mm, serial 7', async () => {
  const content = 'LOT {{serial:3}}';
  const parsed = parseVariableTemplateSource(content);
  if (!parsed.ok) throw new Error(parsed.message);
  const R = (c: string) => textToPolylines({ content: c, sizeMm: 8, alignment: 'left', lineHeight: 1.2, letterSpacing: 0, color: '#000000', fontBuffer: TINOS });
  const tpl = await R(content);
  const text = {
    kind: 'text', id: 'VT', content, variableTemplate: parsed.template, fontKey: 'tinos-regular', sizeMm: 8, alignment: 'left', lineHeight: 1.2, letterSpacing: 0, color: '#000000',
    bounds: tpl.bounds, transform: { ...IDENTITY_TRANSFORM, x: 20, y: 20 }, paths: tpl.paths, operationIds: ['VOP'],
  } as unknown as SceneObject;
  const p0 = proj([text], [op('VOP', 'fill')], { variables: { ...DEFAULT_PROJECT_VARIABLE_DATA, serialValue: 7 } });
  const ev = evaluateVariableTemplate(parsed.template, text, p0, { now: new Date(0) });
  const value = ev.ok ? ev.value : `ERR ${ev.message}`;
  const val = await R(value);
  out(`R3 canPrepareStamp ${canPrepareStamp([text])}; evaluated value "${value}"`);
  const r = await stamp(p0, ['VT'], { mirror: false, taper: 0 });
  // stamp face (tool) in world, compared with two independent even-odd renders placed at the text origin
  const b = r.image.bounds, t = r.image.transform;
  const pw = (b.maxX - b.minX) / r.draft.width, ph = (b.maxY - b.minY) / r.draft.height;
  const tplR = ringsOf(tpl.paths as never, 20, 20), valR = ringsOf(val.paths as never, 20, 20);
  let faceN = 0, agreeTpl = 0, agreeVal = 0, tplN = 0, valN = 0, minFx = Infinity, maxFx = -Infinity;
  for (let j = 0; j < r.draft.height; j++) for (let i = 0; i < r.draft.width; i++) {
    const x = b.minX + t.x + (i + 0.5) * pw, y = b.minY + t.y + (j + 0.5) * ph;
    const f = r.face[j * r.draft.width + i] === 1, ft = evenOdd(x, y, tplR), fv = evenOdd(x, y, valR);
    if (f) { faceN++; minFx = Math.min(minFx, x); maxFx = Math.max(maxFx, x); }
    if (ft) tplN++; if (fv) valN++;
    if (f && ft) agreeTpl++; if (f && fv) agreeVal++;
  }
  const iou = (a: number, n1: number, n2: number) => (100 * a) / (n1 + n2 - a);
  out(`R3 stamp face width ${f3(maxFx - minFx + pw)} mm; template glyphs ${f3(tpl.bounds.maxX - tpl.bounds.minX)} mm; value "${value}" ${f3(val.bounds.maxX - val.bounds.minX)} mm`);
  out(`R3 IoU(stamp face, template render) ${f3(iou(agreeTpl, faceN, tplN))} %; IoU(stamp face, value render) ${f3(iou(agreeVal, faceN, valN))} %`);
  const stObj = r.after.scene.objects.find((o) => o.id === 'STAMP-R') as RasterImage;
  out(`R3 stamp carries variableTemplate: ${'variableTemplate' in stObj}; stamp source label "${stObj.source}"`);
  // what the job burns for the text itself (value) vs the stamp
  const renderer: VariableTextRenderer = ({ text: tx, content: c }) => textToPolylines({ content: c, sizeMm: tx.sizeMm, alignment: tx.alignment, lineHeight: tx.lineHeight, letterSpacing: tx.letterSpacing, color: tx.color, fontBuffer: TINOS });
  const snap = await prepareOutputSnapshot(withOutputOff(r.after, stObj.operationIds![0]!), { clock: () => new Date(0), renderVariableText: renderer });
  if (snap.ok) {
    const segs = decode(emitPreparedGcode(snap).gcode, r.after);
    const xs = segs.flatMap((sg) => [sg.a.x, sg.b.x]);
    out(`R3 source text burn (stamp op off) width ${f3(Math.max(...xs) - Math.min(...xs))} mm → it still burns the value`);
  } else out(`R3 snapshot failed ${JSON.stringify(snap).slice(0, 200)}`);
});

// ---------- R4: variable Code 128; canvas shows current serial; stamp uses stored code ----------
it('R4 variable barcode: stamp vs what the canvas shows', async () => {
  const { createBarcodeObject } = await import('../t31-main/src/core/barcode/materialize-barcode');
  const { defaultBarcodeSpec } = await import('../t31-main/src/core/barcode/barcode-spec');
  const { deriveVariableBarcodeDisplay, encodeWaiting } = await import('../t31-main/src/ui/workspace/variable-barcode-display');
  const parsed = parseVariableTemplateSource('{{serial:5}}');
  if (!parsed.ok) throw new Error(parsed.message);
  const spec = { ...defaultBarcodeSpec('code128'), data: '{{serial:5}}', showText: false, variableTemplate: parsed.template };
  const cap = async () => ({ polylines: [], bounds: { minX: 0, minY: 0, maxX: 0, maxY: 0 } }) as never;
  const made = await createBarcodeObject({ id: 'BC', color: '#000000', spec: spec as never, value: '00100', renderCaption: cap, transform: { ...IDENTITY_TRANSFORM, x: 15, y: 15 } });
  if (!made.ok) throw new Error(made.message);
  const bc = { ...made.object, operationIds: ['BOP'] } as SceneObject;
  const p0 = proj([bc], [op('BOP', 'fill')], { variables: { ...DEFAULT_PROJECT_VARIABLE_DATA, serialValue: 117 } });
  const renderer: VariableTextRenderer = ({ text: tx, content: c }) => textToPolylines({ content: c, sizeMm: tx.sizeMm, alignment: tx.alignment, lineHeight: tx.lineHeight, letterSpacing: tx.letterSpacing, color: tx.color, fontBuffer: ROBOTO });
  const d0 = deriveVariableBarcodeDisplay(p0);
  await encodeWaiting(d0.waiting, p0, renderer);
  const shown = deriveVariableBarcodeDisplay(p0).project.scene.objects[0] as SceneObject & { paths: never; transform: typeof IDENTITY_TRANSFORM };
  const stored = bc as SceneObject & { paths: never };
  out(`R4 canPrepareStamp ${canPrepareStamp([bc])}; canvas object differs from stored: ${JSON.stringify(shown.paths) !== JSON.stringify(stored.paths)}`);
  const r = await stamp(p0, ['BC'], { mirror: false, taper: 0 });
  const b = r.image.bounds, t = r.image.transform;
  const pw = (b.maxX - b.minX) / r.draft.width, ph = (b.maxY - b.minY) / r.draft.height;
  const storedR = ringsOf(stored.paths as never, 15, 15), shownR = ringsOf(shown.paths as never, shown.transform.x, shown.transform.y);
  let n = 0, dStored = 0, dShown = 0;
  for (let j = 0; j < r.draft.height; j++) for (let i = 0; i < r.draft.width; i++) {
    const x = b.minX + t.x + (i + 0.5) * pw, y = b.minY + t.y + (j + 0.5) * ph;
    const f = r.face[j * r.draft.width + i] === 1;
    n++;
    if (f !== evenOdd(x, y, storedR)) dStored++;
    if (f !== evenOdd(x, y, shownR)) dShown++;
  }
  out(`R4 stamp face vs independent even-odd of STORED code (00100): ${dStored}/${n} px differ; vs code the CANVAS shows (00117): ${dShown}/${n} px differ`);
});
