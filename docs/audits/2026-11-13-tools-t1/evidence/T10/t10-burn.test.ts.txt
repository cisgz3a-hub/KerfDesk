const LINE_SPEED = 1234;
const f3 = (n: number) => n.toFixed(3);
const st = () => useStore.getState();

function lineLayer(extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id: 'L', color: '#ff0000', name: 'L', mode: 'line' }), speed: LINE_SPEED, power: 30, ...extra };
}

function svgObj(id: string, paths: ColoredPath[], transform: Transform, extra: Partial<ImportedSvg> = {}): ImportedSvg {
  const all = paths.flatMap((p) => p.polylines.flatMap((pl) => pl.points));
  return {
    kind: 'imported-svg',
    id,
    source: `${id}.svg`,
    bounds: {
      minX: Math.min(...all.map((p) => p.x)),
      minY: Math.min(...all.map((p) => p.y)),
      maxX: Math.max(...all.map((p) => p.x)),
      maxY: Math.max(...all.map((p) => p.y)),
    },
    transform,
    operationIds: ['L'],
    paths,
    ...extra,
  };
}

function load(objects: SceneObject[], selected: string[], layer: Layer = lineLayer()): void {
  const base = createProject();
  const project: Project = {
    ...base,
    optimization: { ...DEFAULT_PROJECT_OPTIMIZATION },
    scene: { ...base.scene, objects, layers: [layer], groups: [] },
  };
  useStore.setState({
    project,
    selectedObjectId: selected[0] ?? null,
    additionalSelectedIds: new Set(selected.slice(1)),
    undoStack: [],
    redoStack: [],
    dirty: false,
  });
}

function burn(project: Project) {
  const prepared = prepareOutput(project);
  const emitted = prepared.ok ? emitPreparedGcode(prepared) : { gcode: '' };
  const moves = litMoves(emitted.gcode, project.device).filter((m) => m.f === LINE_SPEED);
  return { gcode: emitted.gcode, moves, ok: prepared.ok };
}

let lastToast = '';
function apply(tol: number, mode: 'bridge' | 'move-ends') {
  const p0 = st().project;
  const u0 = st().undoStack.length;
  useToastStore.setState({ toasts: [] });
  st().closeSelectedPathsWithTolerance(tol, mode);
  const toasts = useToastStore.getState().toasts;
  lastToast = toasts.length > 0 ? toasts.map((t) => t.message).join(' | ') : '(no toast)';
  return { changed: st().project !== p0, undoAdded: st().undoStack.length - u0 };
}

// ---------- independent sampling / deviation ----------
function densify(pts: ReadonlyArray<Vec>, step = 0.05, closed = false): Vec[] {
  const res: Vec[] = [];
  const list = closed && pts.length > 0 ? [...pts, pts[0]!] : pts;
  for (let i = 0; i + 1 < list.length; i++) {
    const a = list[i]!;
    const b = list[i + 1]!;
    const n = Math.max(1, Math.ceil(Math.hypot(b.x - a.x, b.y - a.y) / step));
    for (let k = 0; k < n; k++) res.push({ x: a.x + ((b.x - a.x) * k) / n, y: a.y + ((b.y - a.y) * k) / n });
  }
  if (list.length > 0) res.push(list.at(-1)!);
  return res;
}
function segDist(p: Vec, a: Vec, b: Vec): number {
  const dx = b.x - a.x, dy = b.y - a.y;
  const l2 = dx * dx + dy * dy;
  const t = l2 === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / l2));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}
function distToPolyline(p: Vec, poly: ReadonlyArray<Vec>): number {
  let d = Infinity;
  for (let i = 0; i + 1 < poly.length; i++) d = Math.min(d, segDist(p, poly[i]!, poly[i + 1]!));
  if (poly.length === 1) d = Math.hypot(p.x - poly[0]!.x, p.y - poly[0]!.y);
  return d;
}
/** max distance from any point of `a` (densified) to polyline `b`. */
function oneSided(a: ReadonlyArray<Vec>, b: ReadonlyArray<Vec>): number {
  let m = 0;
  for (const p of densify(a, 0.1)) m = Math.max(m, distToPolyline(p, b));
  return m;
}
function len(pts: ReadonlyArray<Vec>): number {
  let s = 0;
  for (let i = 0; i + 1 < pts.length; i++) s += Math.hypot(pts[i + 1]!.x - pts[i]!.x, pts[i + 1]!.y - pts[i]!.y);
  return s;
}
const deg = (d: number) => (d * Math.PI) / 180;
function circ(cx: number, cy: number, r: number, a0: number, a1: number, n = 720): Vec[] {
  const pts: Vec[] = [];
  for (let i = 0; i <= n; i++) {
    const a = deg(a0 + ((a1 - a0) * i) / n);
    pts.push({ x: cx + r * Math.cos(a), y: cy + r * Math.sin(a) });
  }
  return pts;
}
function cubicPts(p0: Vec, c1: Vec, c2: Vec, p3: Vec, n = 400): Vec[] {
  const pts: Vec[] = [];
  for (let i = 0; i <= n; i++) {
    const t = i / n, u = 1 - t;
    pts.push({
      x: u * u * u * p0.x + 3 * u * u * t * c1.x + 3 * u * t * t * c2.x + t * t * t * p3.x,
      y: u * u * u * p0.y + 3 * u * u * t * c1.y + 3 * u * t * t * c2.y + t * t * t * p3.y,
    });
  }
  return pts;
}
const world = (pts: ReadonlyArray<Vec>, t: Transform) => pts.map((p) => applyTransform(p, t));

function compiledWorld(path: ColoredPath, t: Transform): Vec[][] {
  return compilationPolylines(path, t, 0.01).map((pl) => world(pl.closed ? [...pl.points, pl.points[0]!] : pl.points, t));
}

// ---------- F: real imports (DXF ARC, SVG A) through the store action ----------
import { parseDxf } from '../t06-main/src/io/dxf/parse-dxf';
import { parseSvg } from '../t06-main/src/io/svg/parse-svg';
it('F real imports', () => {
  out('== F real imports, move-ends');
  const dxf = [[0, 'SECTION'], [2, 'ENTITIES'], [0, 'ARC'], [8, '0'], [10, '0'], [20, '0'], [40, '20'], [50, '5'], [51, '355'], [0, 'ENDSEC'], [0, 'EOF']]
    .map(([c, v]) => `${c}\n${v}`).join('\n') + '\n';
  const d = parseDxf({ dxfText: dxf, id: 'dxf', source: 'arc.dxf' });
  const svgText = '<svg xmlns="http://www.w3.org/2000/svg" width="100mm" height="100mm" viewBox="0 0 100 100"><path d="M 69.92 51.74 A 20 20 0 1 1 69.92 48.26" fill="none" stroke="#ff0000"/></svg>';
  const s = parseSvg({ svgText, id: 'svg', source: 'arc.svg' });
  for (const [name, object] of [['DXF ARC 5..355 r20', d.kind === 'ok' ? d.object : null], ['SVG path A r20 gap 3.48', (s as { object?: unknown }).object ?? null]] as const) {
    if (object === null || object === undefined) { out(`${name}: import failed ${JSON.stringify(d.kind)}`); continue; }
    const o = { ...(object as ImportedSvg), operationIds: ['L'] } as ImportedSvg;
    const segs = o.paths.map((p) => p.curves?.map((c) => c.segments.map((g) => g.kind).join('+')).join(';')).join(' / ');
    load([o], [o.id]);
    const b0 = burn(st().project);
    const r = apply(5, 'move-ends');
    const b1 = burn(st().project);
    out(`${name}: segments [${segs}] move-ends tol 5: changed=${r.changed} toast="${lastToast}" lit length ${f3(totalLen(b0.moves))} -> ${f3(totalLen(b1.moves))}`);
    load([o], [o.id]);
    apply(5, 'bridge');
    const b2 = burn(st().project);
    out(`   bridge tol 5: lit length ${f3(totalLen(b0.moves))} -> ${f3(totalLen(b2.moves))}`);
  }
});

// ---------- G: Optimize Shapes output (arc fitting) then move-ends ----------
import { optimizeObjectPaths } from '../t06-main/src/core/geometry/shape-optimize/optimize-object';
import { DEFAULT_SHAPE_OPTIMIZE_OPTIONS } from '../t06-main/src/core/geometry/shape-optimize/shape-optimize-options';
it('G optimize then close', () => {
  out('== G Optimize Shapes then Close Paths With Tolerance (move-ends)');
  for (const [a0, a1, r] of [[5, 355, 20], [10, 350, 20], [5, 355, 5], [20, 340, 10], [5, 175, 20]] as const) {
    for (const fitWith of ['lines-arcs-curves', 'lines-arcs'] as const) {
      const pts = circ(0, 0, r, a0, a1, 360);
      const base = svgObj('a', [{ color: '#ff0000', polylines: [{ closed: false, points: pts }] }], IDENTITY_TRANSFORM);
      const opt = optimizeObjectPaths(base as never, { ...DEFAULT_SHAPE_OPTIMIZE_OPTIONS, fitWith });
      const o = { ...base, paths: opt.paths } as ImportedSvg;
      const segs = o.paths.map((p) => p.curves?.map((c) => c.segments.map((g) => g.kind[0]).join('')).join(';') ?? 'polyline').join(' / ');
      load([o], ['a']);
      const b0 = burn(st().project);
      const gap = Math.hypot(pts[0]!.x - pts.at(-1)!.x, pts[0]!.y - pts.at(-1)!.y);
      const r1 = apply(Math.max(gap + 0.01, 0.5), 'move-ends');
      const b1 = burn(st().project);
      out(`arc ${a0}..${a1} r${r} (gap ${f3(gap)}) optimized ${fitWith} -> segments [${segs}]; move-ends: changed=${r1.changed} toast="${lastToast}" lit ${f3(totalLen(b0.moves))} -> ${f3(totalLen(b1.moves))}`);
    }
  }
});
