type Shape = { rings: Vec[][]; white: boolean; label: string };

function fillShapes(project: Project): Shape[] {
  const shapes: Shape[] = [];
  for (const o of project.scene.objects) {
    if (o.kind !== 'imported-svg') continue;
    o.paths.forEach((p, i) => {
      const fillOp = project.scene.layers.some(
        (l) => l.mode === 'fill' && l.output && pathUsesOperation(o, p, l),
      );
      if (!fillOp) return;
      shapes.push({
        rings: p.polylines.map((pl) => pl.points.map((pt) => applyTransform(pt, o.transform))),
        white: p.color.toLowerCase() === '#ffffff' && p.whiteInk === undefined,
        label: `${o.id}#${i}${p.color}`,
      });
    });
  }
  return shapes;
}

function insideEO(pt: Vec, rings: Vec[][]): boolean {
  let hit = false;
  for (const poly of rings)
    for (let i = 0, j = poly.length - 1; i < poly.length; j = i, i += 1) {
      const a = poly[i]!;
      const b = poly[j]!;
      if (a.y > pt.y !== b.y > pt.y) {
        const x = ((b.x - a.x) * (pt.y - a.y)) / (b.y - a.y) + a.x;
        if (pt.x < x) hit = !hit;
      }
    }
  return hit;
}

function shouldBurn(pt: Vec, shapes: Shape[]): boolean {
  for (let k = shapes.length - 1; k >= 0; k -= 1) {
    if (insideEO(pt, shapes[k]!.rings)) return !shapes[k]!.white;
  }
  return false;
}

function boxOf(shapes: Shape[]) {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  for (const s of shapes) for (const r of s.rings) for (const p of r) {
    minX = Math.min(minX, p.x); maxX = Math.max(maxX, p.x);
    minY = Math.min(minY, p.y); maxY = Math.max(maxY, p.y);
  }
  return { minX, minY, maxX, maxY };
}

function expectedArea(shapes: Shape[], step = 0.1): number {
  const b = boxOf(shapes);
  let n = 0;
  for (let y = b.minY + step / 2; y < b.maxY; y += step)
    for (let x = b.minX + step / 2; x < b.maxX; x += step) if (shouldBurn({ x, y }, shapes)) n += 1;
  return n * step * step;
}

function burn(project: Project) {
  const prepared = prepareOutput(project);
  const emitted = emitPreparedGcode(prepared);
  return { moves: litMoves(emitted.gcode, project.device), ok: prepared.ok, gcode: emitted.gcode };
}

/** Lit length (fill rows only = horizontal moves) split by oracle. */
function classify(moves: LitMove[], shapes: Shape[]) {
  let good = 0, bad = 0, nonRow = 0;
  for (const m of moves) {
    if (Math.abs(m.to.y - m.from.y) > 1e-6) { nonRow += m.len; continue; }
    const n = Math.max(1, Math.ceil(m.len / 0.02));
    for (let i = 0; i < n; i += 1) {
      const t = (i + 0.5) / n;
      const p = { x: m.from.x + t * (m.to.x - m.from.x), y: m.from.y + t * (m.to.y - m.from.y) };
      if (shouldBurn(p, shapes)) good += m.len / n; else bad += m.len / n;
    }
  }
  return { good, bad, nonRow };
}

function spacing(project: Project): number {
  return project.scene.layers.find((l) => l.mode === 'fill')?.hatchSpacingMm ?? NaN;
}

function whitePathCount(project: Project): number {
  return project.scene.objects.reduce(
    (n, o) => n + ('paths' in o ? (o.paths as ColoredPath[]).filter((p) => p.color.toLowerCase() === '#ffffff' && p.whiteInk !== 'kept').length : 0),
    0,
  );
}

function describeScene(project: Project): string {
  return project.scene.objects
    .map((o) =>
      'paths' in o
        ? `${o.id}[imp=${(o as { svgImport?: { id: string; mode?: string } }).svgImport?.id ?? '-'}:${(o as { svgImport?: { mode?: string } }).svgImport?.mode ?? '-'}] ` +
          (o.paths as ColoredPath[]).map((p) => `${p.color}${p.whiteInk ? '/' + p.whiteInk : ''}{${(p.operationIds ?? o.operationIds ?? ['by-color']).join(',')}}`).join(' ')
        : o.id,
    )
    .join(' | ') + ' || layers ' + project.scene.layers.map((l) => `${l.id}:${l.mode}:${l.color}:out=${l.output}`).join(', ');
}

function lastToast(): string {
  const t = useToastStore.getState() as unknown as { toasts?: { message: string }[] };
  return t.toasts?.at(-1)?.message ?? '(none)';
}

/** Run the tool on the current store project; report oracle vs burn. */
function runTool(name: string, oracleFrom?: Project) {
  const before = st().project;
  const shapes = fillShapes(oracleFrom ?? before);
  const count = countBurningImportedWhites(before.scene);
  const bBefore = burn(before);
  const cBefore = classify(bBefore.moves, shapes);
  const whitesBefore = whitePathCount(before);
  const returned = st().leaveSavedWhiteUnburned();
  const after = st().project;
  const bAfter = burn(after);
  const cAfter = classify(bAfter.moves, shapes);
  const sp = spacing(after);
  const area = expectedArea(shapes);
  out(`== ${name}`);
  out(`  scene before: ${describeScene(before)}`);
  out(`  scene after : ${describeScene(after)}`);
  out(`  count(before)=${count} returned=${returned} whitePaths ${whitesBefore}->${whitePathCount(after)} toast="${lastToast()}" countAfter=${countBurningImportedWhites(after.scene)}`);
  out(`  oracle area ${area.toFixed(1)} mm2 -> expected row length ${(area / sp).toFixed(1)} (spacing ${sp})`);
  out(`  BEFORE rows good ${cBefore.good.toFixed(1)} bad(lit where oracle says paper) ${cBefore.bad.toFixed(1)} nonRow ${cBefore.nonRow.toFixed(1)}`);
  out(`  AFTER  rows good ${cAfter.good.toFixed(1)} bad ${cAfter.bad.toFixed(1)} nonRow ${cAfter.nonRow.toFixed(1)} total ${totalLen(bAfter.moves).toFixed(1)} ok=${bAfter.ok}`);
  return { before, after, bBefore, bAfter, cBefore, cAfter, shapes, count, returned };
}

    project: { ...s.project, scene: { ...s.project.scene, objects: s.project.scene.objects.map(fn) } },
  }));
}

function stripKept(p: Project) {
  return p.scene.objects.map((o) => ('paths' in o ? (o.paths as ColoredPath[]).map(({ whiteInk: _w, ...rest }) => rest) : o));
}

  return added.id;
}
