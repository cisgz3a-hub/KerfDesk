/** 1.0.29's saved import: the parse as is, white on its own operation, no marks. */
function olderBuild(text: string): void {
  importSvg(text);
  st().undo();
  const project = st().project;
  const objects = project.scene.objects.map((object) =>
    'paths' in object
      ? { ...object, paths: object.paths.map(({ whiteInk: _w, ...path }) => path) }
      : object,
  ) as SceneObject[];
  useStore.setState({
    project: { ...project, scene: { ...project.scene, objects } },
    undoStack: [],
    redoStack: [],
  });
}

function segLen(poly: readonly Vec2[]): number {
  let n = 0;
  for (let i = 1; i < poly.length; i += 1)
    n += Math.hypot(poly[i]!.x - poly[i - 1]!.x, poly[i]!.y - poly[i - 1]!.y);
  return n;
}

/** Lit length per operation id, from compileJob. */
function perOperation(project: Project): Record<string, number> {
  const result: Record<string, number> = {};
  for (const group of compileJob(project.scene, project.device).groups) {
    if (!('segments' in group)) continue;
    const id = (group as { layerId: string }).layerId;
    const len = group.segments.reduce((s, seg) => s + segLen(seg.polyline), 0);
    result[id] = (result[id] ?? 0) + len;
  }
  for (const k of Object.keys(result)) result[k] = Math.round(result[k]! * 100) / 100;
  return result;
}

/** White outlines (paper white, Fill-bound) per object, machine coordinates. */
function whiteOutlinesByObject(project: Project): Map<string, Vec2[][]> {
  const map = new Map<string, Vec2[][]>();
  for (const object of project.scene.objects) {
    if (!('paths' in object)) continue;
    const rings = object.paths
      .filter((p) => p.color.toLowerCase() === '#ffffff')
      .flatMap((p) =>
        p.polylines.map((l) =>
          l.points.map((pt) => toMachineCoords(applyTransform(pt, object.transform), project.device)),
        ),
      );
    if (rings.length > 0) map.set(object.id, rings);
  }
  return map;
}

function describe(project: Project): string {
  const { scene } = project;
  return scene.objects
    .map((o) => {
      if (!('paths' in o)) return `${o.id}(${o.kind})`;
      const imp = o.svgImport ? `imp=${o.svgImport.id.slice(0, 6)}:${o.svgImport.mode}` : 'imp=-';
      const ov = o.operationOverride?.mode ?? '-';
      const paths = o.paths
        .map((p) => {
          const ops = scene.layers.filter((l) => pathUsesOperation(o, p, l)).map((l) => `${l.id.slice(-6)}:${l.mode}${l.output ? '' : ':off'}`);
          return `${p.color}${p.whiteInk ? '/' + p.whiteInk : ''}[${ops.join(',')}]`;
        })
        .join(' ');
      return `${o.id.slice(-10)} ${imp} ov=${ov} T=(${o.transform.x.toFixed(1)},${o.transform.y.toFixed(1)}) fillPart=${isSavedFillPart(scene, o as never)} ${paths}`;
    })
    .join('\n    ');
}

function lastToast(): string {
  const t = useToastStore.getState() as unknown as { toasts?: { message: string }[] };
  return t.toasts?.at(-1)?.message ?? '(none)';
}

function surfaces(project: Project): string {
  const notice = burningWhiteOpenNotice(project);
  const review = burningWhiteReviewWarnings(project);
  return `count=${countBurningImportedWhites(project.scene)} canLeave=${countBurningImportedWhites(project.scene) > 0} notice=${notice === null ? 'none' : JSON.stringify(notice.slice(0, 60))} reviewRows=${review.length}`;
}

/** Burned length inside each white outline (taken from `ref`) for `project`. */
function insidePerWhite(project: Project, ref: Project): string {
  const parts: string[] = [];
  for (const [id, rings] of whiteOutlinesByObject(ref)) {
    parts.push(`${id.slice(-10)}:${burnInside(project, rings).toFixed(0)}`);
  }
  return parts.join(' ');
}

function selectAll(): void {
  const ids = st().project.scene.objects.map((o) => o.id);
  useStore.setState({ selectedObjectId: ids[0]!, additionalSelectedIds: new Set(ids.slice(1)) });
}

function runTool(name: string): { before: Project; after: Project } {
  const before = st().project;
  out(`== ${name}`);
  out(`  before: ${surfaces(before)}`);
  out(`    ${describe(before)}`);
  out(`    groups: ${JSON.stringify((before.scene.groups ?? []).map((g) => g.objectIds.length))}`);
  out(`    per-op lit: ${JSON.stringify(perOperation(before))}`);
  out(`    burned inside each white (mm): ${insidePerWhite(before, before)}`);
  const returned = st().leaveSavedWhiteUnburned();
  const after = st().project;
  out(`  tool returned ${returned}; toast "${lastToast()}"`);
  out(`  after: ${surfaces(after)}`);
  out(`    ${describe(after)}`);
  out(`    per-op lit: ${JSON.stringify(perOperation(after))}`);
  out(`    burned inside each ORIGINAL white outline (mm): ${insidePerWhite(after, before)}`);
  return { before, after };
}

it('R: repo SVG corpus - how often a non-fill entry splits fills with a later white over an earlier dark', () => {
  const root = 'C:/Users/stolk/kd-t1/t06-main';
  const files: string[] = [];
  const walk = (dir: string) => {
    for (const name of readdirSync(dir)) {
      if (name === 'node_modules' || name === '.git' || name === 'dist') continue;
      const p = join(dir, name);
      const s = statSync(p);
      if (s.isDirectory()) walk(p);
      else if (name.toLowerCase().endsWith('.svg')) files.push(p);
    }
  };
  walk(join(root, 'src'));
  walk(join(root, 'docs'));
  walk(join(root, 'public'));
  out(`== R corpus: ${files.length} svg files`);
  for (const file of files) {
    let parsed;
    try {
      parsed = parseSvg({ svgText: readFileSync(file, 'utf8'), id: 'r', source: 'r.svg' });
    } catch (e) {
      out(`  ${file.slice(root.length)}: parse error ${(e as Error).message.slice(0, 60)}`);
      continue;
    }
    const entries = parsed.fragment?.entries ?? [];
    const seq = entries.map((e) =>
      e.kind === 'imported-svg'
        ? `${e.operationOverride?.mode === 'fill' ? 'F' : 'L'}${e.paths.some((p) => p.color.toLowerCase() === '#ffffff' && p.whiteInk === undefined) ? 'w' : ''}`
        : 'I',
    );
    const fillEntries = entries.filter((e) => e.kind === 'imported-svg' && e.operationOverride?.mode === 'fill');
    // split risk: a paper white in fill entry j whose box overlaps a dark fill in an earlier fill entry i<j
    let risky = 0;
    fillEntries.forEach((ej, j) => {
      if (ej.kind !== 'imported-svg') return;
      for (const w of ej.paths.filter((p) => p.color.toLowerCase() === '#ffffff' && p.whiteInk === undefined)) {
        const wb = bbox(w);
        for (let i = 0; i < j; i += 1) {
          const ei = fillEntries[i]!;
          if (ei.kind !== 'imported-svg') continue;
          if (ei.paths.some((p) => p.color.toLowerCase() !== '#ffffff' && overlap(bbox(p), wb))) risky += 1;
        }
      }
    });
    const whites = entries.flatMap((e) => (e.kind === 'imported-svg' ? e.paths : [])).filter((p) => p.color.toLowerCase() === '#ffffff').length;
    if (entries.length > 1 || whites > 0)
      out(`  ${file.slice(root.length)}: entries ${seq.join(' ')} whites=${whites} crossEntryWhiteOverDark=${risky}`);
  }
});

function bbox(p: ColoredPath) {
  let a = Infinity, b = Infinity, c = -Infinity, d = -Infinity;
  for (const l of p.polylines) for (const q of l.points) { a = Math.min(a, q.x); b = Math.min(b, q.y); c = Math.max(c, q.x); d = Math.max(d, q.y); }
  return { a, b, c, d };
}
function overlap(x: ReturnType<typeof bbox>, y: ReturnType<typeof bbox>) {
  return x.a < y.c && y.a < x.c && x.b < y.d && y.b < x.d;
}

function addLineOpFor(ids: string[], power: number): string {
  useStore.setState({ selectedObjectId: ids[0]!, additionalSelectedIds: new Set(ids.slice(1)) });
  const before = new Set(st().project.scene.layers.map((l) => l.id));
  st().addOperationForSelection();
  const added = st().project.scene.layers.find((l) => !before.has(l.id))!;
  st().setLayerParam(added.id, { mode: 'line', power });
  return added.id;
}

const SQ = svg(`<rect x="5" y="5" width="30" height="30" fill="#000"/><path d="M20 10 L30 20 L20 30 L10 20 Z" fill="#fff"/>`);
function opOf(color: string): string {
  const p = st().project;
  for (const o of p.scene.objects) {
    if (!('paths' in o)) continue;
    for (const path of o.paths)
      if (path.color.toLowerCase() === color)
        return p.scene.layers.find((l) => pathUsesOperation(o, path, l))!.id;
  }
  throw new Error('no op for ' + color);
}
