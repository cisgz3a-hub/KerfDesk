it('D3 auto tones clip promise on the shipping tone curve', () => {
  let s = 12345 >>> 0; const rnd = () => { s = (s * 1664525 + 1013904223) >>> 0; return s / 4294967296; };
  let n = 0, mismatchOracle = 0, loNotSolid = 0, hiNot255 = 0, hiBelow254 = 0, worstHi = 255, worstCase = '';
  const hiHist = new Map<number, number>();
  for (let t = 0; t < 20000; t++) {
    // random portrait-like histogram: 1-3 gaussian lumps, optional white background
    const N = 2000; const luma = new Uint8Array(N);
    const lumps = 1 + Math.floor(rnd() * 3); const centres = Array.from({ length: lumps }, () => 20 + rnd() * 215); const widths = centres.map(() => 3 + rnd() * 50);
    for (let i = 0; i < N; i++) { const k = Math.floor(rnd() * lumps); const u = rnd() + rnd() + rnd() - 1.5; luma[i] = Math.max(0, Math.min(255, Math.round(centres[k]! + u * widths[k]!))); }
    if (rnd() < 0.3) for (let i = 0; i < N * 0.3; i++) luma[i] = 255;
    const hist = new Uint32Array(256); for (const v of luma) hist[v]++;
    const real = portraitAutoTones(hist); const mine = oracleAutoTones(luma);
    if (real === null || mine === null) continue;
    n++;
    if (real.brightness !== mine.b || real.contrast !== mine.c || real.gamma !== mine.g) mismatchOracle++;
    const lut = lumaAdjustmentLut(real) ?? Uint8Array.from({ length: 256 }, (_, i) => i);
    const capped = 255 / (mine.hi - mine.lo) > 2;
    if (lut[mine.lo]! !== 0) loNotSolid++;
    if (!capped) {
      const v = lut[mine.hi]!; hiHist.set(v, (hiHist.get(v) ?? 0) + 1);
      if (v !== 255) hiNot255++;
      if (v < 254) hiBelow254++;
      if (v < worstHi) { worstHi = v; worstCase = `lo ${mine.lo} hi ${mine.hi} median ${mine.median} -> b${real.brightness} c${real.contrast} g${real.gamma}; lut[hi]=${v}, lut[hi-1]=${lut[mine.hi - 1]}`; }
    }
  }
  out(`D3 ${n} histograms: oracle (ADR text) vs portraitAutoTones mismatches ${mismatchOracle}; darkest-0.5% level not solid (lut!=0) ${loNotSolid}; uncapped cases: lightest-0.5% level not 255 ${hiNot255}, below 254 ${hiBelow254}; lut[hi] distribution ${JSON.stringify([...hiHist.entries()].sort((a, b) => a[0] - b[0]))}; worst ${worstCase}`);
});

it('D4 an image sub-pass after Light marks', async () => {
  const W = 48, H = 32, box = { x: 20, y: 30, w: 4.8, h: 3.2 };
  load([mkImage('A', photoLuma(W, H), W, H, box)], [mkLayer({ ditherAlgorithm: 'floyd-steinberg' })]);
  await act(async () => { st().addLayerSubLayer('img'); });
  const sub = st().project.scene.layers[0]!.subLayers[0]!;
  out(`D4 sub-pass added: ${JSON.stringify({ id: sub.id, mode: sub.settings.mode, dither: sub.settings.ditherAlgorithm, neg: sub.settings.negativeImage, enabled: sub.enabled }).slice(0, 300)}`);
  const g0 = gcodeOf(st().project);
  const d0 = decode(g0, box, W, H, 0.1);
  let both0 = 0; d0.passes.forEach((p) => { if (p >= 2) both0++; });
  const d = await mountHost(); await open();
  await marking(true); await wait(400);
  const prev = workerLog[workerLog.length - 1]?.result;
  out(`D4 dialog op hint "${document.querySelector('.lf-photo-engraving__operation')?.textContent}"; preview scanPasses ${prev?.kind === 'ok' ? prev.scanPasses : '-'} scanCount ${prev?.kind === 'ok' ? prev.scanCount : '-'}; dialog text mentions sub/second pass: ${/sub-|second pass|other pass/i.test(document.body.textContent ?? '')}`);
  await applyBtn(); await wait(50);
  await unmount(d);
  const a = img();
  out(`D4 override ${JSON.stringify(a.operationOverride)}`);
  const g1 = gcodeOf(st().project);
  const d1 = decode(g1, box, W, H, 0.1);
  let lit = 0, twice = 0; d1.passes.forEach((p, i) => { if (d1.s[i]! > 0) lit++; if (p >= 2) twice++; });
  // light marks alone would burn the inverse; count cells burned by at least one pass
  out(`D4 before Apply: cells burned in both passes ${both0}; after Light marks Apply: cells burned by some pass ${lit}/${W * H}, burned twice ${twice}`);
  void useStore;
});
