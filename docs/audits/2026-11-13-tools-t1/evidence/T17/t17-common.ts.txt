export function logger(path: string): (s: string) => void {
  const lines: string[] = [];
  return (s: string) => { lines.push(s); writeFileSync(path, lines.join('\n')); };
}
export const b64 = (u: Uint8Array) => Buffer.from(u).toString('base64');
export const T = (t: Partial<Transform>): Transform => ({ ...IDENTITY_TRANSFORM, ...t });

/** Distinct-ish pattern: every source pixel has its own value (no flat areas). */
export function patternLuma(W: number, H: number, kind: 'pattern' | 'flat' = 'pattern', flat = 200): Uint8Array {
  const l = new Uint8Array(W * H);
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) l[y * W + x] = kind === 'flat' ? flat : (x * 37 + y * 101 + ((x * y) % 7) * 13) % 256;
  return l;
}
export function mkImage(id: string, W: number, H: number, box: { x: number; y: number; w: number; h: number }, transform: Transform = IDENTITY_TRANSFORM, extra: Partial<RasterImage> = {}, luma?: Uint8Array): RasterImage {
  return {
    kind: 'raster-image', id, source: `${id}.png`, dataUrl: 'data:image/png;base64,iVBORw0KGgo=', pixelWidth: W, pixelHeight: H,
    bounds: { minX: box.x, minY: box.y, maxX: box.x + box.w, maxY: box.y + box.h }, transform, color: '#808080',
    dither: 'grayscale', linesPerMm: 5, lumaBase64: b64(luma ?? patternLuma(W, H)), ...extra,
  } as RasterImage;
}
export function imageLayer(extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id: 'img', color: '#808080', name: 'Photo', mode: 'image' }), power: 80, minPower: 0, speed: 3000, linesPerMm: 5,
    ditherAlgorithm: 'grayscale', dotWidthCorrectionMm: 0, imageBidirectional: false, ...extra } as Layer;
}
export function maskLayer(): Layer {
  return { ...createLayer({ id: 'mask', color: '#000000', name: 'Mask', mode: 'line' }), output: false, power: 50, speed: 1000 } as Layer;
}
export function mkMask(id: string, contours: P[][], transform: Transform = IDENTITY_TRANSFORM): SceneObject {
  const all = contours.flat();
  const bounds = { minX: Math.min(...all.map((p) => p.x)), minY: Math.min(...all.map((p) => p.y)), maxX: Math.max(...all.map((p) => p.x)), maxY: Math.max(...all.map((p) => p.y)) };
  const paths = [{ color: '#000000', polylines: contours.map((points) => ({ points, closed: true })) }];
  return { kind: 'shape', id, spec: { kind: 'polyline', points: contours[0], closed: true }, color: '#000000', bounds, transform, paths } as unknown as SceneObject;
}
export function project(objects: SceneObject[], layers: Layer[]): Project {
  const base = createProject(DEV);
  return { ...base, scene: { ...base.scene, objects, layers, groups: [] } } as Project;
}
export function gcodeOf(p: Project): string {
  const r = prepareOutput(p);
  if (!r.ok) return `NOT OK ${JSON.stringify(r).slice(0, 800)}`;
  return emitPreparedGcode(r).gcode;
}
export function encode(raster: VectorRaster) {
  return { dataUrl: 'data:image/png;base64,baked', lumaBase64: b64(raster.luma) };
}

export type Seg = { a: P; b: P; s: number };
export function burnedSegments(g: string, bedH = DEV.bedHeight): Seg[] {
  const out: Seg[] = [];
  let x = 0, y = 0, S = 0, mode = 0, laserOn = false;
  for (const raw of g.split('\n')) {
    const line = raw.replace(/;.*$/, '').trim(); if (!line) continue;
    if (/\bM[34]\b/.test(line)) laserOn = true;
    if (/\bM5\b/.test(line)) laserOn = false;
    const m = line.match(/^G0?([0-3])(?![0-9])/); if (m) mode = Number(m[1]);
    const nx = line.match(/X(-?[\d.]+)/), ny = line.match(/Y(-?[\d.]+)/), ns = line.match(/S(-?[\d.]+)/);
    if (ns) S = Number(ns[1]);
    const x1 = nx ? Number(nx[1]) : x; const y1 = ny ? Number(ny[1]) : y;
    if (mode === 1 && S > 0 && laserOn && (nx || ny)) out.push({ a: { x, y: bedH - y }, b: { x: x1, y: bedH - y1 }, s: S });
    x = x1; y = y1;
  }
  return out;
}

export function xf(p: P, t: Transform): P {
  let x = p.x * t.scaleX, y = p.y * t.scaleY;
  if (t.mirrorX) x = -x; if (t.mirrorY) y = -y;
  const r = (t.rotationDeg * Math.PI) / 180;
  return { x: x * Math.cos(r) - y * Math.sin(r) + t.x, y: x * Math.sin(r) + y * Math.cos(r) + t.y };
}
function segDist(p: P, a: P, b: P): number {
  const dx = b.x - a.x, dy = b.y - a.y; const L = dx * dx + dy * dy;
  const t = L === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / L));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}
export function edgeDist(p: P, contours: P[][]): number {
  let d = Infinity;
  for (const c of contours) for (let i = 0; i < c.length; i++) d = Math.min(d, segDist(p, c[i]!, c[(i + 1) % c.length]!));
  return d;
}
export function circle(cx: number, cy: number, rx: number, ry = rx, n = 96, start = 0.013): P[] {
  return Array.from({ length: n }, (_, i) => { const a = start + (i / n) * 2 * Math.PI; return { x: cx + rx * Math.cos(a), y: cy + ry * Math.sin(a) }; });
}
export const rect = (x0: number, y0: number, x1: number, y1: number): P[] => [{ x: x0, y: y0 }, { x: x1, y: y0 }, { x: x1, y: y1 }, { x: x0, y: y1 }];

export type SMap = { nx: number; ny: number; g: number; x0: number; y0: number; s: Float32Array };
/** Paint each burned segment's ribbon (+-pitch/2) with its S value on a fine grid. Samples the ribbon CENTRE cell row only when ribbonless=true. */
export function sMap(segs: Seg[], pitch: number, region: { x0: number; y0: number; x1: number; y1: number }, g = 0.05): SMap {
  const nx = Math.round((region.x1 - region.x0) / g), ny = Math.round((region.y1 - region.y0) / g);
  const s = new Float32Array(nx * ny);
  const half = pitch / 2;
  for (const sg of segs) {
    const dx = sg.b.x - sg.a.x, dy = sg.b.y - sg.a.y; const L = Math.hypot(dx, dy); if (L < 1e-9) continue;
    const ux = dx / L, uy = dy / L;
    const minX = Math.min(sg.a.x, sg.b.x) - half, maxX = Math.max(sg.a.x, sg.b.x) + half, minY = Math.min(sg.a.y, sg.b.y) - half, maxY = Math.max(sg.a.y, sg.b.y) + half;
    const i0 = Math.max(0, Math.floor((minX - region.x0) / g)), i1 = Math.min(nx - 1, Math.ceil((maxX - region.x0) / g));
    const j0 = Math.max(0, Math.floor((minY - region.y0) / g)), j1 = Math.min(ny - 1, Math.ceil((maxY - region.y0) / g));
    for (let j = j0; j <= j1; j++) for (let i = i0; i <= i1; i++) {
      const px = region.x0 + (i + 0.5) * g - sg.a.x, py = region.y0 + (j + 0.5) * g - sg.a.y;
      const along = px * ux + py * uy, perp = Math.abs(-px * uy + py * ux);
      if (along >= 0 && along <= L && perp < half) s[j * nx + i] = Math.max(s[j * nx + i]!, sg.s);
    }
  }
  return { nx, ny, g, x0: region.x0, y0: region.y0, s };
}

/** Compare before/after S maps; classify differences by distance to mask edge. */
export function diffMaps(before: SMap, after: SMap, maskScene: P[][], farMm: number, sMax: number) {
  const a = before.g * before.g;
  let burnB = 0, burnA = 0, same = 0, diff = 0, diffFar = 0, newBurn = 0, newBurnFar = 0, lost = 0, lostFar = 0, fullPowerNew = 0;
  let maxDS = 0, maxDSFar = 0, sumB = 0, sumA = 0; let maxNewDist = 0; let worstAt: P | null = null;
  for (let j = 0; j < before.ny; j++) for (let i = 0; i < before.nx; i++) {
    const k = j * before.nx + i; const b = before.s[k]!, c = after.s[k]!;
    if (b > 0) burnB++; if (c > 0) burnA++; sumB += b; sumA += c;
    if (b === c) { if (b > 0) same++; continue; }
    const p = { x: before.x0 + (i + 0.5) * before.g, y: before.y0 + (j + 0.5) * before.g };
    const d = maskScene.length ? edgeDist(p, maskScene) : Infinity;
    const far = d > farMm;
    diff++; if (far) diffFar++;
    const ds = Math.abs(b - c); if (ds > maxDS) { maxDS = ds; } if (far && ds > maxDSFar) { maxDSFar = ds; worstAt = p; }
    if (b === 0 && c > 0) { newBurn++; if (far) newBurnFar++; if (c >= sMax && b < sMax) fullPowerNew++; maxNewDist = Math.max(maxNewDist, d); }
    else if (b > 0 && c === 0) { lost++; if (far) lostFar++; }
    else if (c >= sMax && b < sMax) fullPowerNew++;
  }
  const r3 = (v: number) => Math.round(v * 1000) / 1000;
  return {
    burnBeforeMm2: r3(burnB * a), burnAfterMm2: r3(burnA * a), equalBurnMm2: r3(same * a), diffMm2: r3(diff * a), diffFarMm2: r3(diffFar * a),
    newBurnMm2: r3(newBurn * a), newBurnFarMm2: r3(newBurnFar * a), lostMm2: r3(lost * a), lostFarMm2: r3(lostFar * a), raisedToFullPowerMm2: r3(fullPowerNew * a),
    maxDeltaS: maxDS, maxDeltaSFar: maxDSFar, sumSBefore: Math.round(sumB * a), sumSAfter: Math.round(sumA * a), worstFarAt: worstAt,
  };
}
export function rowYs(segs: Seg[]): number[] {
  const ys = new Set<number>();
  for (const s of segs) if (Math.abs(s.a.y - s.b.y) < 1e-6) ys.add(Math.round(s.a.y * 1000) / 1000);
  return [...ys].sort((p, q) => p - q);
}
