export function lineLayer(id = 'L', extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id, color: '#000000', name: id, mode: 'line' }), speed: 1000, power: 80, ...extra } as Layer;
}
export function fillLayer(id = 'F', extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id, color: '#000000', name: id, mode: 'fill' }), speed: 3000, power: 30, hatchSpacingMm: 0.1, ...extra } as Layer;
}
export function proj(objects: SceneObject[], layers: Layer[]): Project {
  const base = createProject();
  return { ...base, optimization: { ...DEFAULT_PROJECT_OPTIMIZATION }, scene: { ...base.scene, objects, layers, groups: [] } };
}
export function setProject(project: Project, selected: string[]): void {
  useStore.setState({ project, selectedObjectId: selected[0] ?? null, additionalSelectedIds: new Set(selected.slice(1)), undoStack: [], redoStack: [], dirty: false });
}
export function burn(project: Project) {
  const prepared = prepareOutput(project);
  if (!prepared.ok) return { ok: false, gcode: '', moves: [] as LitMove[], why: JSON.stringify(prepared.preflight).slice(0, 300) };
  const g = emitPreparedGcode(prepared).gcode;
  return { ok: true, gcode: g, moves: litMoves(g, project.device), why: '' };
}
const rect = (id: string, w: number, h: number, t: Partial<Transform> = {}, extra: Record<string, unknown> = {}) =>
  ({ ...createRectangle({ id, color: '#000000', spec: { widthMm: w, heightMm: h, cornerRadiusMm: 0 }, transform: { ...IDENTITY_TRANSFORM, ...t } }), operationIds: ['L'], ...extra }) as SceneObject;
const circ = (id: string, d: number, t: Partial<Transform> = {}, extra: Record<string, unknown> = {}) =>
  ({ ...createEllipse({ id, color: '#000000', spec: { widthMm: d, heightMm: d }, transform: { ...IDENTITY_TRANSFORM, ...t } }), operationIds: ['L'], ...extra }) as SceneObject;
const ngon = (cx: number, cy: number, r: number, n = 720) => Array.from({ length: n }, (_, i) => ({ x: cx + r * Math.cos((2 * Math.PI * i) / n), y: cy + r * Math.sin((2 * Math.PI * i) / n) }));
const svg = (id: string, polys: { x: number; y: number }[][], extra: Record<string, unknown> = {}, pathExtra: Record<string, unknown> = {}): ImportedSvg =>
  ({
    kind: 'imported-svg',
    id,
    source: `${id}.svg`,
    bounds: { minX: 0, minY: 0, maxX: 1, maxY: 1 },
    transform: IDENTITY_TRANSFORM,
    operationIds: ['L'],
    paths: [{ color: '#000000', ...pathExtra, polylines: polys.map((points) => ({ closed: true, points })) }],
    ...extra,
  }) as ImportedSvg;

/** Oracle: each source burned alone, decoded, filled under its own rule, then unioned. */
function oracleGroups(objs: SceneObject[], layers: Layer[], ruleOf: (o: SceneObject) => 'evenodd' | 'nonzero'): Group[] {
  return objs.map((o) => ({ contours: loops(burn(proj([o], layers)).moves), rule: ruleOf(o) }));
}
const defaultRule = (o: SceneObject): 'evenodd' | 'nonzero' =>
  o.kind === 'text' ? 'nonzero' : 'paths' in o && o.paths[0]?.fillRule === 'nonzero' ? 'nonzero' : 'evenodd';

function runCase(name: string, objs: SceneObject[], analytic?: { area?: number; perim?: number }, layers: Layer[] = [lineLayer()]) {
  const p0 = proj(objs, layers);
  setProject(p0, objs.map((o) => o.id));
  const t0 = performance.now();
  st().weldSelection();
  const ms = performance.now() - t0;
  const p1 = st().project;
  const changed = p1 !== p0;
  const after = burn(p1);
  const actualContours = loops(after.moves);
  const oracle = oracleGroups(objs, layers, defaultRule);
  const cmp = compareRegions(oracle, [{ contours: actualContours, rule: 'evenodd' }]);
  const cmpNZ = compareRegions(oracle, [{ contours: actualContours, rule: 'nonzero' }]);
  const res = p1.scene.objects.filter((o) => o.id.startsWith('welded'));
  out(
    `${name}: changed ${changed} in ${ms.toFixed(0)} ms; objects ${p0.scene.objects.length}->${p1.scene.objects.length}; welded contours ${actualContours.length}; ` +
      `oracle area ${f3(cmp.oracle)} actual(evenodd) ${f3(cmp.actual)} actual(nonzero) ${f3(cmpNZ.actual)} xor ${f3(cmp.xor)}; ` +
      `lit after ${f3(totalLen(after.moves))} perim ${f3(perimeter(actualContours))}` +
      (analytic?.area !== undefined ? `; analytic area ${f3(analytic.area)}` : '') +
      (analytic?.perim !== undefined ? ` analytic perim ${f3(analytic.perim)}` : '') +
      `; welded objs ${res.length}; toasts ${JSON.stringify(toasts())}`,
  );
  return { p0, p1, cmp, after };
}

  runCase('a two rects offset (10,5)', [rect('A', 20, 10, { x: 10, y: 10 }), rect('B', 20, 10, { x: 20, y: 15 })], { area: 350, perim: 90 });
  const lens = 200 * Math.acos(0.5) - 5 * Math.sqrt(300);

  runCase('c touching rects edge', [rect('A', 20, 10, { x: 10, y: 10 }), rect('B', 20, 10, { x: 30, y: 10 })], { area: 400, perim: 100 });

  // ring (svg even-odd donut) with a bar through the hole
  let barArea = 0;
  const N = 200000;
  for (let i = 0; i < N; i++) {
    const y = -2 + (4 * (i + 0.5)) / N;
    const xin = Math.sqrt(100 - y * y);

  }
  const ring = svg('RING', [ngon(50, 50, 20), ngon(50, 50, 10)]);

  runCase('d ring + bar across hole', [ring, bar], { area: Math.PI * 300 + barArea });
  runCase('e small circle inside big (absorbed)', [circ('A', 40, { x: 10, y: 10 }), circ('B', 10, { x: 25, y: 25 })], { area: Math.PI * 400 });
  runCase('e2 island inside ring hole (kept)', [ring, circ('I', 8, { x: 46, y: 46 }), bar], {});
  runCase('f rotated/mirrored/non-uniform', [
    rect('A', 20, 10, { x: 40, y: 40, rotationDeg: 30, scaleX: 2, scaleY: 0.5 }),
    rect('B', 20, 10, { x: 45, y: 35, rotationDeg: -25, mirrorX: true, scaleX: 1.5, scaleY: 1.2 }),

async function text(id: string, content: string, fontKey: string, fontFile: string, sizeMm: number, t: Partial<Transform>, weldOverlaps: boolean): Promise<TextObject> {
  const r = await textToPolylines({ content, sizeMm, alignment: 'left', lineHeight: 1.2, letterSpacing: 0, color: '#000000', fontBuffer: font(fontFile) });
  return {
    kind: 'text',
    id,
    content,
    fontKey,
    sizeMm,
    alignment: 'left',
    lineHeight: 1.2,
    letterSpacing: 0,
    color: '#000000',
    weldOverlaps,
    bounds: r.bounds,
    transform: { ...IDENTITY_TRANSFORM, ...t },
    paths: r.paths.map((p) => ({ ...p, operationIds: ['L'] })),
  } as unknown as TextObject;
}

  for (const [key, file] of [['pacifico', 'Pacifico-Regular.ttf'], ['great-vibes', 'GreatVibes-Regular.ttf'], ['alex-brush', 'AlexBrush-Regular.ttf']] as const) {
    const s = await text('SC', 'Wedding Love', key, file, 15, { x: 10, y: 10, rotationDeg: 12, mirrorX: true }, false);
    runCase(`m script ${key} unwelded alone, rotated+mirrored`, [s]);
    // Fill burn before vs after
    const fl = [fillLayer('L')];
    const p0 = proj([s], fl);
    setProject(p0, ['SC']);
    const b0 = burn(p0);
    st().weldSelection();
    const b1 = burn(st().project);
    out(`m   Fill hatch lit before ${f3(totalLen(b0.moves))} after ${f3(totalLen(b1.moves))} (x0.1 = area ${f3(totalLen(b0.moves) * 0.1)} / ${f3(totalLen(b1.moves) * 0.1)})`);
  }
});
