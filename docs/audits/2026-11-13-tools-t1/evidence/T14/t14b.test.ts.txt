// ---------- fixture ----------
const W = 32, H = 8;
function ramp(): Uint8Array {
  const l = new Uint8Array(W * H);
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const v = Math.round((x * 255) / (W - 1)) + (y % 2 === 0 ? -6 : 6) + (y >> 1);
    l[y * W + x] = Math.max(0, Math.min(255, v));
  }
  return l;
}
const b64 = (u: Uint8Array) => Buffer.from(u).toString('base64');
function mkImage(id: string, luma: Uint8Array, x0: number, extra: Partial<RasterImage> = {}): RasterImage {
  return { kind: 'raster-image', id, source: `${id}.png`, dataUrl: 'data:image/png;base64,iVBORw0KGgo=', pixelWidth: W, pixelHeight: H,
    bounds: { minX: x0, minY: 10, maxX: x0 + W, maxY: 10 + H }, transform: IDENTITY_TRANSFORM, color: '#808080', dither: 'threshold', linesPerMm: 1,
    lumaBase64: b64(luma), ...extra } as RasterImage;
}
function mkLayer(extra: Partial<Layer> = {}): Layer {
  return { ...createLayer({ id: 'img', color: '#808080', name: 'Image', mode: 'image' }), power: 80, minPower: 0, speed: 3000, linesPerMm: 1,
    ditherAlgorithm: 'grayscale', dotWidthCorrectionMm: 0, ...extra } as Layer;
}

// ---------- G-code decode ----------
function gcodeOf(p: Project): string {
  const r = prepareOutput(p);
  if (!r.ok) return `NOT OK ${JSON.stringify((r as never as { preflight: unknown }).preflight).slice(0, 400)}`;
  return emitPreparedGcode(r).gcode;
}
// Returns S per scene pixel for an image at bounds minX=x0, minY=10 (1 px = 1 mm).
function decode(g: string, x0: number, bedH: number): Float64Array {
  const s = new Float64Array(W * H);
  let x = 0, y = 0, S = 0, mode = 0;
  for (const raw of g.split('\n')) {
    const line = raw.replace(/;.*$/, '').trim(); if (!line) continue;
    const m = line.match(/^G0?([0-3])(?![0-9])/); if (m) mode = Number(m[1]);
    const nx = line.match(/X(-?[\d.]+)/), ny = line.match(/Y(-?[\d.]+)/), ns = line.match(/S(-?[\d.]+)/);
    if (ns) S = Number(ns[1]);
    const x1 = nx ? Number(nx[1]) : x; const y1 = ny ? Number(ny[1]) : y;
    if (mode === 1 && S > 0 && nx && y1 === y) {
      const row = Math.round(bedH - y - 10 - 0.5);
      const lo = Math.min(x, x1), hi = Math.max(x, x1);
      for (let px = 0; px < W; px++) { const c = x0 + px + 0.5; if (c > lo && c < hi && row >= 0 && row < H) s[row * W + px] = S; }
    }
    x = x1; y = y1;
  }
  return s;
}

// ---------- oracle ----------
function oracleTone(level: number, b: number, c: number, g: number): number {
  // LightBurn-style: brightness offset b% of full scale, contrast about mid-grey,
  // gamma as exponent 1/g (ADR-671: higher gamma lightens). Each stage clamps; one rounding.
  let v = level;
  v = Math.min(255, Math.max(0, v + (b / 100) * 255));
  v = Math.min(255, Math.max(0, (v - 128) * (1 + c / 100) + 128));
  v = 255 * Math.pow(v / 255, 1 / g);
  return Math.round(v);
}
const KERNELS: Record<string, { k: number[][]; d: number }> = {
  'floyd-steinberg': { k: [[1, 0, 7], [-1, 1, 3], [0, 1, 5], [1, 1, 1]], d: 16 },
  jarvis: { k: [[1, 0, 7], [2, 0, 5], [-2, 1, 3], [-1, 1, 5], [0, 1, 7], [1, 1, 5], [2, 1, 3], [-2, 2, 1], [-1, 2, 3], [0, 2, 5], [1, 2, 3], [2, 2, 1]], d: 48 },
  stucki: { k: [[1, 0, 8], [2, 0, 4], [-2, 1, 2], [-1, 1, 4], [0, 1, 8], [1, 1, 4], [2, 1, 2], [-2, 2, 1], [-1, 2, 2], [0, 2, 4], [1, 2, 2], [2, 2, 1]], d: 42 },
  atkinson: { k: [[1, 0, 1], [2, 0, 1], [-1, 1, 1], [0, 1, 1], [1, 1, 1], [0, 2, 1]], d: 8 },
  burkes: { k: [[1, 0, 8], [2, 0, 4], [-2, 1, 2], [-1, 1, 4], [0, 1, 8], [1, 1, 4], [2, 1, 2]], d: 32 },
  sierra3: { k: [[1, 0, 5], [2, 0, 3], [-2, 1, 2], [-1, 1, 4], [0, 1, 5], [1, 1, 4], [2, 1, 2], [-1, 2, 2], [0, 2, 3], [1, 2, 2]], d: 32 },
  sierra2: { k: [[1, 0, 4], [2, 0, 3], [-2, 1, 1], [-1, 1, 2], [0, 1, 3], [1, 1, 2], [2, 1, 1]], d: 16 },
  'sierra-lite': { k: [[1, 0, 2], [-1, 1, 1], [0, 1, 1]], d: 4 },
};
// Standard 4x4 Bayer index matrix -> thresholds (i + 0.5) * 16
const BAYER = [[0, 8, 2, 10], [12, 4, 14, 6], [3, 11, 1, 9], [15, 7, 13, 5]].map((r) => r.map((i) => i * 16 + 8));
function oracleS(luma: Uint8Array, alg: string, sMax: number, sMin: number): Float64Array {
  const o = new Float64Array(luma.length);
  if (alg === 'grayscale') { for (let i = 0; i < luma.length; i++) { const t = (255 - luma[i]!) / 255; o[i] = t <= 0 ? 0 : Math.round(sMin + t * (sMax - sMin)); } return o; }
  if (alg === 'threshold') { for (let i = 0; i < luma.length; i++) o[i] = luma[i]! < 128 ? sMax : 0; return o; }
  if (alg === 'ordered') { for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) o[y * W + x] = luma[y * W + x]! < BAYER[y % 4]![x % 4]! ? sMax : 0; return o; }
  const K = KERNELS[alg]!; const buf = Float64Array.from(luma);
  for (let y = 0; y < H; y++) {
    const ltr = y % 2 === 0;
    for (let k = 0; k < W; k++) {
      const x = ltr ? k : W - 1 - k; const i = y * W + x; const old = buf[i]!; const q = old < 128 ? 0 : 255; const e = old - q;
      o[i] = q === 0 ? sMax : 0;
      for (const [dx0, dy, w] of K.k) { const xx = x + (ltr ? dx0! : -dx0!), yy = y + dy!; if (xx < 0 || xx >= W || yy >= H) continue; buf[yy * W + xx] += (e * w!) / K.d; }
    }
  }
  return o;
}
function flipY<T extends Uint8Array | Float64Array>(a: T): T { const o = (a instanceof Uint8Array ? new Uint8Array(a.length) : new Float64Array(a.length)) as T; for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) o[(H - 1 - y) * W + x] = a[y * W + x]!; return o; }
function diff(a: Float64Array, b: Float64Array): { n: number; max: number; first: string } {
  let n = 0, max = 0, first = '';
  for (let i = 0; i < a.length; i++) { const d = Math.abs(a[i]! - b[i]!); if (d > 1e-9) { n++; if (!first) first = `px(${i % W},${Math.floor(i / W)}) got ${a[i]} want ${b[i]}`; } max = Math.max(max, d); }
  return { n, max, first };
}

// ---------- dialog driver ----------
type Step = { kind: 'num'; name: string; value: string } | { kind: 'dither'; value: string } | { kind: 'toggle'; name: string };
async function runDialog(imageId: string, steps: Step[], finish: 'ok' | 'cancel' = 'ok'): Promise<{ applied: AdjustImageApply | null; fieldValues: Record<string, string> }> {
  const image = st().project.scene.objects.find((o) => o.id === imageId) as RasterImage;
  const layer = primaryOperationForObject(image, st().project.scene.layers)!;
  let applied: AdjustImageApply | null = null;
  let closed = false;
  // Same as CommandShell.tsx AdjustDialog onApply
  const onApply = (patch: AdjustImageApply) => { applied = patch; closed = true; st().setRasterImageAdjustments(image.id, patch.imagePatch, { id: layer.id, patch: patch.layerPatch }); };
  const host = document.createElement('div'); document.body.appendChild(host);
  let root: Root | null = null;
  await act(async () => { root = createRoot(host); root.render(createElement(AdjustImageDialog, { image, layer, onApply, onCancel: () => { closed = true; } })); });
  const fieldValues: Record<string, string> = {};
  for (const n of ['brightness', 'contrast', 'gamma', 'minPower', 'imageDpi', 'lineIntervalMm', 'dotWidthCorrectionMm']) fieldValues[n] = host.querySelector<HTMLInputElement>(`input[name="${n}"]`)?.value ?? '?';
  fieldValues['dither'] = host.querySelector<HTMLInputElement>('input[name="ditherAlgorithm"]:checked')?.value ?? '?';
  fieldValues['negative'] = String(host.querySelector<HTMLInputElement>('input[name="negativeImage"]')?.checked);
  for (const s of steps) {
    if (s.kind === 'num') { const el = host.querySelector<HTMLInputElement>(`input[name="${s.name}"]`)!; act(() => { el.value = s.value; Simulate.change(el); }); act(() => { Simulate.blur(el); }); }
    else if (s.kind === 'dither') { const el = host.querySelector<HTMLInputElement>(`input[name="ditherAlgorithm"][value="${s.value}"]`)!; act(() => el.click()); }
    else { const el = host.querySelector<HTMLInputElement>(`input[name="${s.name}"]`)!; act(() => { el.checked = !el.checked; Simulate.change(el); }); }
  }
  if (finish === 'ok') await act(async () => { Simulate.submit(host.querySelector('form')!); });
  else { const cancel = [...host.querySelectorAll('button')].find((b) => b.textContent === 'Cancel')!; await act(async () => { Simulate.click(cancel); }); }
  await act(async () => root!.unmount()); host.remove();
  void closed;
  return { applied, fieldValues };
}
const num = (name: string, value: string): Step => ({ kind: 'num', name, value });

it('F shared operation and sub-pass', async () => {
  const luma = ramp(); const bed = DEFAULT_DEVICE_PROFILE.bedHeight;
  // F1: two photos on one operation, neither owns an override. Adjust Image on B, tick Negative.
  load([mkImage('A', luma, 10), mkImage('B', luma, 60)], mkLayer({ ditherAlgorithm: 'grayscale' }));
  const a0 = decode(gcodeOf(st().project), 10, bed);
  await runDialog('B', [{ kind: 'toggle', name: 'negativeImage' }, num('brightness', '10')]);
  const a1 = decode(gcodeOf(st().project), 10, bed);
  let sum0 = 0, sum1 = 0; for (let i = 0; i < a0.length; i++) { sum0 += a0[i]!; sum1 += a1[i]!; }
  out(`F1 OK on B (Negative ticked): layer negative ${st().project.scene.layers[0]!.negativeImage}; photo A pixels changed ${diff(a1, a0).n}/${W * H}; A total S ${sum0} -> ${sum1}`);
  // F2: sub-pass. Operation with a second image pass (threshold); image A owns a primary override.
  const sub = { id: 's2', label: 'Pass 2', enabled: true, settings: { ...mkLayer({ ditherAlgorithm: 'threshold', power: 30 }) } } as never;

  load([A], mkLayer({ ditherAlgorithm: 'grayscale', subLayers: [sub] }));
  const g0 = gcodeOf(st().project);
  await runDialog('A', [{ kind: 'dither', value: 'stucki' }]);
  const g1 = gcodeOf(st().project);
  const passes = (g: string) => g.split(/; image layer /).slice(1).map((p) => p.split('\n')[0]);

});
