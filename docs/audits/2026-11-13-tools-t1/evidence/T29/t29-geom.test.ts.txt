/** Oracle groups: each source path in world space with its own fill rule (text non-zero). */
function oracleGroups(objects: SceneObject[]): Group[] {
  return objects.flatMap((o) => {
    const m = materializeVectorObject(o as never);
    return m.paths.map((p) => ({
      contours: p.polylines.map((l) => l.points.map((q) => ({ x: q.x, y: q.y }))),
      rule: o.kind === 'text' || p.fillRule === 'nonzero' ? ('nonzero' as const) : ('evenodd' as const),
    }));
  });
}

function runCase(name: string, objects: SceneObject[], expect: { area?: number; perim?: number } = {}) {
  const L = layer('L', 'line', 1000, 50);
  const objs = objects.map((o) => ({ ...o, operationIds: ['L'] }) as SceneObject);
  const p0 = proj(objs, [L]);
  const ids = objs.map((o) => o.id);
  setProject(p0, ids);
  const ops = unionSilhouetteOperations(p0.scene, ids).map((l) => l.id);
  const t0 = performance.now();
  const applied = st().unionSilhouetteSelection('L');
  const ms = performance.now() - t0;
  const p1 = st().project;
  const res = p1.scene.objects.find((o) => o.id.startsWith('union-')) as ImportedSvg | undefined;
  if (!applied || res === undefined) {
    out(`${name}: menu ops [${ops}] applied=${applied} objects=${p1.scene.objects.length} toasts ${JSON.stringify(toasts())}`);
    return;
  }
  const oracle = oracleGroups(objs);
  const resContours = res.paths.flatMap((p) => p.polylines.map((l) => l.points.map((q) => ({ x: q.x, y: q.y }))));
  const geo = compareRegions(oracle, [{ contours: resContours, rule: 'evenodd' }], 0.01);
  const b = burn(p1);
  const burnLoops = loops(b.moves);
  const bg = compareRegions(oracle, [{ contours: burnLoops, rule: 'evenodd' }], 0.01);
  const lit = totalLen(b.moves);
  out(
    `${name}: ops [${ops}] ${f3(ms)} ms; result contours ${resContours.length}; area oracle ${f3(geo.oracle)} result ${f3(geo.actual)} xor ${f3(geo.xor)}; ` +
      `burn loops ${burnLoops.length} burn area ${f3(bg.actual)} xor ${f3(bg.xor)}; lit ${f3(lit)} perimeter(result) ${f3(perimeter(resContours))}` +
      (expect.area !== undefined ? `; ANALYTIC area ${f3(expect.area)}` : '') +
      (expect.perim !== undefined ? ` perim ${f3(expect.perim)}` : '') +
      `; objects after ${p1.scene.objects.length}; toast ${JSON.stringify(toasts())}`,
  );
}

it('G1 overlapping rectangles (ADR fixture)', () => {
  runCase('G1 overlap 60x40@(40,40)+(70,60)', [rect('A', 60, 40, { x: 40, y: 40 }), rect('B', 60, 40, { x: 70, y: 60 })], { area: 4200, perim: 300 });
});
it('G2 touching and disjoint', () => {
  runCase('G2a touching edge', [svg('A', [box(0, 0, 10, 10)]), svg('B', [box(10, 0, 10, 10)])], { area: 200, perim: 60 });
  runCase('G2b corner touch', [svg('A', [box(0, 0, 10, 10)]), svg('B', [box(10, 10, 10, 10)])], { area: 200, perim: 80 });
  runCase('G2c disjoint', [svg('A', [box(0, 0, 10, 10)]), svg('B', [box(30, 0, 5, 5)])], { area: 125, perim: 60 });
});
it('G3 nested, holes, islands', () => {
  // separate objects nested: inner swallowed
  runCase('G3a nested separate squares', [svg('A', [box(0, 0, 40, 40)]), svg('B', [box(10, 10, 20, 20)])], { area: 1600, perim: 160 });
  // ring as one object (even-odd hole), plus an island inside the hole
  runCase('G3b ring+island', [svg('R', [box(0, 0, 40, 40), box(10, 10, 20, 20)]), svg('I', [box(15, 15, 10, 10)])], { area: 1200 + 100, perim: 160 + 80 + 40 });
  // ring + shape filling half the hole
  runCase('G3c ring+half hole', [svg('R', [box(0, 0, 40, 40), box(10, 10, 20, 20)]), svg('H', [box(5, 5, 15, 30)])], { area: 1200 + 200, perim: 160 + 40 + 20 * 2 });
  // island inside a hole inside a ring, as ONE object with three nested contours (even-odd)
  runCase('G3d one-object triple nest', [svg('T', [box(0, 0, 40, 40), box(10, 10, 20, 20), box(15, 15, 10, 10)]), svg('X', [box(50, 0, 5, 5)])], { area: 1300 + 25, perim: 160 + 80 + 40 + 20 });
});

  runCase('G4b text "OB8a" + bar under it', [await textObj('T', 'OB8a', { x: 20, y: 20 }, ['L']), svg('BAR', [box(18, 30, 60, 3)])]);

it('G5 self-crossing and curves', () => {
  const star: P[] = Array.from({ length: 5 }, (_, i) => {
    const a = -Math.PI / 2 + (i * 4 * Math.PI) / 5;
    return { x: 50 + 20 * Math.cos(a), y: 50 + 20 * Math.sin(a) };
  });
  runCase('G5a pentagram even-odd', [svg('S', [star])]);
  runCase('G5b pentagram nonzero', [svg('S', [star], {}, true, 'nonzero')]);
  runCase('G5c two circles d20 overlapping', [circ('C1', 20, { x: 40, y: 40 }), circ('C2', 20, { x: 50, y: 40 })], {
    // analytic lens: two r=10 circles, centres 10 apart
    area: 2 * Math.PI * 100 - (2 * 100 * Math.acos(10 / 20) - 5 * Math.sqrt(400 - 100)),
    perim: 2 * (2 * Math.PI * 10) - 2 * (2 * 10 * Math.acos(10 / 20)),
  });
});

  runCase('G7a open path in selection', [svg('O', [[{ x: 0, y: 0 }, { x: 10, y: 0 }, { x: 10, y: 10 }]], {}, false), svg('A', [box(0, 0, 5, 5)])]);

  runCase('G7g huge 2000 mm squares', [svg('A', [box(0, 0, 2000, 2000)]), svg('B', [box(1000, 1000, 2000, 2000)])], { area: 7_000_000, perim: 12000 });
