// Decoder: returns machine-coordinate midpoints of each contiguous lit run whose S
// is below the run maximum (the tab gaps burned at tab power).
function tabSpans(gcode: string): { mid: Vec2; len: number; s: number }[] {
  let x = 0, y = 0, s = 0, mode = 0, on = false;
  const lit: { a: Vec2; b: Vec2; s: number }[] = [];
  for (const raw of gcode.split(/\r?\n/)) {
    const line = raw.replace(/;.*$/, '').toUpperCase();
    const w: Record<string, number> = {};
    const ms: number[] = [];
    for (const m of line.matchAll(/([A-Z])(-?\d*\.?\d+)/g)) {
      if (m[1] === 'M') ms.push(Number(m[2]));
      else if (m[1] === 'G') { const g = Number(m[2]); if (g <= 3) mode = g; }
      else w[m[1]!] = Number(m[2]);
    }
    for (const m of ms) { if (m === 3 || m === 4) on = true; if (m === 5) on = false; }
    if (w.S !== undefined) s = w.S;
    const nx = w.X ?? x, ny = w.Y ?? y;
    if ((w.X !== undefined || w.Y !== undefined) && mode === 1 && on && s > 0) lit.push({ a: { x, y }, b: { x: nx, y: ny }, s });
    x = nx; y = ny;
  }
  const sMax = Math.max(...lit.map((m) => m.s));
  const spans: { mid: Vec2; len: number; s: number }[] = [];
  let cur: { a: Vec2; b: Vec2; s: number }[] = [];
  const flush = () => {
    if (cur.length === 0) return;
    const len = cur.reduce((t, m) => t + Math.hypot(m.b.x - m.a.x, m.b.y - m.a.y), 0);
    // midpoint along the run
    let half = len / 2, mid = cur[0]!.a;
    for (const m of cur) {
      const l = Math.hypot(m.b.x - m.a.x, m.b.y - m.a.y);
      if (half <= l) { mid = { x: m.a.x + ((m.b.x - m.a.x) * half) / l, y: m.a.y + ((m.b.y - m.a.y) * half) / l }; break; }
      half -= l;
    }
    spans.push({ mid, len, s: cur[0]!.s });
    cur = [];
  };
  for (const m of lit) {
    if (m.s < sMax) {
      if (cur.length > 0 && Math.hypot(cur.at(-1)!.b.x - m.a.x, cur.at(-1)!.b.y - m.a.y) > 1e-6) flush();
      cur.push(m);
    } else flush();
  }
  flush();
  return spans;
}

function build(points: Vec2[], hand: Vec2[], tx: number, ty: number) {
  let o = {
    kind: 'imported-svg', id: 'panel', name: 'Panel', source: 'panel.svg',
    transform: { ...IDENTITY_TRANSFORM, x: tx, y: ty },
    bounds: { minX: Math.min(...points.map((p) => p.x)), minY: Math.min(...points.map((p) => p.y)), maxX: Math.max(...points.map((p) => p.x)), maxY: Math.max(...points.map((p) => p.y)) },
    paths: [{ color: '#000000', operationIds: ['cut'], polylines: [{ closed: true, points: [...points, points[0]!] }] }],
    operationIds: ['cut'],
  } as unknown as ImportedSvg;
  const anchors = hand.map((p) => {
    const an = projectCncTabAnchor(o, '#000000', { x: p.x + tx, y: p.y + ty });
    if (an === null) throw new Error('off outline');
    return an;
  });
  o = { ...o, laserTabAnchors: anchors } as ImportedSvg;
  const layer = { ...createLayer({ id: 'cut', name: 'cut', color: '#000000' }), mode: 'line' as const, power: 60, speed: 1200, tabsEnabled: true, tabSizeMm: 1, tabsPerShape: 4, tabCutPowerPercent: 20 };
  const base = createProject();
  const p: Project = { ...base, scene: { ...base.scene, objects: [o], layers: [layer], groups: [] } } as Project;
  return { p, o };
}

function runCase(label: string, points: Vec2[], hand: Vec2[], expected: Vec2[]) {
  const tx = 30, ty = 30;
  const { p, o } = build(points, hand, tx, ty);
  useStore.setState({ project: p, projectDocumentEpoch: 7, selectedObjectId: 'panel', additionalSelectedIds: new Set(), undoStack: [], redoStack: [], dirty: false });
  const toastsBefore = useToastStore.getState().toasts?.length ?? 0;
  const req = { currentWidthMm: 3, materialThicknessMm: 4, fitAllowanceMm: 0.1, detectionToleranceMm: 0.05 };
  const a = analyseJointResize([o], req);
  if (a.kind !== 'ok') throw new Error(a.error.message);
  const r = jointResizeSelectionMutation(useStore.getState(), { project: p, documentEpoch: 7, ids: ['panel'] }, req, a.value.candidates.map((c) => c.id));
  if (r.kind !== 'ok') throw new Error(r.error.message);
  useStore.setState(r.value);
  const after = useStore.getState().project;
  const o2 = after.scene.objects[0] as ImportedSvg;
  const toastsAfter = useToastStore.getState().toasts?.length ?? 0;
  log(`== ${label}: ${a.value.candidates.length} candidates (${a.value.candidates.map((c) => c.kind).join(',')}); toasts added ${toastsAfter - toastsBefore}; notices ${JSON.stringify(a.value.notices)}`);
  const followed = withTabsFollowingEdit(o, o2);
  hand.forEach((h, i) => {
    const pa = laserTabAnchorPosition(o2, o2.laserTabAnchors![i]!)!;
    const pf = laserTabAnchorPosition(followed.object, followed.object.laserTabAnchors![i]!)!;
    const loc = { x: pa.x - tx, y: pa.y - ty };
    const e = expected[i]!;
    log(`  tab ${i}: placed (${f(h.x)},${f(h.y)}) after (${f(loc.x)},${f(loc.y)}) moved ${f(Math.hypot(loc.x - h.x, loc.y - h.y))}; analytic fraction-kept (${f(e.x)},${f(e.y)}) err ${f(Math.hypot(loc.x - e.x, loc.y - e.y))}; withTabsFollowingEdit -> (${f(pf.x - tx)},${f(pf.y - ty)}) moved ${f(Math.hypot(pf.x - tx - h.x, pf.y - ty - h.y))}`);
  });
  const burn = (proj: Project) => {
    const prep = prepareOutput(proj);
    if (!prep.ok) throw new Error('prep ' + JSON.stringify(prep).slice(0, 300));
    return tabSpans(emitPreparedGcode(prep).gcode);
  };
  const sb = burn(p), sa = burn(after);
  log(`  G-code tab runs before (${sb.length}): ${sb.map((t) => `(${f(t.mid.x)},${f(t.mid.y)}) len ${f(t.len)} S${t.s}`).join(' ')}`);
  log(`  G-code tab runs after  (${sa.length}): ${sa.map((t) => `(${f(t.mid.x)},${f(t.mid.y)}) len ${f(t.len)} S${t.s}`).join(' ')}`);
  // pair nearest
  const pairs = sb.map((b) => Math.min(...sa.map((q) => Math.hypot(q.mid.x - b.mid.x, q.mid.y - b.mid.y))));
  log(`  nearest-run shift in machine mm: ${pairs.map(f).join(', ')}`);
}

it('R2a 100x50 panel, three 10-wide 3-deep finger tabs on the top edge', () => {
  // start (0,0), along bottom (+x), up right, back along top (fingers at y 50..53), down left.
  const pts: Vec2[] = [{ x: 0, y: 0 }, { x: 100, y: 0 }, { x: 100, y: 50 }];
  for (const x0 of [80, 55, 30]) pts.push({ x: x0, y: 50 }, { x: x0, y: 53 }, { x: x0 - 10, y: 53 }, { x: x0 - 10, y: 50 });
  pts.push({ x: 0, y: 50 });
  // L0 = 300 + 6*3 = 318; L1 = 324. Fraction-kept arc s1 = s0 * 324/318.
  const hand = [{ x: 50, y: 0 }, { x: 100, y: 25 }, { x: 0, y: 25 }, { x: 56, y: 50 }, { x: 62.5, y: 50 }];
  // analytic: bottom s0 50 -> 50.943; right s0 125 -> 127.358 => (100,27.358);
  // left s0 293 -> 298.528, left edge starts at 274 from (0,50) => (0,25.472);
  // (56,50): s0 = 150+20+3+10+3+14 = 200 -> 203.774; new (55,50) at s 203 => up the finger flank (55,50.774)
  // (62.5,50): s0 = 150+20+3+10+3+7.5 = 193.5 -> 196.934; new (70,50) at s 188 => x 70-8.934 = 61.066
  const k = 324 / 318;
  const expected = [{ x: 50 * k, y: 0 }, { x: 100, y: 125 * k - 100 }, { x: 0, y: 50 - (293 * k - 274) }, { x: 55, y: 50 + (200 * k - 203) }, { x: 70 - (193.5 * k - 188), y: 50 }];
  runCase('R2a', pts, hand, expected);
});

it('R2b 300x100 panel, 25 fingers 6 wide 3 deep on the top edge', () => {
  const pts: Vec2[] = [{ x: 0, y: 0 }, { x: 300, y: 0 }, { x: 300, y: 100 }];
  for (let i = 0; i < 25; i++) { const x0 = 297 - i * 12; pts.push({ x: x0, y: 100 }, { x: x0, y: 103 }, { x: x0 - 6, y: 103 }, { x: x0 - 6, y: 100 }); }
  pts.push({ x: 0, y: 100 });
  // L0 = 800 + 50*3 = 950; L1 = 1000.
  const k = 1000 / 950;
  const hand = [{ x: 150, y: 0 }, { x: 300, y: 50 }, { x: 0, y: 50 }];
  const expected = [{ x: 150 * k, y: 0 }, { x: 300, y: 350 * k - 300 }, { x: 0, y: 100 - (900 * k - 900) }];
  runCase('R2b', pts, hand, expected);
});
