type V = { x: number; y: number };
const svgOpen = (
  id: string,
  pts: V[][],
  t: Partial<Transform> = {},
  extra: Record<string, unknown> = {},
  op = 'L',
): ImportedSvg =>
  ({
    kind: 'imported-svg', id, source: `${id}.svg`, operationIds: [op],
    bounds: { minX: 0, minY: 0, maxX: 100, maxY: 100 }, transform: { ...IDENTITY_TRANSFORM, ...t },
    paths: [{ color: '#000000', polylines: pts.map((points) => ({ points, closed: false })) }],
    ...extra,
  }) as ImportedSvg;

function diag(p: Project): string {
  try {
    const j = compileJob(p) as unknown as { diagnostics?: unknown[] };
    return JSON.stringify(j.diagnostics ?? []).slice(0, 400);
  } catch (e) {
    return `compileJob threw ${String(e)}`;
  }
}
function report(tag: string, p: Project) {
  const b = burn(p);
  if (!b.ok) {
    out(`${tag}: NOT OK ${b.why}`);
    return 0;
  }
  const bb = bbox(b.moves);
  out(`${tag}: lit ${f3(totalLen(b.moves))} mm, moves ${b.moves.length}, bbox ${f3(bb.minX)},${f3(bb.minY)}..${f3(bb.maxX)},${f3(bb.maxY)} diag ${diag(p)}`);
  return totalLen(b.moves);
}
function closeAll(objects: SceneObject[], layers: Layer[], sel: string[]) {
  const p0 = proj(objects, layers);
  setProject(p0, sel);
  st().closeSelectedPaths();
  return { before: p0, after: st().project };
}
const polyInfo = (o: SceneObject | undefined) =>
  o && 'paths' in o
    ? o.paths.map((p) => p.polylines.map((l) => `${l.closed ? 'C' : 'O'}${l.points.length}`).join(' ')).join(' | ')
    : 'n/a';

it('S1 three-sided open box closes with a straight line; gap reported; G-code +gap', () => {
  const L = layer('L', 'line', 1000, 50);
  const o = svgOpen('a', [[{ x: 10, y: 10 }, { x: 50, y: 10 }, { x: 50, y: 40 }, { x: 10, y: 40 }]]);
  const { before, after } = closeAll([o], [L], ['a']);
  out(`S1 toast ${toasts().join(' // ')}; shape ${polyInfo(after.scene.objects[0])}`);
  const a = report('S1 before', before);
  const b = report('S1 after', after);
  out(`S1 delta ${f3(b - a)} (oracle 30.000)`);
  st().undo();
  out(`S1 undo restores exactly: ${JSON.stringify(st().project.scene) === JSON.stringify(before.scene)}`);
  st().redo();
  out(`S1 redo repeats exactly: ${JSON.stringify(st().project.scene) === JSON.stringify(after.scene)}`);
  const re = deserializeProject(serializeProject(st().project));
  const ro = (re as unknown as { ok?: boolean; project?: Project; value?: Project });
  const rp = ro.project ?? ro.value ?? (re as unknown as Project);
  out(`S1 save/reopen: ${polyInfo(rp.scene?.objects?.[0])}`);
});

it('S4 partial retrace: closing line runs back along the first segment (spur), kerf 0 and 0.1', () => {
  for (const kerf of [0, 0.1, -0.1]) {
    resetStore();
    clearToasts();
    const L = layer('L', 'line', 1000, 50, { kerfOffsetMm: kerf } as Partial<Layer>);
    // start (10,10) -> (40,10) -> (40,30) -> (25,30) -> (25,10): closing (25,10)->(10,10) retraces 15 mm of the first edge

  }
});

it('S8 near-coincident ends: duplicate seam point and zero-length segment', () => {
  const L = layer('L', 'line', 1000, 50);
  for (const eps of [0, 1e-12, 1e-6, 1e-3]) {
    resetStore();
    clearToasts();
    const pts = [{ x: 10, y: 10 }, { x: 40, y: 10 }, { x: 40, y: 40 }, { x: 10 + eps, y: 10 }];
    const o = svgOpen('a', [pts]);
    const { before, after } = closeAll([o], [L], ['a']);
    const obj = after.scene.objects[0] as ImportedSvg;
    const p = obj.paths[0]!.polylines[0]!;
    const last = p.points.at(-1)!;
    const prev = p.points.at(-2)!;
    out(`S8 eps ${eps}: toast ${toasts().join(' // ')}; ${polyInfo(obj)}; last seg ${Math.hypot(last.x - prev.x, last.y - prev.y)}`);
    const a = report(`S8 eps ${eps} before`, before);
    const b = report(`S8 eps ${eps} after`, after);
    out(`S8 eps ${eps} delta ${f3(b - a)}`);
  }
});

it('S9 locked object, mixed selection, many objects, degenerate inputs', () => {
  const L = layer('L', 'line', 1000, 50);
  const locked = svgOpen('lk', [[{ x: 0, y: 0 }, { x: 10, y: 0 }, { x: 10, y: 10 }]], {}, { locked: true });
  const free = svgOpen('fr', [[{ x: 20, y: 0 }, { x: 30, y: 0 }, { x: 30, y: 10 }]]);
  closeAll([locked, free], [L], ['lk', 'fr']);
  out(`S9a locked+free toast ${toasts().join(' // ')}; locked ${polyInfo(st().project.scene.objects[0])} free ${polyInfo(st().project.scene.objects[1])}`);
  resetStore(); clearToasts();
  closeAll([locked], [L], ['lk']);
  out(`S9b locked only toast ${toasts().join(' // ')}`);
  // degenerate: NaN point, two-point line, single point, empty
  const t0 = performance.now();
  const r = closeOpenPaths(
    [{ color: '#000000', polylines: [
      { points: [{ x: 0, y: 0 }, { x: NaN, y: 1 }, { x: 2, y: 2 }], closed: false },
      { points: [{ x: 0, y: 0 }, { x: 5, y: 0 }], closed: false },
      { points: [{ x: 0, y: 0 }], closed: false },
      { points: [], closed: false },
      { points: [{ x: 0, y: 0 }, { x: 0, y: 0 }, { x: 0, y: 0 }], closed: false },
    ] }],
    IDENTITY_TRANSFORM,
  );
  out(`S9c degenerate: closed ${r.closed} gap ${r.longestGapMm} ${r.paths[0]!.polylines.map((l) => `${l.closed ? 'C' : 'O'}${l.points.length}`).join(' ')} (${f3(performance.now() - t0)} ms)`);
  // many
  const many: SceneObject[] = [];
  for (let i = 0; i < 400; i++) many.push(svgOpen(`m${i}`, [Array.from({ length: 50 }, (_, k) => ({ x: (i % 20) * 10 + Math.cos(k / 8) * 4, y: Math.floor(i / 20) * 10 + Math.sin(k / 8) * 4 }))]));
  const t1 = performance.now();
  closeAll(many, [L], many.map((o) => o.id));
  out(`S9d 400 objects x 50 pts: ${f3(performance.now() - t1)} ms; toast ${toasts().join(' // ')}`);
});
