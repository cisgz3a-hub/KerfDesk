// @vitest-environment jsdom
// T08 dialog: validation of typed tolerances, what the counts say, submit path (checkedSubmit),
// Cancel/Escape.
import { act } from 'react';
import { createRoot } from 'react-dom/client';
import { it, vi } from 'vitest';
import { writeFileSync } from 'node:fs';
import { createLayer, createProject, IDENTITY_TRANSFORM, type ImportedSvg, type Project } from '../t06-main/src/core/scene';
import { CloseOpenFillContoursDialog } from '../t06-main/src/ui/commands/CloseOpenFillContoursDialog';

(globalThis as typeof globalThis & { IS_REACT_ACT_ENVIRONMENT?: boolean }).IS_REACT_ACT_ENVIRONMENT = true;

const log: string[] = [];
const out = (s: string) => {
  log.push(s);
  writeFileSync('C:/Users/stolk/kd-t1/zz-t08/t08-dialog.log', log.join('\n'));
};

function obj(id: string, last: { x: number; y: number }): ImportedSvg {
  return {
    kind: 'imported-svg',
    id,
    source: `${id}.svg`,
    bounds: { minX: 0, minY: 0, maxX: 10, maxY: 10 },
    transform: IDENTITY_TRANSFORM,
    paths: [{ color: '#000000', polylines: [{ closed: false, points: [{ x: 0, y: 0 }, { x: 10, y: 0 }, { x: 10, y: 10 }, last] }] }],
  };
}

function project(): Project {
  return {
    ...createProject(),
    scene: {
      layers: [createLayer({ id: '#000000', color: '#000000', mode: 'fill' })],
      objects: [obj('near', { x: 0, y: 0.3 }), obj('mid', { x: 0, y: 0.75 }), obj('far', { x: 0, y: 3 })],
      groups: [],
    },
  };
}

function setValue(input: HTMLInputElement, value: string) {
  Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value')!.set!.call(input, value);
  input.dispatchEvent(new Event('input', { bubbles: true }));
}

it('dialog validation sweep', async () => {
  for (const typed of ['1', '0', '-1', '', 'abc', '0,5', '0.5in', 'Infinity', '1e308', '0.2', '0.5', '0.75', '1.25', '3']) {
    const host = document.createElement('div');
    document.body.appendChild(host);
    const onApply = vi.fn();
    const onCancel = vi.fn();
    const root = createRoot(host);
    await act(async () => {
      root.render(
        <CloseOpenFillContoursDialog
          project={project()}
          selectedObjectId="near"
          additionalSelectedIds={new Set(['mid', 'far'])}
          onCancel={onCancel}
          onApply={onApply}
        />,
      );
    });
    const input = host.ownerDocument.querySelector('input') as HTMLInputElement;
    await act(async () => setValue(input, typed));
    const apply = [...host.ownerDocument.querySelectorAll('button')].find((b) => b.textContent?.trim() === 'Apply Close')!;
    const form = host.ownerDocument.querySelector('form')!;
    const reported: string[] = [];
    input.reportValidity = () => {
      reported.push(input.validationMessage || JSON.stringify(input.validity));
      return false;
    };
    const body = host.ownerDocument.querySelector('.lf-dialog-body')?.textContent ?? '';
    await act(async () => {
      form.dispatchEvent(new Event('submit', { bubbles: true, cancelable: true }));
    });
    out(
      `typed ${JSON.stringify(typed)} -> input.value ${JSON.stringify(input.value)} | Apply disabled=${apply.disabled} | rangeUnderflow=${input.validity.rangeUnderflow} | submit -> onApply ${JSON.stringify(onApply.mock.calls)} reported=${JSON.stringify(reported)} | body: ${body}`,
    );
    // Escape / Cancel
    await act(async () => {
      [...host.ownerDocument.querySelectorAll('button')].find((b) => b.textContent?.trim() === 'Cancel')!.click();
    });
    if (typed === '1') {
      await act(async () => {
        host.ownerDocument.dispatchEvent(new KeyboardEvent('keydown', { key: 'Escape', bubbles: true }));
      });
      out(`Cancel/Escape -> onCancel calls ${onCancel.mock.calls.length}, onApply calls ${onApply.mock.calls.length}`);
    }
    await act(async () => root.unmount());
    host.remove();
  }
});
