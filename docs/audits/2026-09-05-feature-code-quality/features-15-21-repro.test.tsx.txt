const mounted: Root[] = [];
async function render(element: JSX.Element): Promise<HTMLDivElement> {
  const host = document.createElement('div');
  document.body.appendChild(host);
  const root = createRoot(host);
  mounted.push(root);
  await act(async () => root.render(element));
  return host;
}
function input(host: HTMLElement, label: string): HTMLInputElement {
  const found = host.querySelector(`input[aria-label="${label}"]`);
  if (!(found instanceof HTMLInputElement)) throw new Error(`Missing ${label}`);
  return found;
}
async function setInput(host: HTMLElement, label: string, value: string): Promise<void> {
  const element = input(host, label);
  await act(async () => { element.value = value; Simulate.change(element); });
}
async function click(host: HTMLElement, text: string): Promise<void> {
  const button = [...host.querySelectorAll('button')].find((item) => item.textContent?.trim() === text);
  if (button === undefined) throw new Error(`Missing button ${text}`);
  await act(async () => button.click());
}

  it('blank interval speed is browser-valid and generates a zero-speed job that cannot emit', async () => {
    const generated = vi.fn();
    const host = await render(<IntervalTestDialog onCancel={() => undefined} onGenerate={generated} maxFeedMmPerMin={3000} />);
    await setInput(host, 'Speed', '');
    expect(input(host, 'Speed').checkValidity()).toBe(true);
    await click(host, 'Generate');
    expect(generated).toHaveBeenCalledOnce();
    const options = generated.mock.calls[0]?.[0];
    expect(options.speed).toBe(0);
    const grid = generateIntervalTestGrid(options);
    expect(grid.cells[0]?.effectiveSpeed).toBe(0);
    expect(() => grblStrategy.emit(compileJob(grid.scene, DEFAULT_DEVICE_PROFILE), DEFAULT_DEVICE_PROFILE)).toThrow(/speed must be finite and > 0/);
    // Required invariant: incomplete speed must not replace the scene or claim successful generation.
  });

  it('blank CNC fit-test relief diameter is accepted and generates unrelieved coupons', async () => {
    const generated = vi.fn();
    const host = await render(<BoxFitTestDialog machine={{ kind: 'cnc', stockThicknessMm: 3, toolDiameterMm: 3.175 }} onCancel={() => undefined} onGenerate={generated} />);
    await setInput(host, 'Relief tool diameter', '');
    expect(input(host, 'Relief tool diameter').checkValidity()).toBe(true);
    await click(host, 'Generate');
    expect(generated).toHaveBeenCalledOnce();
    const specification = { thicknessMm: 3, fingerWidthMm: 9, startClearanceMm: .05, stepClearanceMm: .05, rungCount: 6 };
    const zero = generateFitCoupon({ ...specification, relief: { kind: 'corner-overcut', toolDiameterMm: 0 } });
    const plain = generateFitCoupon({ ...specification, relief: { kind: 'none' } });
    expect(zero.kind).toBe('generated');
    expect(plain.kind).toBe('generated');
    if (zero.kind !== 'generated' || plain.kind !== 'generated') throw new Error('Expected current generated behavior');
    // Clipper normalizes closure/order and rounds nominal coordinates. A zero
    // radius leaves the original orthogonal comb/slots with no relief arcs.
    const lengths = (parts: typeof zero.parts) => parts.map((part) => part.rings.outline.points.length);
    expect(lengths(zero.parts)).toEqual(lengths(plain.parts));
    for (const part of zero.parts) {
      const points = part.rings.outline.points;
      for (let i = 1; i < points.length; i += 1) {
        const previous = points[i - 1]!;
        const current = points[i]!;
        expect(previous.x === current.x || previous.y === current.y).toBe(true);
      }
    }
    const relieved = generateFitCoupon({ ...specification, relief: { kind: 'corner-overcut', toolDiameterMm: 3.175 } });
    if (relieved.kind !== 'generated') throw new Error('Expected valid positive relief');
    expect(relieved.parts[1]!.rings.outline.points.length).toBeGreaterThan(zero.parts[1]!.rings.outline.points.length);
    // Required invariant: CNC relief requires a finite positive diameter before any coupon is generated.
  });
