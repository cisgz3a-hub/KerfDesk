const circle = (cx: number, cy: number, r: number, n = 96): Polyline => ({
  closed: true,
  points: Array.from({ length: n }, (_, i) => ({
    x: cx + r * Math.cos((2 * Math.PI * i) / n),
    y: cy + r * Math.sin((2 * Math.PI * i) / n),
  })),
});
const rect = (x0: number, y0: number, x1: number, y1: number): Polyline => ({
  closed: true,
  points: [
    { x: x0, y: y0 },
    { x: x1, y: y0 },
    { x: x1, y: y1 },
    { x: x0, y: y1 },
  ],
});

type Case = {

  readonly shapes: ReadonlyArray<Polyline>;
  readonly probes: ReadonlyArray<{ label: string; loop: number }>;

describe('audit tab swallow on mixed layers', () => {
  it.each(CASES)(
    '$name',
    (c) => {
      const layer = {
        ...createLayer({ id: 'L1', color: '#ff0000' }),
        cnc: { ...DEFAULT_CNC_LAYER_SETTINGS, ...c.patch },
      };
      const machine = {
        ...DEFAULT_CNC_MACHINE_CONFIG,
        stock: { ...DEFAULT_CNC_MACHINE_CONFIG.stock, thicknessMm: c.stockMm ?? 6.35 },
      };
      const project: Project = {
        ...createProject(),
        machine,
        scene: {
          objects: [
            {
              kind: 'imported-svg',
              id: 'o',
              source: 'o.svg',
              bounds: { minX: 0, minY: 0, maxX: 120, maxY: 100 },
              transform: IDENTITY_TRANSFORM,
              paths: [{ color: '#ff0000', polylines: c.shapes }],
            },
          ],
          layers: [layer],
        },
      };
      const job = compileCncJob(project.scene, project.device, machine);
      const program = cncGrblStrategy.emit(job, project.device);
      writeFileSync(`${OUT}/t-${c.name}.nc`, program);
      const warnings = detectMachineJobWarnings(project);
      const parsed = parseProgram(program);
      const profiles = new Map<string, Profile>();
      for (const tool of machine.tools) profiles.set(tool.id, profileFor(tool as OracleTool));
      const loops = c.shapes.map((p) => p.points.map((q) => toMachineCoords(q, project.device)));
      const xs = loops.flatMap((l) => l.map((p) => p.x));
      const ys = loops.flatMap((l) => l.map((p) => p.y));
      const field = emptyField(Math.min(...xs) - 6, Math.min(...ys) - 6, Math.max(...xs) + 6, Math.max(...ys) + 6, 0.1);
      stampMoves(field, parsed.moves, profiles);
      const outside = c.patch.cutType === 'profile-outside';
      const results = c.probes.map((probe) => {
        const loop = loops[probe.loop] ?? [];
        // Waste side of this loop: for an outside profile the plate's waste is outside loop 0
        // and a hole's waste is inside the hole loop; for an inside profile it is inside.
        const isHole = outside ? probe.loop > 0 && c.name.startsWith('plate') : true;
        let cells = 0;
        let deepest = 0;
        const hist: Record<string, number> = {};
        for (let j = 0; j < field.ny; j += 1) {
          const y = field.y0 + (j + 0.5) * field.cell;
          for (let i = 0; i < field.nx; i += 1) {
            const x = field.x0 + (i + 0.5) * field.cell;
            const d = distanceToLoops(x, y, [loop]);
            if (d > 3 || d < 0.2) continue;
            const inLoop = insideEvenOdd(x, y, [loop]);
            if (isHole ? !inLoop : inLoop) continue;
            const got = field.depth[j * field.nx + i] ?? 0;
            if (got <= 0) continue;
            cells += 1;
            deepest = Math.max(deepest, got);
            const key = got.toFixed(2);
            hist[key] = (hist[key] ?? 0) + 1;
          }
        }
        return { label: probe.label, cells, deepest, hist };
      });
      writeFileSync(
        `${OUT}/t-${c.name}.json`,
        JSON.stringify({ name: c.name, warnings, results }, null, 1),
      );
    },
    600_000,
  );
});

import { detectCompiledVCarveDepthWarningsForJob } from './cnc-compiled-depth-warnings';

describe('audit v-carve at the stock bottom', () => {
  it.each([
    { name: 'vb90-default-stock', toolId: 'vb-90', stock: 6.35, w: 20 },
    { name: 'vb90-stock-6.0', toolId: 'vb-90', stock: 6.0, w: 20 },
    { name: 'vb90-stock-6.5', toolId: 'vb-90', stock: 6.5, w: 20 },
    { name: 'vb60-default-stock', toolId: 'vb-60', stock: 6.35, w: 20 },
  ])('$name', (c) => {
    const layer = {
      ...createLayer({ id: 'L1', color: '#ff0000' }),
      cnc: { ...DEFAULT_CNC_LAYER_SETTINGS, cutType: 'v-carve' as const, toolId: c.toolId },
    };
    const machine = {
      ...DEFAULT_CNC_MACHINE_CONFIG,
      stock: { ...DEFAULT_CNC_MACHINE_CONFIG.stock, thicknessMm: c.stock },
    };
    const project: Project = {
      ...createProject(),
      machine,
      scene: {
        objects: [
          {
            kind: 'imported-svg',
            id: 'o',
            source: 'o.svg',
            bounds: { minX: 0, minY: 0, maxX: 120, maxY: 100 },
            transform: IDENTITY_TRANSFORM,
            paths: [{ color: '#ff0000', polylines: [rect(20, 20, 60, 20 + c.w)] }],
          },
        ],
        layers: [layer],
      },
    };
    const job = compileCncJob(project.scene, project.device, machine);
    const program = cncGrblStrategy.emit(job, project.device);
    const zs = [...program.matchAll(/Z(-\d+\.\d+)/g)].map((m) => Number(m[1]));
    writeFileSync(
      `${OUT}/s-${c.name}.json`,
      JSON.stringify(
        {
          name: c.name,
          deepestZ: Math.min(...zs),
          stock: c.stock,
          depthWarnings: detectCompiledVCarveDepthWarningsForJob(project, job),
          jobWarnings: detectMachineJobWarnings(project),
        },
        null,
        1,
      ),
    );
  });
});
