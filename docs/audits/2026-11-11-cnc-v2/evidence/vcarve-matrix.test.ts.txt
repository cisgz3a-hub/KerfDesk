const poly = (pts: ReadonlyArray<readonly [number, number]>): Polyline => ({
  closed: true,
  points: pts.map(([x, y]) => ({ x, y })),
});
const rect = (x0: number, y0: number, x1: number, y1: number): Polyline =>
  poly([
    [x0, y0],
    [x1, y0],
    [x1, y1],
    [x0, y1],
  ]);
const ngon = (cx: number, cy: number, rx: number, ry: number, n: number): Polyline =>
  poly(
    Array.from({ length: n }, (_, i) => [
      cx + rx * Math.cos((2 * Math.PI * i) / n),
      cy + ry * Math.sin((2 * Math.PI * i) / n),
    ]),
  );

function run(c: Case) {
  const shape = SHAPES[c.shape] ?? [];
  const scene: Scene = {
    objects: [
      {
        kind: 'imported-svg',
        id: 'source',
        source: 'source.svg',
        bounds: { minX: 0, minY: 0, maxX: 60, maxY: 60 },
        transform: IDENTITY_TRANSFORM,
        paths: [{ color: '#ff0000', polylines: shape }],
      },
    ],
    layers: [
      {
        ...createLayer({ id: 'carve', color: '#ff0000' }),
        cnc: { ...DEFAULT_CNC_LAYER_SETTINGS, cutType: 'v-carve', toolId: c.tool, ...c.patch },
      },
    ],
  };
  const started = Date.now();
  const job = compileCncJob(scene, DEFAULT_DEVICE_PROFILE, {
    ...DEFAULT_CNC_MACHINE_CONFIG,
    tools: TOOLS,
    toolId: c.tool,
  });
  const program = cncGrblStrategy.emit(job, DEFAULT_DEVICE_PROFILE);
  const compileMs = Date.now() - started;
  console.log(c.name, "compiled", compileMs, "ms", program.length, "bytes");
  writeFileSync(`${OUT}/v-${c.name}.nc`, program);
  const parsed = parseProgram(program);
  const profiles = new Map<string, Profile>();
  for (const tool of TOOLS) profiles.set(tool.id, profileFor(tool as OracleTool));
  const loops: Loop[] = shape.map((p) => p.points.map((q) => toMachineCoords(q, DEFAULT_DEVICE_PROFILE)));
  const xs = loops.flatMap((l) => l.map((p) => p.x));
  const ys = loops.flatMap((l) => l.map((p) => p.y));
  const field = emptyField(Math.min(...xs) - 2, Math.min(...ys) - 2, Math.max(...xs) + 2, Math.max(...ys) + 2, 0.05);
  stampMoves(field, parsed.moves, profiles);
  const vTool = TOOLS.find((t) => t.id === c.tool);
  if (vTool === undefined) throw new Error('tool');
  const tanHalf = Math.tan((((vTool.tipAngleDeg ?? 0) / 2) * Math.PI) / 180);
  const r0 = vTool.kind === 'engraving' ? (vTool.tipDiameterMm ?? 0) / 2 : 0;
  const flank = (vTool.diameterMm / 2 - r0) / tanHalf;
  const capMm = Math.min(c.patch.vCarveFlatDepthEnabled === true ? (c.patch.depthMm ?? 1) : Infinity, flank);
  const target = (x: number, y: number): number => {
    if (!insideEvenOdd(x, y, loops)) return 0;
    const d = distanceToLoops(x, y, loops);
    // Wall law: the edge of a tip flat traces the same wall as a point, d / tan.
    return Math.min(capMm, d / tanHalf);
  };
  const compare = compareField(field, target, 0.02, 0.15);
  const motion = auditMotion(parsed, DEFAULT_CNC_MACHINE_CONFIG.params.safeZMm);
  const groups = job.groups.map((g) =>
    g.kind === 'cnc'
      ? {
          cutType: g.cutType,
          toolId: g.toolId,
          passes: g.passes.length,
          feed: g.feedMmPerMin,
          plunge: g.plungeMmPerMin,
          depth: g.depthMm,
        }
      : { kind: g.kind },
  );
  return {
    name: c.name,
    capMm,
    flank,
    compileMs,
    lines: program.split('\n').length,
    compare,
    motion,
    notes: parsed.notes.slice(0, 12),
    toolOrder: parsed.toolOrder,
    groups,
    warnings: (job as unknown as { warnings?: unknown }).warnings ?? null,
  };
}

describe('audit v-carve matrix', () => {
  it.each(CASES)(
    '$name',
    (c) => {
      const result = run(c);
      writeFileSync(`${OUT}/v-${c.name}.json`, JSON.stringify(result, null, 1));
    },
    600_000,
  );
});
